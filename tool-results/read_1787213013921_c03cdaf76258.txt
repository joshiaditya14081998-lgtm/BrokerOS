   213→// ─────────────────────────────────────────────────────────────────────────────
   214→// ScoreRing — circular SVG progress for the supplier reliability score.
   215→// ─────────────────────────────────────────────────────────────────────────────
   216→function ScoreRing({ score, size = 48 }: { score: number; size?: number }) {
   217→  const radius = (size - 8) / 2;
   218→  const circumference = 2 * Math.PI * radius;
   219→  const offset = circumference - (score / 100) * circumference;
   220→  const color = score >= 85 ? COLORS.emerald
   221→    : score >= 70 ? COLORS.teal
   222→    : score >= 55 ? COLORS.amber
   223→    : COLORS.rose;
   224→  return (
   225→    <div className="relative" style={{ width: size, height: size }}>
   226→      <svg width={size} height={size} className="-rotate-90">
   227→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="oklch(0.7 0.02 160 / 0.2)" strokeWidth={4} />
   228→        <circle
   229→          cx={size / 2}
   230→          cy={size / 2}
   231→          r={radius}
   232→          fill="none"
   233→          stroke={color}
   234→          strokeWidth={4}
   235→          strokeDasharray={circumference}
   236→          strokeDashoffset={offset}
   237→          strokeLinecap="round"
   238→          className="transition-all duration-500"
   239→        />
   240→      </svg>
   241→      <span className="absolute inset-0 grid place-items-center text-xs font-semibold kpi-num" style={{ color }}>
   242→        {score}
   243→      </span>
   244→    </div>
   245→  );
   246→}

   350→// ─────────────────────────────────────────────────────────────────────────────
   351→// TagInsightsCard — top-of-view summary card shown when a tag filter is
   352→// active. Surfaces tag name + color, entity counts, total business value,
   353→// average-metric comparison, and a tagged-vs-all share bar.
   354→// ─────────────────────────────────────────────────────────────────────────────
   355→function TagInsightsCard({ appliedFilter, insights }: { appliedFilter: AppliedFilter; insights: TagInsights }) {
   356→  const swatchClass = TAG_SWATCH_CLASS[appliedFilter.tagColor] ?? TAG_SWATCH_CLASS.emerald;
   357→  const colorClass = TAG_COLOR_CLASS[appliedFilter.tagColor] ?? TAG_COLOR_CLASS.emerald;
   358→
   359→  // Build the comparison "share of total" bars for the included entity types.
   360→  type ShareRow = {
   361→    label: string;
   362→    tagged: number;
   363→    total: number;
   364→    icon: React.ReactNode;
   365→  };
   366→  const shareRows: ShareRow[] = [];
   367→  if (insights.client) {
   368→    shareRows.push({
   369→      label: "Clients",
   370→      tagged: insights.client.taggedCount,
   371→      total: insights.client.totalCount,
   372→      icon: <Users className="size-3.5" />,
   373→    });
   374→  }
   375→  if (insights.supplier) {
   376→    shareRows.push({
   377→      label: "Suppliers",
   378→      tagged: insights.supplier.taggedCount,
   379→      total: insights.supplier.totalCount,
   380→      icon: <Factory className="size-3.5" />,
   381→    });
   382→  }
   383→  if (insights.purchaseOrder) {
   384→    shareRows.push({
   385→      label: "Purchase Orders",
   386→      tagged: insights.purchaseOrder.taggedCount,
   387→      total: insights.purchaseOrder.totalCount,
   388→      icon: <BarChart3 className="size-3.5" />,
   389→    });
   390→  }
   391→
   392→  // Average-metric comparison insights (one per included entity type).
   393→  type InsightRow = {
   394→    label: string;
   395→    taggedValue: string;
   396→    allValue: string;
   397→    delta: number; // % difference vs all
   398→  };
   399→  const insightRows: InsightRow[] = [];
   400→  if (insights.client) {
   401→    const delta = insights.client.allAvgOutstanding > 0
   402→      ? Math.round(((insights.client.taggedAvgOutstanding - insights.client.allAvgOutstanding) / insights.client.allAvgOutstanding) * 100)
   403→      : 0;
   404→    insightRows.push({
   405→      label: "Avg outstanding / client",
   406→      taggedValue: formatCurrency(insights.client.taggedAvgOutstanding, { compact: true }),
   407→      allValue: formatCurrency(insights.client.allAvgOutstanding, { compact: true }),
   408→      delta,
   409→    });
   410→  }
   411→  if (insights.supplier) {
   412→    const delta = insights.supplier.allFulfillment > 0
   413→      ? insights.supplier.taggedFulfillment - insights.supplier.allFulfillment
   414→      : 0;
   415→    insightRows.push({
   416→      label: "Avg fulfillment / supplier",
   417→      taggedValue: `${insights.supplier.taggedFulfillment}%`,
   418→      allValue: `${insights.supplier.allFulfillment}%`,
   419→      delta,
   420→    });
   421→  }
   422→  if (insights.purchaseOrder) {
   423→    const avgTagged = insights.purchaseOrder.taggedCount > 0
   424→      ? insights.purchaseOrder.taggedValue / insights.purchaseOrder.taggedCount
   425→      : 0;
   426→    const avgAll = insights.purchaseOrder.totalCount > 0
   427→      ? insights.purchaseOrder.allValue / insights.purchaseOrder.totalCount
   428→      : 0;
   429→    const delta = avgAll > 0 ? Math.round(((avgTagged - avgAll) / avgAll) * 100) : 0;
   430→    insightRows.push({
   431→      label: "Avg PO value",
   432→      taggedValue: formatCurrency(Math.round(avgTagged), { compact: true }),
   433→      allValue: formatCurrency(Math.round(avgAll), { compact: true }),
   434→      delta,
   435→    });
   436→  }
   437→
   438→  return (
   439→    <GlassCard className="relative overflow-hidden p-5">
   440→      <div className="pointer-events-none absolute inset-0 bg-gradient-to-br from-emerald-500/5 to-transparent" />
   441→      <div className="relative">
   442→        <div className="flex flex-wrap items-start justify-between gap-3">
   443→          <div className="flex items-center gap-3">
   444→            <div className={cn("grid size-10 place-items-center rounded-xl border", colorClass)}>
   445→              <TagIcon className="size-5" />
   446→            </div>
   447→            <div>
   448→              <div className="flex items-center gap-2">
   449→                <h3 className="text-lg font-semibold tracking-tight text-foreground">
   450→                  {appliedFilter.tagName}
   451→                </h3>
   452→                <span className={cn("size-2.5 rounded-full", swatchClass)} aria-hidden />
   453→              </div>
   454→              <p className="text-xs text-muted-foreground">
   455→                {appliedFilter.entityType === null
   456→                  ? "Filter applied across Clients, Suppliers & Purchase Orders"
   457→                  : `Filter scoped to ${appliedFilter.entityType === "PurchaseOrder" ? "Purchase Orders" : `${appliedFilter.entityType}s`}`}
   458→              </p>
   459→            </div>
   460→          </div>
   461→          <Badge variant="outline" className="border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300">
   462→            <PieChart className="size-3" />
   463→            Tag insights
   464→          </Badge>
   465→        </div>
   466→
   467→        {/* Share-of-total bars */}
   468→        <div className="mt-5 grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   469→          {shareRows.map((row) => {
   470→            const pct = row.total > 0 ? Math.round((row.tagged / row.total) * 100) : 0;
   471→            return (
   472→              <div key={row.label} className="rounded-xl border border-border/50 bg-card/40 p-3">
   473→                <div className="flex items-center justify-between gap-2">
   474→                  <span className="inline-flex items-center gap-1.5 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   475→                    {row.icon}
   476→                    {row.label}
   477→                  </span>
   478→                  <span className="kpi-num text-xs text-muted-foreground">
   479→                    {row.tagged} of {row.total}
   480→                  </span>
   481→                </div>
   482→                <div className="mt-2 flex items-center gap-2">
   483→                  <div className="h-2 flex-1 overflow-hidden rounded-full bg-muted">
   484→                    <div
   485→                      className="h-full rounded-full bg-emerald-500 transition-all duration-500"
   486→                      style={{ width: `${Math.max(2, pct)}%` }}
   487→                    />
   488→                  </div>
   489→                  <span className="kpi-num w-10 text-right text-xs font-medium text-foreground">{pct}%</span>
   490→                </div>
   491→              </div>
   492→            );
   493→          })}
   494→        </div>
   495→
   496→        {/* Tagged vs all comparison metrics */}
   497→        <div className="mt-4">
   498→          <p className="mb-2 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   499→            Tagged vs. all entities
   500→          </p>
   501→          <div className="overflow-x-auto rounded-xl border border-border/50">
   502→            <Table>
   503→              <TableHeader>
   504→                <TableRow>
   505→                  <TableHead>Metric</TableHead>
   506→                  <TableHead className="text-right">Tagged</TableHead>
   507→                  <TableHead className="text-right">All</TableHead>
   508→                  <TableHead className="text-right">Δ vs. all</TableHead>
   509→                </TableRow>
   510→              </TableHeader>
   511→              <TableBody>
   512→                {insightRows.map((row) => {
   513→                  const positive = row.delta > 0;
   514→                  const neutral = row.delta === 0;
   515→                  const tone = neutral
   516→                    ? "text-muted-foreground"
   517→                    : positive
   518→                      ? "text-emerald-600 dark:text-emerald-400"
   519→                      : "text-rose-600 dark:text-rose-400";
   520→                  return (
   521→                    <TableRow key={row.label} className="hover-lift">

   542→// ─────────────────────────────────────────────────────────────────────────────
   543→// TagComparisonChart — grouped BarChart comparing Tagged vs All across the
   544→// key metrics for each included entity type.
   545→// ─────────────────────────────────────────────────────────────────────────────
   546→function TagComparisonChart({ insights, tagColor }: { insights: TagInsights; tagColor: string }) {
   547→  // Build chart data — one row per entity-type/metric.
   548→  type ChartRow = { metric: string; tagged: number; all: number };
   549→  const rows: ChartRow[] = [];
   550→  if (insights.client) {
   551→    rows.push({ metric: "Client · Business", tagged: insights.client.taggedBusiness, all: insights.client.allBusiness });
   552→    rows.push({ metric: "Client · Outstanding", tagged: insights.client.taggedOutstanding, all: insights.client.allOutstanding });
   553→    rows.push({ metric: "Client · Brokerage", tagged: insights.client.taggedBrokerage, all: insights.client.allBrokerage });
   554→  }
   555→  if (insights.supplier) {
   556→    rows.push({ metric: "Supplier · Supplied", tagged: insights.supplier.taggedSupplied, all: insights.supplier.allSupplied });
   557→    // Fulfillment is a % — scale to ₹10k-per-% so it shows up on the same axis
   558→    // (we surface the raw % in the tooltip via formatter).
   559→    rows.push({
   560→      metric: "Supplier · Fulfillment",
   561→      tagged: insights.supplier.taggedFulfillment * 1000,
   562→      all: insights.supplier.allFulfillment * 1000,
   563→    });
   564→  }
   565→  if (insights.purchaseOrder) {
   566→    rows.push({ metric: "PO · Value", tagged: insights.purchaseOrder.taggedValue, all: insights.purchaseOrder.allValue });
   567→    rows.push({
   568→      metric: "PO · Fulfillment",
   569→      tagged: insights.purchaseOrder.taggedFulfillment * 1000,
   570→      all: insights.purchaseOrder.allFulfillment * 1000,
   571→    });
   572→  }
   573→
   574→  if (rows.length === 0) return null;
   575→
   576→  const tagSwatchColor = TAG_SWATCH_CLASS[tagColor] ? COLORS[colorsKeyFor(tagColor)] : COLORS.emerald;
   577→  const tooltipFormatter = (value: number, name: string, item: { payload?: { metric?: string } }) => {
   578→    const metric = item?.payload?.metric ?? "";
   579→    if (metric.includes("Fulfillment")) {
   580→      return [`${Math.round(value / 1000)}%`, name];
   581→    }
   582→    return [formatCurrency(value), name];
   583→  };
   584→
   585→  return (
   586→    <GlassCard className="p-5">
   587→      <SectionHeader
   588→        title="Tagged vs. All — Comparison"
   589→        description="Side-by-side view of how the tagged segment stacks up against the full portfolio"
   590→        action={
   591→          <div className="hidden items-center gap-3 text-xs sm:flex">
   592→            <span className="inline-flex items-center gap-1.5 text-muted-foreground">
   593→              <span className="size-2.5 rounded-full" style={{ background: tagSwatchColor }} />
   594→              Tagged
   595→            </span>
   596→            <span className="inline-flex items-center gap-1.5 text-muted-foreground">
   597→              <span className="size-2.5 rounded-full" style={{ background: COLORS.teal }} />
   598→              All
   599→            </span>
   600→          </div>
   601→        }
   602→      />
   603→      <div className="mt-4 h-72 w-full">
   604→        <ResponsiveContainer width="100%" height="100%">
   605→          <BarChart data={rows} margin={{ top: 4, right: 8, bottom: 0, left: 0 }}>
   606→            <CartesianGrid strokeDasharray="3 3" stroke="oklch(0.7 0.02 160 / 0.2)" vertical={false} />
   607→            <XAxis
   608→              dataKey="metric"
   609→              tick={{ fontSize: 10, fill: "oklch(0.5 0.02 165)" }}
   610→              axisLine={false}
   611→              tickLine={false}
   612→              interval={0}
   613→              angle={-25}
   614→              textAnchor="end"
   615→              height={60}
   616→            />
   617→            <YAxis
   618→              tick={{ fontSize: 11, fill: "oklch(0.5 0.02 165)" }}
   619→              axisLine={false}
   620→              tickLine={false}
   621→              tickFormatter={(v) => formatCurrency(v as number, { compact: true })}
   622→              width={55}
   623→            />
   624→            <Tooltip
   625→              cursor={{ fill: "oklch(0.7 0.02 160 / 0.08)" }}
   626→              contentStyle={{ background: "oklch(0.16 0.012 170 / 0.92)", border: "1px solid oklch(1 0 0 / 0.12)", borderRadius: 12, color: "white", backdropFilter: "blur(12px)" }}
   627→              formatter={(v: number, name: string, item) => tooltipFormatter(v, name, item as { payload?: { metric?: string } })}
   628→            />
   629→            <Legend
   630→              wrapperStyle={{ fontSize: 11 }}
   631→              formatter={(v) => (v === "tagged" ? "Tagged" : "All")}
   632→            />
   633→            <Bar dataKey="tagged" name="tagged" fill={tagSwatchColor} radius={[4, 4, 0, 0]} />
   634→            <Bar dataKey="all" name="all" fill={COLORS.teal} radius={[4, 4, 0, 0]} fillOpacity={0.55} />
   635→          </BarChart>
   636→        </ResponsiveContainer>
   637→      </div>
   638→      <p className="mt-2 text-[11px] text-muted-foreground">
   639→        Fulfillment metrics are scaled for visual comparison — hover for the true percentage.
   640→      </p>
   641→    </GlassCard>
   642→  );
   643→}

   645→// Maps a tag color name (from the shared palette) to the oklch chart color so
   646→// the TagComparisonChart bars match the tag's pill swatch.
   647→function colorsKeyFor(color: string): keyof typeof COLORS {
   648→  if (color === "emerald") return "emerald";
   649→  if (color === "amber") return "amber";
   650→  if (color === "rose") return "rose";
   651→  if (color === "teal") return "teal";
   652→  if (color === "plum") return "plum";
   653→  return "emerald";
   654→}
