   155→   155→function formatBytes(bytes: number): string {
   156→   156→  if (bytes < 1024) return `${bytes} B`;
   157→   157→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   158→   158→  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
   159→   159→}
   160→   160→
   161→   161→function formatRelative(iso: string | null): string {
   162→   162→  if (!iso) return "Never";
   163→   163→  const t = new Date(iso).getTime();
   164→   164→  if (Number.isNaN(t)) return "Never";
   165→   165→  const diff = Date.now() - t;
   166→   166→  const sec = Math.floor(diff / 1000);
   167→   167→  if (sec < 60) return "Just now";
   168→   168→  const min = Math.floor(sec / 60);
   169→   169→  if (min < 60) return `${min} min ago`;
   170→   170→  const hr = Math.floor(min / 60);
   171→   171→  if (hr < 24) return `${hr} hr ago`;
   172→   172→  const day = Math.floor(hr / 24);
   173→   173→  if (day < 30) return `${day} day${day === 1 ? "" : "s"} ago`;
   174→   174→  const mon = Math.floor(day / 30);
   175→   175→  if (mon < 12) return `${mon} month${mon === 1 ? "" : "s"} ago`;
   176→   176→  return new Date(iso).toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   177→   177→}
   178→   178→
   179→   179→function formatDateTime(iso: string | null | undefined): string {
   180→   180→  if (!iso) return "Unknown";
   181→   181→  const d = new Date(iso);
   182→   182→  if (Number.isNaN(d.getTime())) return "Unknown";
   183→   183→  return d.toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" });
   184→   184→}

   186→   186→function parseBackupFile(text: string, fileName: string, fileSize: number): { ok: true; backup: ParsedBackup } | { ok: false; error: string } {
   187→   187→  let json: unknown;
   188→   188→  try {
   189→   189→    json = JSON.parse(text);
   190→   190→  } catch {
   191→   191→    return { ok: false, error: `"${fileName}" is not valid JSON.` };
   192→   192→  }
   193→   193→  if (typeof json !== "object" || json === null || Array.isArray(json)) {
   194→   194→    return { ok: false, error: "Backup file must be a JSON object." };
   195→   195→  }
   196→   196→  const obj = json as Record<string, unknown>;
   197→   197→  if (obj.version !== 1) {
   198→   198→    return { ok: false, error: `Unsupported backup version (expected 1, got ${String(obj.version)}).` };
   199→   199→  }
   200→   200→  if (typeof obj.tables !== "object" || obj.tables === null || Array.isArray(obj.tables)) {
   201→   201→    return { ok: false, error: "Backup file is missing the 'tables' object." };
   202→   202→  }
   203→   203→  const tables = obj.tables as Record<string, unknown>;
   204→   204→  for (const t of BACKUP_TABLE_LABELS) {
   205→   205→    if (!Array.isArray(tables[t.key])) {
   206→   206→      return { ok: false, error: `Backup file is missing the "${t.key}" table.` };
   207→   207→    }
   208→   208→  }
   209→   209→  return {
   210→   210→    ok: true,
   211→   211→    backup: {
   212→   212→      version: 1,
   213→   213→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : undefined,
   214→   214→      tables: tables as Record<string, unknown[]>,
   215→   215→      bytes: fileSize,
   216→   216→    },
   217→   217→  };
   218→   218→}
