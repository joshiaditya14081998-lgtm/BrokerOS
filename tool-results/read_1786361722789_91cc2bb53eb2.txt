   520→   520→function TimelineTab({ po }: { po: PoDetail }) {
   521→   521→  const entries: TimelineEntry[] = React.useMemo(() => {
   522→   522→    const list: TimelineEntry[] = [];
   523→   523→    for (const l of po.dispatchDateLogs) {
   524→   524→      list.push({
   525→   525→        id: l.id, date: l.createdAt, kind: "log",
   526→   526→        title: `Dispatch date revised → ${formatDate(l.newDate)}`,
   527→   527→        detail: l.reason || (l.oldDate ? `from ${formatDate(l.oldDate)}` : null),
   528→   528→      });
   529→   529→    }
   530→   530→    if (po.bill) {
   531→   531→      list.push({
   532→   532→        id: `bill-${po.bill.id}`, date: po.bill.createdAt, kind: "bill",
   533→   533→        title: `Bill ${po.bill.billNumber} generated`,
   534→   534→        detail: `Final: ${formatCurrency(po.bill.finalAmount)} · status: ${titleCase(po.bill.status)}`,
   535→   535→      });
   536→   536→      for (const p of po.bill.payments) {
   537→   537→        list.push({
   538→   538→          id: `pay-${p.id}`, date: p.date, kind: "payment",
   539→   539→          title: `Payment · ${titleCase(p.mode)}`,
   540→   540→          detail: `${formatCurrency(p.amount)}${p.reference ? ` · ref ${p.reference}` : ""}`,
   541→   541→        });
   542→   542→      }
   543→   543→      if (po.bill.brokerage) {
   544→   544→        const b = po.bill.brokerage;
   545→   545→        list.push({
   546→   546→          id: `brok-${b.id}`, date: b.eligibleAt ?? po.bill.createdAt, kind: "bill",
   547→   547→          title: `Brokerage ${b.eligible ? "eligible" : "accrued"}`,
   548→   548→          detail: `${formatCurrency(b.brokerageAmount)} · ${titleCase(b.payoutStatus)}`,
   549→   549→        });
   550→   550→      }
   551→   551→    }
   552→   552→    for (const d of po.disputes) {
   553→   553→      list.push({
   554→   554→        id: `disp-${d.id}`, date: d.createdAt, kind: "dispute",
   555→   555→        title: `Dispute · ${titleCase(d.type)}`,
   556→   556→        detail: d.description || `${d.quantityAffected} units · ${formatCurrency(d.valueAffected)}`,
   557→   557→      });
   558→   558→    }
   559→   559→    return list.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   560→   560→  }, [po]);
   561→   561→
   562→   562→  if (entries.length === 0) {
   563→   563→    return (
   564→   564→      <GlassCard className="p-8">
   565→   565→        <EmptyState title="No timeline events" hint="Dispatches, payments, and disputes will appear here." icon={<Clock className="size-5" />} />
   566→   566→      </GlassCard>
   567→   567→    );
   568→   568→  }
   569→   569→
   570→   570→  return (
   571→   571→    <div className="glass rounded-xl p-4">
   572→   572→      <ol className="relative space-y-4 border-l border-border/60 pl-4">
   573→   573→        {entries.map((e) => (
   574→   574→          <li key={e.id} className="relative">
   575→   575→            <span className={`absolute -left-[1.41rem] top-1 grid size-3.5 place-items-center rounded-full border-2 border-background ${kindClass(e.kind)}`}>
   576→   576→              <span className="size-1.5 rounded-full bg-current opacity-80" />
   577→   577→            </span>
   578→   578→            <div className="flex flex-wrap items-center gap-1.5">
   579→   579→              <span className="text-sm font-medium">{e.title}</span>
   580→   580→              <span className="text-[10px] uppercase tracking-wider text-muted-foreground">
   581→   581→                {kindLabel(e.kind)}
   582→   582→              </span>
   583→   583→            </div>
   584→   584→            <p className="text-xs text-muted-foreground">{formatDateTime(e.date)}</p>
   585→   585→            {e.detail ? <p className="mt-0.5 text-xs text-foreground/80">{e.detail}</p> : null}
   586→   586→          </li>
   587→   587→        ))}
   588→   588→      </ol>
   589→   589→    </div>
   590→   590→  );
   591→   591→}
