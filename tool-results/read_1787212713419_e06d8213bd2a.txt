    37→function fmtCurrency(n: number): string {
    38→  const v = Number.isFinite(n) ? n : 0;
    39→  return new Intl.NumberFormat("en-IN", {
    40→    style: "currency",
    41→    currency: "INR",
    42→    maximumFractionDigits: 0,
    43→  }).format(v);
    44→}

    46→function fmtDate(d: Date | string | null | undefined): string {
    47→  if (!d) return "—";
    48→  const date = typeof d === "string" ? new Date(d) : d;
    49→  return date.toLocaleDateString("en-IN", {
    50→    day: "2-digit",
    51→    month: "short",
    52→    year: "numeric",
    53→  });
    54→}
    55→
    56→function fmtDateTime(d: Date | string | null | undefined): string {
    57→  if (!d) return "—";
    58→  const date = typeof d === "string" ? new Date(d) : d;
    59→  return date.toLocaleString("en-IN", {
    60→    day: "2-digit",
    61→    month: "short",
    62→    year: "numeric",
    63→    hour: "2-digit",
    64→    minute: "2-digit",
    65→  });
    66→}
    67→
    68→function titleCase(s: string): string {
    69→  return s.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase());
    70→}
    71→
    72→function escapeHtml(s: string | null | undefined): string {
    73→  if (s == null) return "";
    74→  return String(s)
    75→    .replace(/&/g, "&amp;")
    76→    .replace(/</g, "&lt;")
    77→    .replace(/>/g, "&gt;")
    78→    .replace(/"/g, "&quot;")
    79→    .replace(/'/g, "&#39;");
    80→}

    82→// ── Range bounds (reused from dashboard route) ───────────────────────────────
    83→
    84→function getRangeBounds(range: Range): { start: Date | undefined; end: Date; label: string } {
    85→  const now = new Date();
    86→  if (range === "all") {
    87→    return { start: undefined, end: now, label: "All time" };
    88→  }
    89→  const y = now.getFullYear();
    90→  const m = now.getMonth();
    91→  let start: Date;
    92→  let label: string;
    93→  if (range === "month") {
    94→    start = new Date(y, m, 1);
    95→    label = start.toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    96→  } else if (range === "quarter") {
    97→    const qStartMonth = Math.floor(m / 3) * 3;
    98→    start = new Date(y, qStartMonth, 1);
    99→    const qNum = Math.floor(qStartMonth / 3) + 1;
   100→    label = `Q${qNum} ${y}`;
   101→  } else {
   102→    start = new Date(y, 0, 1);
   103→    label = `Year ${y}`;
   104→  }
   105→  return { start, end: now, label };
   106→}

   393→// ── Status pill helper ───────────────────────────────────────────────────────

   395→function statusPill(status: string): string {
   396→  const s = status.toLowerCase();
   397→  let cls = "zinc";
   398→  if (["fully_paid", "fully_delivered", "delivered", "resolved", "occurred", "done", "paid", "scheduled"].includes(s)) {
   399→    cls = "emerald";
   400→  } else if (["partially_paid", "partially_delivered", "in_transit", "accrued"].includes(s)) {
   401→    cls = "amber";
   402→  } else if (["pending", "open", "no_show", "short_shipment", "defective_return", "overdue"].includes(s)) {
   403→    cls = "rose";
   404→  } else if (["closed", "rejected", "dismissed"].includes(s)) {
   405→    cls = "zinc";
   406→  }
   407→  return `<span class="pill ${cls}">${escapeHtml(titleCase(status))}</span>`;
   408→}
   409→
   410→function eligiblePill(eligible: boolean, forceEligible?: boolean): string {
   411→  if (!eligible) return `<span class="pill rose">No</span>`;
   412→  return forceEligible
   413→    ? `<span class="pill amber">Forced</span>`
   414→    : `<span class="pill emerald">Yes</span>`;
   415→}

   417→// ── Brokerage statement report ───────────────────────────────────────────────
   418→
   419→async function buildBrokerageStatement(range: Range): Promise<string> {
   420→  const { start, end, label } = getRangeBounds(range);
   421→
   422→  const brokerages = await db.brokerage.findMany({
   423→    where: start
   424→      ? {
   425→          OR: [
   426→            { eligibleAt: { gte: start, lte: end } },
   427→            { eligibleAt: null, createdAt: { gte: start, lte: end } },
   428→          ],
   429→        }
   430→      : undefined,
   431→    include: {
   432→      bill: { select: { billNumber: true, baseAmount: true, status: true, po: { select: { poNumber: true } } } },
   433→      client: { select: { name: true } },
   434→      supplier: { select: { name: true } },
   435→      payout: { select: { id: true, status: true, paidAt: true } },
   436→    },
   437→    orderBy: { eligibleAt: "asc" },
   438→  });
   439→
   440→  const payouts = await db.brokeragePayout.findMany({
   441→    where: start ? { createdAt: { gte: start, lte: end } } : undefined,
   442→    include: { client: { select: { name: true } }, brokerages: { select: { id: true, brokerageAmount: true } } },
   443→    orderBy: { createdAt: "desc" },
   444→  });
   445→
   446→  const totalBrokerage = brokerages.reduce((s, b) => s + b.brokerageAmount, 0);
   447→  const accruedTotal = brokerages
   448→    .filter((b) => b.eligible && b.payoutStatus === "accrued")
   449→    .reduce((s, b) => s + b.brokerageAmount, 0);
   450→  const scheduledTotal = brokerages
   451→    .filter((b) => b.eligible && b.payoutStatus === "scheduled")
   452→    .reduce((s, b) => s + b.brokerageAmount, 0);
   453→  const paidTotal = brokerages
   454→    .filter((b) => b.payoutStatus === "paid")
   455→    .reduce((s, b) => s + b.brokerageAmount, 0);
   456→  const pendingTotal = brokerages
   457→    .filter((b) => !b.eligible)
   458→    .reduce((s, b) => s + b.brokerageAmount, 0);
   459→  const billsCount = new Set(brokerages.map((b) => b.billId)).size;
   460→
   461→  // Group by client for the detail table (multiple sub-totals)
   462→  const byClient = new Map<string, { name: string; rows: typeof brokerages }>();
   463→  for (const b of brokerages) {
   464→    const key = b.clientId;
   465→    if (!byClient.has(key)) {
   466→      byClient.set(key, { name: b.client.name, rows: [] });
   467→    }
   468→    byClient.get(key)!.rows.push(b);
   469→  }
   470→  // Sort clients by name
   471→  const clientGroups = Array.from(byClient.entries())
   472→    .map(([id, g]) => ({ id, name: g.name, rows: g.rows }))
   473→    .sort((a, b) => a.name.localeCompare(b.name));
   474→
   475→  // Build the detail table — grouped by client with sub-totals
   476→  const detailRows: string[] = [];
   477→  let grandBase = 0;
   478→  let grandBrokerage = 0;
   479→  for (const group of clientGroups) {
   480→    detailRows.push(
   481→      `<tr class="group-header"><td colspan="9">${escapeHtml(group.name)} — ${group.rows.length} entr${group.rows.length === 1 ? "y" : "ies"}</td></tr>`,
   482→    );
   483→    let subBase = 0;
   484→    let subBrokerage = 0;
   485→    for (const b of group.rows) {
   486→      subBase += b.baseAmount;
   487→      subBrokerage += b.brokerageAmount;
   488→      grandBase += b.baseAmount;
   489→      grandBrokerage += b.brokerageAmount;
   490→      detailRows.push(
   491→        `<tr>
   492→          <td>${escapeHtml(b.bill.billNumber)}</td>
   493→          <td>${escapeHtml(b.client.name)}</td>
   494→          <td>${escapeHtml(b.supplier.name)}</td>
   495→          <td class="num">${fmtCurrency(b.baseAmount)}</td>
   496→          <td class="num">${b.commissionRate.toFixed(1)}%</td>
   497→          <td class="num emerald">${fmtCurrency(b.brokerageAmount)}</td>
   498→          <td class="center">${eligiblePill(b.eligible, b.forceEligible)}</td>
   499→          <td class="center">${statusPill(b.payoutStatus)}</td>
   500→          <td>${fmtDate(b.eligibleAt)}</td>
   501→        </tr>`,
   502→      );
   503→    }
   504→    detailRows.push(
   505→      `<tr class="sub-total">
   506→        <td colspan="3">Sub-total — ${escapeHtml(group.name)}</td>
   507→        <td class="num">${fmtCurrency(subBase)}</td>
   508→        <td></td>
   509→        <td class="num">${fmtCurrency(subBrokerage)}</td>
   510→        <td colspan="3"></td>
   511→      </tr>`,
   512→    );
   513→  }
   514→

  1136→// Maps an action string to a print-safe pill color (matches UI tone mapping).
  1137→function auditActionPill(action: string): string {
  1138→  const a = action.toLowerCase();
  1139→  if (a === "create" || a === "payout") {
  1140→    return `<span class="pill emerald">${escapeHtml(titleCase(action))}</span>`;
  1141→  }
  1142→  if (a === "update") {
  1143→    return `<span class="pill amber">${escapeHtml(titleCase(action))}</span>`;
  1144→  }
  1145→  if (a === "delete") {
  1146→    return `<span class="pill rose">${escapeHtml(titleCase(action))}</span>`;
  1147→  }
  1148→  if (a === "force_eligible") {
  1149→    return `<span class="pill emerald" style="background:#f0fdfa;color:#0f766e;border-color:#99f6e4;">${escapeHtml(titleCase(action))}</span>`;
  1150→  }
  1151→  return `<span class="pill zinc">${escapeHtml(titleCase(action))}</span>`;
  1152→}

  1154→async function buildAuditTrail(params: URLSearchParams): Promise<string> {
  1155→  const entityType = params.get("entityType")?.trim();
  1156→  const fromRaw = params.get("from");
  1157→  const toRaw = params.get("to");
  1158→
  1159→  // Note: SQLite (Prisma's provider here) does not support `mode: "insensitive"`.
  1160→  // String comparison uses the default BINARY collation (case-sensitive).
  1161→  // Stored entityType values are PascalCase and the UI dropdown uses matching
  1162→  // values, so case-sensitive equals is sufficient.
  1163→  const where: Prisma.AuditLogWhereInput = {};
  1164→  if (entityType) {
  1165→    where.entityType = entityType;
  1166→  }
  1167→  const range: { gte?: Date; lte?: Date } = {};
  1168→  if (fromRaw) {
  1169→    const d = new Date(fromRaw);
  1170→    if (!Number.isNaN(d.getTime())) range.gte = d;
  1171→  }
  1172→  if (toRaw) {
  1173→    const d = new Date(toRaw);
  1174→    if (!Number.isNaN(d.getTime())) {
  1175→      d.setHours(23, 59, 59, 999);
  1176→      range.lte = d;
  1177→    }
  1178→  }
  1179→  if (range.gte || range.lte) {
  1180→    where.createdAt = range;
  1181→  }
  1182→
  1183→  const [logs, byActionAgg, byEntityAgg, minMax] = await Promise.all([
  1184→    db.auditLog.findMany({
  1185→      where,
  1186→      orderBy: { createdAt: "desc" },
  1187→      take: 1000,
  1188→    }),
  1189→    db.auditLog.groupBy({ by: ["action"], where, _count: { _all: true } }),
  1190→    db.auditLog.groupBy({ by: ["entityType"], where, _count: { _all: true } }),
  1191→    db.auditLog.aggregate({
  1192→      where,
  1193→      _min: { createdAt: true },
  1194→      _max: { createdAt: true },
  1195→    }),
  1196→  ]);
  1197→
  1198→  const byAction: Record<string, number> = {};
  1199→  for (const g of byActionAgg) byAction[g.action] = g._count._all;
  1200→  const byEntity: Record<string, number> = {};
  1201→  for (const g of byEntityAgg) byEntity[g.entityType] = g._count._all;
  1202→
  1203→  const total = logs.length;
  1204→  const earliest = minMax._min.createdAt;
  1205→  const latest = minMax._max.createdAt;
  1206→
  1207→  // Build the date-range label shown in the report header.
  1208→  let rangeLabel: string;
  1209→  if (range.gte && range.lte) {
  1210→    rangeLabel = `${fmtDate(range.gte)} → ${fmtDate(range.lte)}`;
  1211→  } else if (range.gte) {
  1212→    rangeLabel = `From ${fmtDate(range.gte)}`;
  1213→  } else if (range.lte) {
  1214→    rangeLabel = `Up to ${fmtDate(range.lte)}`;
  1215→  } else {
  1216→    rangeLabel = "All time";
  1217→  }
  1218→  if (entityType) {
  1219→    rangeLabel += ` · ${entityType} only`;
  1220→  }
  1221→
  1222→  // KPI grid: total + per-action counts
  1223→  const createCount = byAction["create"] ?? 0;
  1224→  const updateCount = byAction["update"] ?? 0;
  1225→  const deleteCount = byAction["delete"] ?? 0;
  1226→  const forceCount = byAction["force_eligible"] ?? 0;
  1227→  const payoutCount = byAction["payout"] ?? 0;
  1228→
  1229→  // Action breakdown list (all actions observed, sorted by count desc)
  1230→  const actionRows = Object.entries(byAction)
  1231→    .sort((a, b) => b[1] - a[1])
  1232→    .map(
  1233→      ([action, n]) =>
  1234→        `<tr><td>${auditActionPill(action)}</td><td class="num">${n}</td><td class="num">${total ? Math.round((n / total) * 100) : 0}%</td></tr>`,
  1235→    )
  1236→    .join("") ||
  1237→    `<tr><td colspan="3" style="text-align:center;padding:24px;color:#a1a1aa;">No entries.</td></tr>`;
  1238→
  1239→  const entityRows = Object.entries(byEntity)
  1240→    .sort((a, b) => b[1] - a[1])
  1241→    .map(
  1242→      ([entity, n]) =>
  1243→        `<tr><td>${escapeHtml(titleCase(entity))}</td><td class="num">${n}</td><td class="num">${total ? Math.round((n / total) * 100) : 0}%</td></tr>`,
  1244→    )
  1245→    .join("") ||
  1246→    `<tr><td colspan="3" style="text-align:center;padding:24px;color:#a1a1aa;">No entries.</td></tr>`;
  1247→
  1248→  // Detail table — page break every 50 rows via a wrapper div with class
  1249→  // `section-break` on every 50th row group. Simpler approach: insert a
  1250→  // page-break row class on every 50th row.
  1251→  const PAGE_SIZE = 50;
  1252→  const detailRows = logs.length
  1253→    ? logs
  1254→        .map((l, i) => {
  1255→          const breakClass = i > 0 && i % PAGE_SIZE === 0 ? "page-break-row" : "";
  1256→          return `<tr class="${breakClass}">
  1257→            <td class="nowrap">${escapeHtml(fmtDateTime(l.createdAt))}</td>
  1258→            <td>${escapeHtml(l.userName ?? "System")}</td>
  1259→            <td>${escapeHtml(titleCase(l.entityType))}</td>
  1260→            <td>${auditActionPill(l.action)}</td>
  1261→            <td>${escapeHtml(l.reason ?? "—")}</td>
  1262→          </tr>`;
  1263→        })
  1264→        .join("")
  1265→    : `<tr><td colspan="5" style="text-align:center;padding:24px;color:#a1a1aa;">No audit entries match these filters.</td></tr>`;
  1266→
  1267→  const body = `
  1268→    <div class="kpi-grid">
  1269→      <div class="kpi-card">
  1270→        <div class="kpi-label">Total entries</div>
  1271→        <div class="kpi-value">${total}</div>
  1272→        <div class="kpi-sub">${range.gte || range.lte ? "In selected range" : "All time"}</div>
  1273→      </div>
  1274→      <div class="kpi-card">
  1275→        <div class="kpi-label">Created</div>
  1276→        <div class="kpi-value emerald">${createCount}</div>
  1277→        <div class="kpi-sub">New records</div>
  1278→      </div>
  1279→      <div class="kpi-card">
  1280→        <div class="kpi-label">Updated</div>
  1281→        <div class="kpi-value amber">${updateCount}</div>
  1282→        <div class="kpi-sub">Field edits</div>
  1283→      </div>
  1284→      <div class="kpi-card">
  1285→        <div class="kpi-label">Deleted</div>
  1286→        <div class="kpi-value rose">${deleteCount}</div>
  1287→        <div class="kpi-sub">Removals</div>
  1288→      </div>
  1289→      <div class="kpi-card">
  1290→        <div class="kpi-label">Force-eligible</div>
  1291→        <div class="kpi-value" style="color:#0f766e;">${forceCount}</div>
  1292→        <div class="kpi-sub">Manual overrides</div>
  1293→      </div>
  1294→      <div class="kpi-card">
  1295→        <div class="kpi-label">Payouts</div>
  1296→        <div class="kpi-value emerald">${payoutCount}</div>
  1297→        <div class="kpi-sub">Brokerage settled</div>

  1299→      <div class="kpi-card">
  1300→        <div class="kpi-label">First entry</div>
  1301→        <div class="kpi-value" style="font-size:13px;">${earliest ? escapeHtml(fmtDate(earliest)) : "—"}</div>
  1302→        <div class="kpi-sub">${earliest ? escapeHtml(fmtDateTime(earliest)) : ""}</div>
  1303→      </div>
  1304→      <div class="kpi-card">
  1305→        <div class="kpi-label">Latest entry</div>

  1357→export async function GET(req: NextRequest) {
  1358→  const { searchParams } = new URL(req.url);
  1359→  const typeParam = (searchParams.get("type") ?? "").toLowerCase() as ReportType;
  1360→  if (!VALID_TYPES.includes(typeParam)) {
  1361→    return NextResponse.json(
  1362→      { error: `Invalid report type. Valid: ${VALID_TYPES.join(", ")}` },
  1363→      { status: 400 },
  1364→    );
  1365→  }
  1366→
  1367→  try {
  1368→    let html: string;
  1369→    if (typeParam === "brokerage-statement") {
  1370→      const rawRange = (searchParams.get("range") ?? "all").toLowerCase() as Range;
  1371→      const range: Range = VALID_RANGES.includes(rawRange) ? rawRange : "all";
  1372→      html = await buildBrokerageStatement(range);
  1373→    } else if (typeParam === "client-ledger") {
  1374→      const clientId = searchParams.get("clientId");
  1375→      if (!clientId) {
  1376→        return NextResponse.json({ error: "clientId is required for client-ledger report" }, { status: 400 });
  1377→      }
  1378→      html = await buildClientLedger(clientId);
  1379→    } else if (typeParam === "supplier-summary") {
  1380→      const supplierId = searchParams.get("supplierId");
  1381→      if (!supplierId) {
  1382→        return NextResponse.json({ error: "supplierId is required for supplier-summary report" }, { status: 400 });
  1383→      }
  1384→      html = await buildSupplierSummary(supplierId);
  1385→    } else {
  1386→      // audit-trail
  1387→      html = await buildAuditTrail(searchParams);
  1388→    }
  1389→
  1390→    return new NextResponse(html, {
  1391→      status: 200,
  1392→      headers: {
  1393→        "Content-Type": "text/html; charset=utf-8",
  1394→        "Cache-Control": "no-store, no-cache, must-revalidate",
  1395→        "X-Content-Type-Options": "nosniff",
  1396→      },
  1397→    });
  1398→  } catch (e) {
  1399→    const message = e instanceof Error ? e.message : "Unknown error";
  1400→    return NextResponse.json({ error: `Report failed: ${message}` }, { status: 500 });
  1401→  }
