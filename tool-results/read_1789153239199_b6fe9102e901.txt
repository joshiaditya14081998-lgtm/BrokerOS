   230→function Section({
   231→  id,
   232→  children,
   233→  className,
   234→}: {
   235→  id?: string;
   236→  children: React.ReactNode;
   237→  className?: string;
   238→}) {
   239→  const reduce = useReducedMotion();
   240→  return (
   241→    <motion.section
   242→      id={id}
   243→      className={cn("scroll-mt-24", className)}
   244→      initial={reduce ? false : { opacity: 0, y: 24 }}
   245→      whileInView={reduce ? undefined : { opacity: 1, y: 0 }}
   246→      viewport={{ once: true, margin: "-80px" }}
   247→      transition={{ duration: 0.6, ease: [0.2, 0.8, 0.2, 1] }}
   248→    >
   249→      {children}
   250→    </motion.section>
   251→  );
   252→}
