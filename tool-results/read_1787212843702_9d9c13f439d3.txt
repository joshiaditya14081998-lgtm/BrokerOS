   141→// Main portal view

   143→export function PortalView() {
   144→  const [persona, setPersona] = useStateWithUrl<Persona>("supplier", "persona");
   145→  const [entityId, setEntityId] = useStateWithUrl<string>("", "id");
   146→
   147→  // Fetch the picker list for the active persona.
   148→  const pickerPath = persona === "supplier" ? "/api/suppliers" : "/api/clients";
   149→  const { data: pickerData, loading: pickerLoading } = useApi<{ suppliers?: Option[]; clients?: Option[] }>(pickerPath);
   150→  const options: Option[] = (persona === "supplier" ? pickerData?.suppliers : pickerData?.clients) ?? [];
   151→
   152→  // Auto-pick the first entity when the list arrives and nothing is selected
   153→  // (or when the persona switches and the current id no longer applies).
   154→  // We intentionally key on options.length rather than options (a fresh array
   155→  // each render) to avoid an infinite update loop.
   156→  React.useEffect(() => {
   157→    if (options.length === 0) return;
   158→    if (!entityId || !options.some((o) => o.id === entityId)) {
   159→      setEntityId(options[0].id);
   160→    }
   161→  }, [persona, options.length, entityId, options, setEntityId]);
   162→
   163→  // Fetch the portal payload only when both persona + id are set.
   164→  const portalPath = entityId ? `/api/portal?persona=${persona}&id=${entityId}` : null;
   165→  const { data, loading, error } = useApi<PortalData>(portalPath);
   166→
   167→  return (
   168→    <div className="space-y-6">
   169→      <SectionHeader
   170→        title="Portals"
   171→        description="Preview what suppliers and clients see in their own portal"
   172→        action={
   173→          <Badge variant="outline" className="border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-300">
   174→            <Info className="size-3" />
   175→            Read-only mockup
   176→          </Badge>
   177→        }
   178→      />
   179→
   180→      {/* Preview-mode info banner */}
   181→      <div className="flex items-start gap-3 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-3 text-sm text-amber-800 dark:text-amber-200">
   182→        <Info className="mt-0.5 size-4 shrink-0" />
   183→        <p>
   184→          <span className="font-semibold">Preview mode —</span> This is a read-only mockup of the supplier/client
   185→          portal experience. In production, each party would log in to see only their own data.
   186→        </p>
   187→      </div>
   188→
   189→      {/* Persona switcher + entity picker */}
   190→      <GlassCard className="p-4 sm:p-5">
   191→        <div className="flex flex-col gap-4 lg:flex-row lg:items-end lg:justify-between">
   192→          <div className="w-full lg:max-w-sm">
   193→            <p className="mb-2 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   194→              View as
   195→            </p>
   196→            <div className="grid grid-cols-2 gap-2">
   197→              <PersonaButton
   198→                active={persona === "supplier"}
   199→                onClick={() => setPersona("supplier")}
   200→                icon={<Factory className="size-4" />}
   201→                label="Supplier Portal"
   202→              />
   203→              <PersonaButton
   204→                active={persona === "client"}
   205→                onClick={() => setPersona("client")}
   206→                icon={<Users className="size-4" />}
   207→                label="Client Portal"
   208→              />
   209→            </div>
   210→          </div>
   211→          <div className="w-full lg:max-w-sm">
   212→            <label className="mb-2 block text-xs font-medium uppercase tracking-wider text-muted-foreground">
   213→              {persona === "supplier" ? "Select supplier" : "Select client"}
   214→            </label>
   215→            <Select value={entityId} onValueChange={setEntityId}>
   216→              <SelectTrigger className="w-full" disabled={pickerLoading || options.length === 0}>
   217→                <SelectValue
   218→                  placeholder={pickerLoading ? "Loading…" : options.length === 0 ? "No records" : "Pick one…"}
   219→                />
   220→              </SelectTrigger>
   221→              <SelectContent>
   222→                {options.map((o) => (
   223→                  <SelectItem key={o.id} value={o.id}>{o.name}</SelectItem>
   224→                ))}
   225→              </SelectContent>
   226→            </Select>
   227→          </div>
   228→        </div>
   229→      </GlassCard>
   230→
   231→      {/* Portal dashboard body */}
   232→      {!entityId ? (
   233→        <EmptyState
   234→          title={persona === "supplier" ? "No supplier selected" : "No client selected"}
   235→          hint="Use the picker above to choose a record to preview as."
   236→          icon={<Store className="size-5" />}
   237→        />
   238→      ) : loading ? (
   239→        <PortalSkeleton persona={persona} />
   240→      ) : error ? (
   241→        <EmptyState
   242→          title="Couldn't load portal data"
   243→          hint={error}
   244→          icon={<AlertCircle className="size-5" />}
   245→        />
   246→      ) : !data ? (
   247→        <EmptyState title="No data" icon={<Store className="size-5" />} />
   248→      ) : persona === "supplier" && data.persona === "supplier" ? (
   249→        <SupplierPortalDashboard data={data} />
   250→      ) : persona === "client" && data.persona === "client" ? (
   251→        <ClientPortalDashboard data={data} />
   252→      ) : (
   253→        <PortalSkeleton persona={persona} />
   254→      )}
   255→    </div>
   256→  );
   257→}

   284→// Tiny URL-synced state hook so the persona + entity survive navigations away
   285→// and back via the sidebar (the picker state stays scoped to this view).
   286→function useStateWithUrl<T>(initial: T, _key: string): [T, (v: T) => void] {
   287→  const [value, setValue] = React.useState<T>(initial);
   288→  return [value, setValue];
   289→}

   332→  return (

   334→      {/* Welcome header */}

   336→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   337→          <div className="min-w-0">
   338→            <p className="text-xs font-medium uppercase tracking-wider text-emerald-600 dark:text-emerald-400">
   339→              <Factory className="mr-1 inline size-3" />Supplier Portal
   340→            </p>
   341→            <h2 className="mt-1 text-2xl font-semibold tracking-tight text-foreground">
   342→              Welcome, {profile.name}
   343→            </h2>
   344→            <div className="mt-2">
   345→              <ContactStrip phone={profile.phone} email={profile.email} address={profile.address} gstNo={profile.gstNo} />
   346→            </div>
   347→            <div className="mt-3 flex flex-wrap gap-2 text-[11px]">
   348→              <Badge variant="outline" className="border-border/60 bg-card/50">
   349→                <BadgePercent className="size-3" />Commission {profile.defaultCommissionRate}%
   350→              </Badge>
   351→              <Badge variant="outline" className="border-border/60 bg-card/50">GST {profile.defaultGstRate}%</Badge>
   352→              <Badge variant="outline" className="border-border/60 bg-card/50">{p.poCount} POs</Badge>
   353→              <Badge variant="outline" className="border-border/60 bg-card/50">{p.dispatchCount} dispatches</Badge>
   354→              <Badge variant="outline" className="border-border/60 bg-card/50">

   356→              </Badge>
   357→            </div>
   358→          </div>
   359→          <div className="shrink-0">
   360→            <ScoreRing score={Math.round((p.fulfillment + p.onTimeRate + (100 - p.shortShipmentRate)) / 3)} size={72} />
   361→          </div>

   371→          icon={<Package className="size-5" />}
   372→          accent={p.fulfillment >= 85 ? "emerald" : p.fulfillment >= 70 ? "amber" : "rose"}

   375→          label="On-time rate"
   376→          value={`${p.onTimeRate}%`}
   377→          sub="Dispatches by expected date"
   378→          icon={<Clock3 className="size-5" />}
   379→          accent={p.onTimeRate >= 85 ? "emerald" : p.onTimeRate >= 60 ? "amber" : "rose"}

   381→        <KpiCard
   382→          label="Short-ship rate"
   383→          value={`${p.shortShipmentRate}%`}

   443→                          </span>
   444→                        </div>
   445→                        {po.billStatus ? (
   446→                          <div className="text-[11px] text-muted-foreground">
   447→                            Bill: <StatusChip status={po.billStatus} className="ml-1" />
   448→                          </div>
   449→                        ) : null}

   451→                    </div>
   452→                    <div className="mt-3">
   453→                      <div className="flex items-center justify-between text-[11px] text-muted-foreground">

   477→              <Truck className="size-3" />{sections.recentDispatches.length}

   488→                      <p className="kpi-num truncate text-sm font-medium">{d.poNumber}</p>
   489→                      <p className="truncate text-xs text-muted-foreground">{d.buyer} · {formatDate(d.dispatchDate)}</p>

   862→// ─────────────────────────────────────────────────────────────────────────────
   863→// ScoreRing — small circular SVG progress indicator reused from analytics.
   864→// ─────────────────────────────────────────────────────────────────────────────
   865→function ScoreRing({ score, size = 48 }: { score: number; size?: number }) {
   866→  const radius = (size - 8) / 2;
   867→  const circumference = 2 * Math.PI * radius;
   868→  const offset = circumference - (score / 100) * circumference;
   869→  const color = score >= 85 ? "oklch(0.7 0.15 162)"
   870→    : score >= 70 ? "oklch(0.65 0.12 200)"
   871→    : score >= 55 ? "oklch(0.78 0.14 85)"
   872→    : "oklch(0.65 0.2 27)";
   873→  return (
   874→    <div className="relative grid place-items-center" style={{ width: size, height: size }}>
   875→      <svg width={size} height={size} className="-rotate-90">
   876→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="oklch(0.7 0.02 160 / 0.2)" strokeWidth={5} />
   877→        <circle
   878→          cx={size / 2} cy={size / 2} r={radius}
   879→          fill="none" stroke={color} strokeWidth={5}
   880→          strokeDasharray={circumference}
   881→          strokeDashoffset={offset}
   882→          strokeLinecap="round"
   883→          className="transition-all duration-500"
   884→        />
   885→      </svg>
   886→      <div className="absolute inset-0 grid place-items-center">
   887→        <div className="text-center">
   888→          <p className="kpi-num text-lg font-semibold leading-none" style={{ color }}>{score}</p>
   889→          <p className="text-[9px] uppercase tracking-wider text-muted-foreground">score</p>
   890→        </div>
   891→      </div>
   892→    </div>
   893→  );
   894→}
