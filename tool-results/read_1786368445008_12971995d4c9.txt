    49→const BACKUP_REMINDER_KEY = "broker-os:backup-reminder-dismissed";
    50→const BACKUP_STALE_DAYS = 7;

    52→function backupAgeDays(iso: string | null): number | null {
    53→  if (!iso) return null;
    54→  const t = new Date(iso).getTime();
    55→  if (Number.isNaN(t)) return null;
    56→  return Math.floor((Date.now() - t) / (24 * 60 * 60 * 1000));
    57→}

   659→// ─────────────────────────────────────────────────────────────────────────────
   660→// SortableCard — wraps a dashboard section with dnd-kit's useSortable hook.
   661→//
   662→// In customize mode, a thin bar appears above the card content with:
   663→//   • a grip drag handle (the activator for dnd-kit)
   664→//   • the card label
   665→//   • a "Hide" button (eye-off icon) that removes the card from the layout
   666→//
   667→// Out of customize mode, the bar is hidden and a small floating grip appears
   668→// at the top-right corner of the card on hover — clicking it enters customize
   669→// mode (so the hover affordance isn't a tease). The floating grip uses
   670→// `pointer-events-none` until hovered so it never blocks clicks on underlying
   671→// card content (e.g. KPI tile drill-downs).
   672→//
   673→// While dragging, the card lifts with `scale-[1.02] shadow-xl z-10` per spec.
   674→// ─────────────────────────────────────────────────────────────────────────────
   675→function SortableCard({
   676→  id,
   677→  label,
   678→  customizeMode,
   679→  children,
   680→}: {
   681→  id: DashboardCardId;
   682→  label: string;
   683→  customizeMode: boolean;
   684→  children: React.ReactNode;
   685→}) {
   686→  const {
   687→    attributes,
   688→    listeners,
   689→    setNodeRef,
   690→    setActivatorNodeRef,
   691→    transform,
   692→    transition,
   693→    isDragging,
   694→  } = useSortable({ id });
   695→
   696→  const toggleCard = useDashboardLayout((s) => s.toggleCard);
   697→  const setCustomizeMode = useDashboardLayout((s) => s.setCustomizeMode);
   698→
   699→  const style: React.CSSProperties = {
   700→    transform: CSS.Transform.toString(transform),
   701→    transition,
   702→  };
   703→
   704→  return (
   705→    <div
   706→      ref={setNodeRef}
   707→      style={style}
   708→      className={cn(
   709→        "group/sort relative",
   710→        isDragging && "z-10 scale-[1.02] shadow-xl",
   711→      )}
   712→    >
   713→      {customizeMode ? (
   714→        <div className="mb-2 flex items-center justify-between rounded-xl border border-dashed border-emerald-500/40 bg-emerald-500/5 px-3 py-1.5">
   715→          <div className="flex min-w-0 items-center gap-2">
   716→            <button
   717→              type="button"
   718→              ref={setActivatorNodeRef}
   719→              {...attributes}
   720→              {...listeners}
   721→              className="grid size-6 shrink-0 cursor-grab place-items-center rounded-md text-muted-foreground transition-colors hover:bg-emerald-500/10 hover:text-emerald-600 active:cursor-grabbing dark:hover:text-emerald-400"
   722→              aria-label={`Drag ${label} to reorder`}
   723→            >
   724→              <GripVertical className="size-4" />
   725→            </button>
   726→            <span className="truncate text-xs font-medium text-muted-foreground">{label}</span>
   727→          </div>
   728→          <Button
   729→            type="button"
   730→            variant="ghost"
   731→            size="sm"
   732→            className="h-7 shrink-0 gap-1.5 px-2 text-[11px] text-muted-foreground hover:text-rose-600 dark:hover:text-rose-400"
   733→            onClick={() => toggleCard(id)}
   734→            aria-label={`Hide ${label} from dashboard`}
   735→          >
   736→            <EyeOff className="size-3.5" />
   737→            <span className="hidden sm:inline">Hide</span>
   738→          </Button>
   739→        </div>
   740→      ) : (
   741→        // Floating grip — visible on hover only. Clicking it enters customize
   742→        // mode so the broker can start rearranging. pointer-events-none until
   743→        // hovered so it never blocks underlying card interactions.
   744→        <button
   745→          type="button"
   746→          onClick={() => setCustomizeMode(true)}
   747→          aria-label="Customize dashboard layout"
   748→          className="absolute right-2 top-2 z-20 grid size-7 cursor-pointer place-items-center rounded-md bg-background/70 text-muted-foreground opacity-0 backdrop-blur-sm transition-all hover:text-emerald-600 group-hover/sort:opacity-100 group-hover/sort:pointer-events-auto pointer-events-none dark:hover:text-emerald-400"
   749→        >
   750→          <GripVertical className="size-3.5" />
   751→        </button>
   752→      )}
   753→      {children}
   754→    </div>
   755→  );
   756→}
