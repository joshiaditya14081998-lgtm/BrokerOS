   179→function formatBytes(bytes: number): string {
   180→  if (bytes < 1024) return `${bytes} B`;
   181→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   182→  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
   183→}
   184→
   185→function formatRelative(iso: string | null): string {
   186→  if (!iso) return "Never";
   187→  const t = new Date(iso).getTime();
   188→  if (Number.isNaN(t)) return "Never";
   189→  const diff = Date.now() - t;
   190→  const sec = Math.floor(diff / 1000);
   191→  if (sec < 60) return "Just now";
   192→  const min = Math.floor(sec / 60);
   193→  if (min < 60) return `${min} min ago`;
   194→  const hr = Math.floor(min / 60);
   195→  if (hr < 24) return `${hr} hr ago`;
   196→  const day = Math.floor(hr / 24);
   197→  if (day < 30) return `${day} day${day === 1 ? "" : "s"} ago`;
   198→  const mon = Math.floor(day / 30);
   199→  if (mon < 12) return `${mon} month${mon === 1 ? "" : "s"} ago`;
   200→  return new Date(iso).toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   201→}

   203→function formatDateTime(iso: string | null | undefined): string {
   204→  if (!iso) return "Unknown";
   205→  const d = new Date(iso);
   206→  if (Number.isNaN(d.getTime())) return "Unknown";
   207→  return d.toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" });
   208→}

   210→function parseBackupFile(text: string, fileName: string, fileSize: number): { ok: true; backup: ParsedBackup } | { ok: false; error: string } {
   211→  let json: unknown;
   212→  try {
   213→    json = JSON.parse(text);
   214→  } catch {
   215→    return { ok: false, error: `"${fileName}" is not valid JSON.` };
   216→  }
   217→  if (typeof json !== "object" || json === null || Array.isArray(json)) {
   218→    return { ok: false, error: "Backup file must be a JSON object." };
   219→  }
   220→  const obj = json as Record<string, unknown>;
   221→  if (obj.version !== 1) {
   222→    return { ok: false, error: `Unsupported backup version (expected 1, got ${String(obj.version)}).` };
   223→  }
   224→  if (typeof obj.tables !== "object" || obj.tables === null || Array.isArray(obj.tables)) {
   225→    return { ok: false, error: "Backup file is missing the 'tables' object." };
   226→  }
   227→  const tables = obj.tables as Record<string, unknown>;
   228→  for (const t of BACKUP_TABLE_LABELS) {
   229→    if (!Array.isArray(tables[t.key])) {
   230→      return { ok: false, error: `Backup file is missing the "${t.key}" table.` };
   231→    }
   232→  }
   233→  return {
   234→    ok: true,
   235→    backup: {
   236→      version: 1,
   237→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : undefined,
   238→      tables: tables as Record<string, unknown[]>,
   239→      bytes: fileSize,
   240→    },
   241→  };
   242→}

  1520→// Returns "in 23 min" / "in 1 hr 4 min" / "now" / "—" for an ISO timestamp.
  1521→function formatCountdown(iso: string | null): string {
  1522→  if (!iso) return "—";
  1523→  const t = new Date(iso).getTime();
  1524→  if (Number.isNaN(t)) return "—";
  1525→  const diff = t - Date.now();
  1526→  if (diff <= 0) return "now";
  1527→  const min = Math.floor(diff / 60000);
  1528→  if (min < 60) return `in ${min} min`;
  1529→  const hr = Math.floor(min / 60);
  1530→  const rem = min % 60;
  1531→  return rem === 0 ? `in ${hr} hr` : `in ${hr} hr ${rem} min`;
  1532→}
