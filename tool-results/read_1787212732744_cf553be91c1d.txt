    71→    71→  short_shipment: "Short shipment",
    72→    72→  defective_return: "Defective return",
    73→    73→  other: "Other",

    77→    77→

    84→    84→
    85→    85→export function DisputesView() {
    86→    86→  const { data, loading, refresh } = useApi<{ disputes: Dispute[] }>("/api/disputes");
    87→    87→  const [filter, setFilter] = React.useState<FilterKey>("all");
    88→    88→  const [q, setQ] = React.useState("");
    89→    89→  const [open, setOpen] = React.useState(false);
    90→    90→  const { drillFilter, clearDrill } = useUI();
    91→    91→
    92→    92→  // Bulk-select state
    93→    93→  const [selectMode, setSelectMode] = React.useState(false);
    94→    94→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
    95→    95→  const [bulkBusy, setBulkBusy] = React.useState(false);
    96→    96→  const [resolveAllOpen, setResolveAllOpen] = React.useState(false);
    97→    97→  const [rejectAllOpen, setRejectAllOpen] = React.useState(false);
    98→    98→  const [bulkResolution, setBulkResolution] = React.useState("");
    99→    99→
   100→   100→  // Consume drill-down preset from the dashboard "Active POs / Disputes" KPI —
   101→   101→  // preset "open" selects the existing Open filter chip. The chip highlight +
   102→   102→  // the Filtered badge below both surface the active filter to the user.
   103→   103→  const consumedDrill = React.useRef<string | null>(null);
   104→   104→  React.useEffect(() => {
   105→   105→    if (!drillFilter) return;
   106→   106→    const key = `${drillFilter.view}:${drillFilter.preset}`;
   107→   107→    if (consumedDrill.current === key) return;
   108→   108→    if (drillFilter.view === "disputes" && drillFilter.preset === "open") {
   109→   109→      setFilter("open");
   110→   110→      consumedDrill.current = key;
   111→   111→      clearDrill();
   112→   112→    }
   113→   113→  }, [drillFilter, clearDrill]);
   114→   114→
   115→   115→  const disputes = data?.disputes ?? [];
   116→   116→
   117→   117→  const filtered = disputes.filter((d) => {
   118→   118→    if (filter !== "all" && d.status !== filter) return false;
   119→   119→    const term = q.toLowerCase().trim();
   120→   120→    if (!term) return true;
   121→   121→    return (
   122→   122→      d.po.poNumber.toLowerCase().includes(term) ||
   123→   123→      d.po.client.name.toLowerCase().includes(term) ||
   124→   124→      d.po.supplier.name.toLowerCase().includes(term) ||
   125→   125→      (d.description ?? "").toLowerCase().includes(term) ||
   126→   126→      TYPE_LABEL[d.type].toLowerCase().includes(term)
   127→   127→    );
   128→   128→  });
   129→   129→
   130→   130→  // Select-all operates over the currently-filtered OPEN disputes (the only
   131→   131→  // status where bulk resolve/reject meaningfully applies).
   132→   132→  const openFilteredIds = React.useMemo(
   133→   133→    () => filtered.filter((d) => d.status === "open").map((d) => d.id),
   134→   134→    [filtered],
   135→   135→  );
   136→   136→  const allOpenSelected =
   137→   137→    openFilteredIds.length > 0 &&
   138→   138→    openFilteredIds.every((id) => selectedIds.has(id));
   139→   139→
   140→   140→  // --- Bulk helpers -------------------------------------------------------
   141→   141→  const toggleSelect = (id: string) => {
   142→   142→    setSelectedIds((prev) => {
   143→   143→      const next = new Set(prev);
   144→   144→      if (next.has(id)) next.delete(id);
   145→   145→      else next.add(id);
   146→   146→      return next;
   147→   147→    });
   148→   148→  };
   149→   149→
   150→   150→  const toggleSelectAll = () => {
   151→   151→    if (allOpenSelected) {
   152→   152→      setSelectedIds((prev) => {
   153→   153→        const next = new Set(prev);
   154→   154→        for (const id of openFilteredIds) next.delete(id);
   155→   155→        return next;
   156→   156→      });
   157→   157→    } else {
   158→   158→      setSelectedIds((prev) => {
   159→   159→        const next = new Set(prev);
   160→   160→        for (const id of openFilteredIds) next.add(id);
   161→   161→        return next;
   162→   162→      });
   163→   163→    }
   164→   164→  };
   165→   165→
   166→   166→  const clearSelection = () => setSelectedIds(new Set());
   167→   167→
   168→   168→  const exitSelectMode = () => {
   169→   169→    setSelectMode(false);
   170→   170→    clearSelection();
   171→   171→  };
   172→   172→
   173→   173→  const runBulkResolve = async () => {
   174→   174→    const ids = Array.from(selectedIds);
   175→   175→    if (ids.length === 0) return;
   176→   176→    setBulkBusy(true);
   177→   177→    try {
   178→   178→      const res = await api<{ updated: number }>("/api/disputes/bulk", {
   179→   179→        method: "PATCH",
   180→   180→        body: JSON.stringify({
   181→   181→          ids,
   182→   182→          action: "resolve",
   183→   183→          resolution: bulkResolution.trim() || "Bulk resolved.",
   184→   184→        }),
   185→   185→      });
   186→   186→      toast.success(
   187→   187→        `${res.updated} dispute${res.updated === 1 ? "" : "s"} resolved`,
   188→   188→        { description: bulkResolution.trim() || undefined },
   189→   189→      );
   190→   190→      setBulkResolution("");
   191→   191→      setResolveAllOpen(false);
   192→   192→      clearSelection();
   193→   193→      refresh();
   194→   194→    } catch (e) {
   195→   195→      toast.error(e instanceof Error ? e.message : "Bulk resolve failed");
   196→   196→    } finally {
   197→   197→      setBulkBusy(false);
   198→   198→    }
   199→   199→  };
   200→   200→
   201→   201→  const runBulkReject = async () => {
   202→   202→    const ids = Array.from(selectedIds);
   203→   203→    if (ids.length === 0) return;
   204→   204→    setBulkBusy(true);
   205→   205→    try {
   206→   206→      const res = await api<{ updated: number }>("/api/disputes/bulk", {
   207→   207→        method: "PATCH",
   208→   208→        body: JSON.stringify({ ids, action: "reject" }),
   209→   209→      });
   210→   210→      toast.success(
   211→   211→        `${res.updated} dispute${res.updated === 1 ? "" : "s"} rejected`,
   212→   212→      );
   213→   213→      setRejectAllOpen(false);
   214→   214→      clearSelection();
   215→   215→      refresh();
   216→   216→    } catch (e) {
   217→   217→      toast.error(e instanceof Error ? e.message : "Bulk reject failed");
   218→   218→    } finally {
   219→   219→      setBulkBusy(false);
   220→   220→    }
   221→   221→  };
   222→   222→
   223→   223→  return (
   224→   224→    <div className="space-y-5">
   225→   225→      <SectionHeader
   226→   226→        title="Disputes & Returns"
   227→   227→        description="Short-shipments & defects"
   228→   228→        action={
   229→   229→          <div className="flex flex-wrap items-center gap-2">
   230→   230→            <Button
   231→   231→              size="sm"
   232→   232→              variant={selectMode ? "default" : "outline"}
   233→   233→              onClick={() => (selectMode ? exitSelectMode() : setSelectMode(true))}
   234→   234→              aria-pressed={selectMode}
   235→   235→              className={cn(
   236→   236→                !selectMode &&
   237→   237→                  "border-emerald-500/30 bg-emerald-500/10 text-emerald-600 hover:bg-emerald-500/20 hover:text-emerald-600 dark:text-emerald-400",
   238→   238→              )}
   239→   239→            >
   240→   240→              <CheckSquare className="mr-1.5 size-4" />
   241→   241→              {selectMode ? "Exit select" : "Select"}
   242→   242→            </Button>
   243→   243→            <Dialog open={open} onOpenChange={setOpen}>
   244→   244→              <DialogTrigger asChild>
   245→   245→                <Button size="sm"><Plus className="mr-1.5 size-4" />Log dispute</Button>
   246→   246→              </DialogTrigger>
   247→   247→              <LogDisputeDialog onDone={() => { setOpen(false); refresh(); }} />
   248→   248→            </Dialog>
   249→   249→          </div>
   250→   250→        }
   251→   251→      />
   252→   252→
   253→   253→      <SavedViewsBar
   254→   254→        view="disputes"
   255→   255→        currentFilter={{ q, filter }}
   256→   256→        isFilterActive={isFilterActive({ q, filter })}
   257→   257→        onApply={(f) => {
   258→   258→          setQ(typeof f.q === "string" ? f.q : "");
   259→   259→          if (typeof f.filter === "string") setFilter(f.filter as FilterKey);
   260→   260→        }}
   261→   261→      />
   262→   262→
   263→   263→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   264→   264→        <div className="inline-flex rounded-xl border border-border/60 bg-card/30 p-1">
   265→   265→          {FILTERS.map((f) => (
   266→   266→            <button
   267→   267→              key={f.key}
   268→   268→              onClick={() => setFilter(f.key)}
   269→   269→              className={cn(
   270→   270→                "rounded-lg px-3 py-1.5 text-xs font-medium transition-colors",
   271→   271→                filter === f.key
   272→   272→                  ? "bg-primary text-primary-foreground shadow-sm"
   273→   273→                  : "text-muted-foreground hover:text-foreground"
   274→   274→              )}
   275→   275→            >
   276→   276→              {f.label}
   277→   277→              <span className="ml-1.5 text-[10px] opacity-70">
   278→   278→                {f.key === "all" ? disputes.length : disputes.filter((d) => d.status === f.key).length}
   279→   279→              </span>
   280→   280→            </button>
   281→   281→          ))}
   282→   282→        </div>
   283→   283→        <Input
   284→   284→          placeholder="Search PO, party, type…"
   285→   285→          value={q}
   286→   286→          onChange={(e) => setQ(e.target.value)}
   287→   287→          className="h-9 sm:w-64"
   288→   288→        />
   289→   289→      </div>
   290→   290→
   291→   291→      {filter !== "all" && (
   292→   292→        <div className="-mt-1">
   293→   293→          <Badge
   294→   294→            variant="outline"
   295→   295→            className="border-emerald-500/40 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300"
   296→   296→          >
   297→   297→            <Filter className="mr-1 size-3" />
   298→   298→            Filtered: {FILTERS.find((f) => f.key === filter)?.label.toLowerCase()} only
   299→   299→            <button
   300→   300→              type="button"
   301→   301→              onClick={() => setFilter("all")}
   302→   302→              aria-label="Clear dispute filter"
   303→   303→              className="ml-1 inline-flex size-3.5 items-center justify-center rounded-full hover:bg-emerald-500/20"
   304→   304→            >
   305→   305→              <X className="size-3" />
   306→   306→            </button>
   307→   307→          </Badge>
   308→   308→        </div>
   309→   309→      )}
   310→   310→
   311→   311→      {selectMode && (
   312→   312→        <div className="flex items-center justify-between rounded-xl border border-border/60 bg-card/30 px-3 py-2">
   313→   313→          <label
   314→   314→            htmlFor="dispute-select-all"
   315→   315→            className="inline-flex cursor-pointer items-center gap-2 text-xs font-medium text-foreground"
   316→   316→          >
   317→   317→            <Checkbox
   318→   318→              id="dispute-select-all"
   319→   319→              checked={allOpenSelected}
   320→   320→              onCheckedChange={toggleSelectAll}
   321→   321→              disabled={openFilteredIds.length === 0}
   322→   322→              className="data-[state=checked]:bg-emerald-600 data-[state=checked]:border-emerald-600 text-white"
   323→   323→            />
   324→   324→            Select all open
   325→   325→            <span className="rounded-full bg-card/80 px-1.5 py-0.5 text-[10px] text-muted-foreground">
   326→   326→              {openFilteredIds.length}
   327→   327→            </span>
   328→   328→          </label>
   329→   329→          <span className="text-[11px] text-muted-foreground">
   330→   330→            {selectedIds.size === 0
   331→   331→              ? "Tap a card to select"
   332→   332→              : `${selectedIds.size} selected`}
   333→   333→          </span>
   334→   334→        </div>
   335→   335→      )}
   336→   336→
   337→   337→      {loading ? (
   338→   338→        <div className="space-y-2">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-2xl" />)}</div>
   339→   339→      ) : filtered.length === 0 ? (
   340→   340→        <GlassCard className="p-8">
   341→   341→          <EmptyState
   342→   342→            title="No disputes to show"
   343→   343→            hint="Log a short-shipment or defective return to start tracking. Resolved defective returns auto-recompute the bill."
   344→   344→            icon={<AlertTriangle className="size-5" />}
   345→   345→          />
   346→   346→        </GlassCard>
   347→   347→      ) : (
   348→   348→        <div className="grid gap-3 lg:grid-cols-2">
   349→   349→          {filtered.map((d) => (
   350→   350→            <DisputeCard
   351→   351→              key={d.id}
   352→   352→              dispute={d}
   353→   353→              onDone={refresh}
   354→   354→              selectMode={selectMode}
   355→   355→              selected={selectedIds.has(d.id)}
   356→   356→              onToggleSelect={toggleSelect}
   357→   357→            />
   358→   358→          ))}
   359→   359→        </div>
   360→   360→      )}
   361→   361→
   362→   362→      {/* Bulk action bar — fixed at the bottom of the viewport */}
   363→   363→      <BulkActionBar selectedCount={selectedIds.size} onClear={clearSelection}>
   364→   364→        <Button
   365→   365→          size="sm"
   366→   366→          onClick={() => { setBulkResolution(""); setResolveAllOpen(true); }}
   367→   367→          disabled={bulkBusy}
   368→   368→          className="h-8 bg-emerald-600 text-white hover:bg-emerald-700"
   369→   369→        >
   370→   370→          <CheckCheck className="mr-1.5 size-3.5" />
   371→   371→          Resolve all
   372→   372→        </Button>
   373→   373→        <Button
   374→   374→          size="sm"
   375→   375→          variant="outline"
   376→   376→          onClick={() => setRejectAllOpen(true)}
   377→   377→          disabled={bulkBusy}
   378→   378→          className="h-8 border-rose-500/40 bg-rose-500/10 text-rose-700 hover:bg-rose-500/20 hover:text-rose-700 dark:text-rose-300"
   379→   379→        >
   380→   380→          <Ban className="mr-1.5 size-3.5" />
   381→   381→          Reject all
   382→   382→        </Button>
   383→   383→      </BulkActionBar>
   384→   384→
   385→   385→      {/* Bulk resolve dialog — shared resolution note for all selected */}
   386→   386→      <Dialog open={resolveAllOpen} onOpenChange={(o) => { setResolveAllOpen(o); if (!o) setBulkResolution(""); }}>
   387→   387→        <DialogContent className="glass-strong max-w-md">
   388→   388→          <DialogHeader>
   389→   389→            <DialogTitle>Resolve {selectedIds.size} dispute{selectedIds.size === 1 ? "" : "s"}</DialogTitle>
   390→   390→            <DialogDescription className="sr-only">
   391→   391→              Apply one resolution note to every selected dispute. Resolved defective-return disputes will trigger an automatic bill recompute per linked PO.
   392→   392→            </DialogDescription>
   393→   393→          </DialogHeader>
   394→   394→          <div className="space-y-2 py-2">
   395→   395→            <Label className="text-xs font-medium text-muted-foreground">Shared resolution note</Label>
   396→   396→            <Textarea
   397→   397→              placeholder="e.g. Verified with supplier — credit note issued for the affected quantity."
   398→   398→              value={bulkResolution}
   399→   399→              onChange={(e) => setBulkResolution(e.target.value)}
   400→   400→              rows={4}
   401→   401→            />
   402→   402→            <p className="rounded-lg border border-amber-500/30 bg-amber-500/10 px-3 py-2 text-[11px] text-amber-700 dark:text-amber-300">
   403→   403→              Any defective-return dispute in the selection will automatically recompute its linked bill (base − short-shipment − returns).
   404→   404→            </p>
   405→   405→          </div>
   406→   406→          <DialogFooter>
   407→   407→            <Button variant="outline" onClick={() => setResolveAllOpen(false)} disabled={bulkBusy}>Cancel</Button>
   408→   408→            <Button onClick={runBulkResolve} disabled={bulkBusy} className="bg-emerald-600 text-white hover:bg-emerald-700">
   409→   409→              {bulkBusy ? "Resolving…" : `Resolve ${selectedIds.size}`}
   410→   410→            </Button>
   411→   411→          </DialogFooter>
   412→   412→        </DialogContent>
   413→   413→      </Dialog>
   414→   414→
   415→   415→      {/* Bulk reject confirmation */}
   416→   416→      <AlertDialog open={rejectAllOpen} onOpenChange={setRejectAllOpen}>
   417→   417→        <AlertDialogContent className="glass-strong">
   418→   418→          <AlertDialogHeader>
   419→   419→            <AlertDialogTitle>Reject {selectedIds.size} dispute{selectedIds.size === 1 ? "" : "s"}?</AlertDialogTitle>
   420→   420→            <AlertDialogDescription>
   421→   421→              All selected disputes will be marked <span className="font-medium text-foreground">rejected</span>. Linked bills are not affected. This action is recorded in the audit log.
   422→   422→            </AlertDialogDescription>
   423→   423→          </AlertDialogHeader>
   424→   424→          <AlertDialogFooter>
   425→   425→            <AlertDialogCancel disabled={bulkBusy}>Cancel</AlertDialogCancel>
   426→   426→            <AlertDialogAction
   427→   427→              className="bg-rose-600 text-white hover:bg-rose-700"
   428→   428→              onClick={runBulkReject}
   429→   429→              disabled={bulkBusy}
   430→   430→            >
   431→   431→              {bulkBusy ? "Rejecting…" : `Reject ${selectedIds.size}`}
   432→   432→            </AlertDialogAction>
   433→   433→          </AlertDialogFooter>
   434→   434→        </AlertDialogContent>
   435→   435→      </AlertDialog>
   436→   436→    </div>
   437→   437→  );
   438→   438→}
   439→   439→
   440→   440→function DisputeCard({
   441→   441→  dispute,
   442→   442→  onDone,
   443→   443→  selectMode,
   444→   444→  selected,
   445→   445→  onToggleSelect,
