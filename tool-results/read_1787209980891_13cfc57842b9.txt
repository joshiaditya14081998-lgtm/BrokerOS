   177→function formatBytes(bytes: number): string {
   178→  if (bytes < 1024) return `${bytes} B`;
   179→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   180→  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
   181→}
   182→
   183→function formatRelative(iso: string | null): string {
   184→  if (!iso) return "Never";
   185→  const t = new Date(iso).getTime();
   186→  if (Number.isNaN(t)) return "Never";
   187→  const diff = Date.now() - t;
   188→  const sec = Math.floor(diff / 1000);
   189→  if (sec < 60) return "Just now";
   190→  const min = Math.floor(sec / 60);
   191→  if (min < 60) return `${min} min ago`;
   192→  const hr = Math.floor(min / 60);
   193→  if (hr < 24) return `${hr} hr ago`;
   194→  const day = Math.floor(hr / 24);
   195→  if (day < 30) return `${day} day${day === 1 ? "" : "s"} ago`;
   196→  const mon = Math.floor(day / 30);
   197→  if (mon < 12) return `${mon} month${mon === 1 ? "" : "s"} ago`;
   198→  return new Date(iso).toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   199→}

   201→function formatDateTime(iso: string | null | undefined): string {
   202→  if (!iso) return "Unknown";
   203→  const d = new Date(iso);
   204→  if (Number.isNaN(d.getTime())) return "Unknown";
   205→  return d.toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" });
   206→}

   208→function parseBackupFile(text: string, fileName: string, fileSize: number): { ok: true; backup: ParsedBackup } | { ok: false; error: string } {
   209→  let json: unknown;
   210→  try {
   211→    json = JSON.parse(text);
   212→  } catch {
   213→    return { ok: false, error: `"${fileName}" is not valid JSON.` };
   214→  }
   215→  if (typeof json !== "object" || json === null || Array.isArray(json)) {
   216→    return { ok: false, error: "Backup file must be a JSON object." };
   217→  }
   218→  const obj = json as Record<string, unknown>;
   219→  if (obj.version !== 1) {
   220→    return { ok: false, error: `Unsupported backup version (expected 1, got ${String(obj.version)}).` };
   221→  }
   222→  if (typeof obj.tables !== "object" || obj.tables === null || Array.isArray(obj.tables)) {
   223→    return { ok: false, error: "Backup file is missing the 'tables' object." };
   224→  }
   225→  const tables = obj.tables as Record<string, unknown>;
   226→  for (const t of BACKUP_TABLE_LABELS) {
   227→    if (!Array.isArray(tables[t.key])) {
   228→      return { ok: false, error: `Backup file is missing the "${t.key}" table.` };
   229→    }
   230→  }
   231→  return {
   232→    ok: true,
   233→    backup: {
   234→      version: 1,
   235→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : undefined,
   236→      tables: tables as Record<string, unknown[]>,
   237→      bytes: fileSize,
   238→    },
   239→  };
   240→}

  1393→  const lastRun = status?.lastRun ?? null;

  1397→
  1398→  return (

  1513→// Returns "in 23 min" / "in 1 hr 4 min" / "now" / "—" for an ISO timestamp.
  1514→function formatCountdown(iso: string | null): string {
  1515→  if (!iso) return "—";
  1516→  const t = new Date(iso).getTime();
  1517→  if (Number.isNaN(t)) return "—";
  1518→  const diff = t - Date.now();
  1519→  if (diff <= 0) return "now";
  1520→  const min = Math.floor(diff / 60000);
  1521→  if (min < 60) return `in ${min} min`;
  1522→  const hr = Math.floor(min / 60);
  1523→  const rem = min % 60;
  1524→  return rem === 0 ? `in ${hr} hr` : `in ${hr} hr ${rem} min`;
  1525→}
