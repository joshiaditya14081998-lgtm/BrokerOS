   457→// ─────────────────────────────────────────────────────────────────────────────
   458→// Photos — stage-polymorphic (S3-migrable: swap URL for object storage later)
   459→//
   460→// Polymorphic pattern: `entityType` + `entityId` carry the logical link
   461→// (so the API can query /api/photos?entityType=Dispatch&entityId=… uniformly),
   462→// AND a dedicated nullable FK column per parent table is kept in sync so
   463→// Prisma relations work for `include: { photos: true }` reads. Exactly one
   464→// of the five *Id columns is set per row (the one matching `entityType`).
   465→// ─────────────────────────────────────────────────────────────────────────────
   466→
   467→model Photo {
   468→  id         String   @id @default(cuid())
   469→  brokerId      String
   470→  broker        Broker   @relation(fields: [brokerId], references: [id], onDelete: Cascade)
   471→  stage      String   // booking | dispatch | receiving | dispute | visit | payment
   472→  entityType String   // Visit | Booking | Dispatch | Dispute | Payment
   473→  entityId   String   // polymorphic — kept in sync with the matching *Id column below
   474→  url        String   // path under /uploads/...  (full-resolution original)
   475→  thumbnailUrl String? // path under /uploads/thumb_... (400px-wide JPEG q80) — null on legacy records
   476→  caption    String?
   477→  createdAt  DateTime @default(now())
   478→
   479→  // Optional FK columns — exactly one is set per row, matching entityType.
   480→  visitId    String?
   481→  bookingId  String?
   482→  dispatchId String?
   483→  disputeId  String?
   484→  paymentId  String?
   485→
   486→  visit      Visit?   @relation("VisitPhotos",    fields: [visitId],    references: [id], onDelete: Cascade)
   487→  booking    Booking? @relation("BookingPhotos",  fields: [bookingId],  references: [id], onDelete: Cascade)
   488→  dispatch   Dispatch? @relation("DispatchPhotos", fields: [dispatchId], references: [id], onDelete: Cascade)
   489→  dispute    Dispute? @relation("DisputePhotos",  fields: [disputeId],  references: [id], onDelete: Cascade)
   490→  payment    Payment? @relation("PaymentPhotos",  fields: [paymentId],  references: [id], onDelete: Cascade)
   491→
   492→  @@index([entityType, entityId])
   493→}

   546→//
   547→// A Tag is a reusable label (e.g. "VIP", "Festive Season") with a colour.
   548→// An EntityTag links a Tag to an entity (Client | Supplier | PurchaseOrder).
   549→// The polymorphic pattern matches the Photo model: `entityType` + `entityId`
   550→// carry the logical link for uniform API queries, AND a dedicated nullable FK
   551→// column per parent table is kept in sync so Prisma relations work for
   552→// `include: { tags: true }` reads + cascade deletes. Exactly one of the
   553→// three optional *Id columns is set per row (the one matching `entityType`).
   554→// ─────────────────────────────────────────────────────────────────────────────
   555→
   556→model Tag {
   557→  id        String   @id @default(cuid())
   558→  brokerId      String
   559→  broker        Broker   @relation(fields: [brokerId], references: [id], onDelete: Cascade)
   560→  name      String   @unique
   561→  color     String   @default("emerald") // emerald | amber | rose | teal | plum | slate
   562→  createdAt DateTime @default(now())
   563→
   564→  entityTags EntityTag[]
   565→}
   566→
   567→model EntityTag {
   568→  id         String   @id @default(cuid())
   569→  tagId      String
   570→  tag        Tag      @relation(fields: [tagId], references: [id], onDelete: Cascade)
   571→  entityType String   // Client | Supplier | PurchaseOrder
   572→  entityId   String   // polymorphic — kept in sync with the matching *Id column below
   573→  createdAt  DateTime @default(now())
   574→
   575→  // Optional FK columns — exactly one is set per row, matching entityType.
   576→  clientId          String?
   577→  supplierId        String?
   578→  purchaseOrderId   String?
   579→
   580→  client            Client?        @relation(fields: [clientId],        references: [id], onDelete: Cascade)
   581→  supplier          Supplier?      @relation(fields: [supplierId],       references: [id], onDelete: Cascade)
   582→  purchaseOrder     PurchaseOrder? @relation(fields: [purchaseOrderId],  references: [id], onDelete: Cascade)
   583→
   584→  @@unique([entityType, entityId, tagId])
   585→  @@index([entityType, entityId])
   586→  @@index([tagId])
   587→}

   589→// ─────────────────────────────────────────────────────────────────────────────
   590→// Saved Views — user-persisted filter combinations for list views
   591→//
   592→// Brokers often apply the same filter combos repeatedly (e.g. "VIP clients
   593→// with outstanding > ₹50k", "Overdue dispatches", "Open disputes this week").
   594→// A SavedView captures the filter state of a view as JSON so it can be
   595→// re-applied with a single click from the SavedViewsBar or the Saved Views
   596→// management page.
   597→//
   598→// `view` is a ViewKey string ("clients" | "suppliers" | "pos" | "bills" |
   599→// "disputes" | "audit" | …). `filterJson` is the serialised filter state
   600→// (shape varies per view — search query + active tags + status chip + date
   601→// range, etc.). `entityType` is optional, set for tag-filtered views so the
   602→// Saved Views page can group / describe them. `icon` is one of
   603→// star|heart|pin|flag|bookmark (lucide icon names).
   604→// ─────────────────────────────────────────────────────────────────────────────
   605→
   606→model SavedView {
   607→  id         String   @id @default(cuid())
   608→  brokerId      String
   609→  broker        Broker   @relation(fields: [brokerId], references: [id], onDelete: Cascade)
   610→  name       String
   611→  view       String   // ViewKey — clients | suppliers | pos | bills | disputes | audit | …
   612→  filterJson String   // JSON of the filter state (shape varies per view)
   613→  entityType String?  // optional — for tag-filtered views (Client | Supplier | PurchaseOrder)
   614→  icon       String   @default("star") // star | heart | pin | flag | bookmark
   615→  createdAt  DateTime @default(now())
   616→  updatedAt  DateTime @updatedAt
   617→
   618→  @@index([view])
   619→}

   621→// ─────────────────────────────────────────────────────────────────────────────
   622→// Report Templates — user-defined custom report configurations
   623→//
   624→// Brokers have varied reporting needs — some want a weekly client performance
   625→// summary, others a monthly supplier comparison, others a custom date-range
   626→// brokerage breakdown. A ReportTemplate captures the user-defined report
   627→// configuration (entity type, columns, filters, grouping, sorting, options)
   628→// as JSON in `configJson` so the Custom Report Builder view can edit it and
   629→// `/api/reports/custom?templateId=…` can render a print-optimized HTML PDF
   630→// from it on demand.
   631→//
   632→// `type` is one of: summary | comparison | ledger | custom.
   633→// `configJson` shape (see `src/lib/report-columns.ts` + ReportBuilderView):
   634→//   {
   635→//     entityType: "client"|"supplier"|"bill"|"payment"|"brokerage"|"dispatch"|"dispute",
   636→//     columns: string[],                 // column keys from REPORT_COLUMNS[entityType]
   637→//     filters: {
   638→//       dateRange?: "all"|"week"|"month"|"quarter"|"year"|"custom",
   639→//       fromDate?: string,               // ISO date — when dateRange === "custom"
   640→//       toDate?: string,                 // ISO date — when dateRange === "custom"
   641→//       status?: string,                 // entity-specific status (e.g. "fully_paid")
   642→//       tagId?: string                   // restrict to entities with this Tag
   643→//     },
   644→//     groupBy?: string | null,           // column key to group rows by (sub-totals per group)
   645→//     sortBy?: { field: string; direction: "asc"|"desc" },
   646→//     title: string,                     // report heading (often === template name)
   647→//     includeCharts: boolean             // embed an inline SVG bar chart in the PDF
   648→//   }
   649→// ─────────────────────────────────────────────────────────────────────────────
   650→
   651→model ReportTemplate {
   652→  id          String   @id @default(cuid())
   653→  brokerId      String
   654→  broker        Broker   @relation(fields: [brokerId], references: [id], onDelete: Cascade)
   655→  name        String
   656→  description String?
   657→  type        String   // summary | comparison | ledger | custom
   658→  configJson  String   // JSON of the template config (see module doc)
   659→  createdAt   DateTime @default(now())
   660→  updatedAt   DateTime @updatedAt
   661→
   662→  @@index([type])
   663→}
