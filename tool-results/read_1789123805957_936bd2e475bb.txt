    39→function fmtCurrency(n: number): string {
    40→  const v = Number.isFinite(n) ? n : 0;
    41→  return new Intl.NumberFormat("en-IN", {
    42→    style: "currency",
    43→    currency: "INR",
    44→    maximumFractionDigits: 0,
    45→  }).format(v);
    46→}

    48→function fmtDate(d: Date | string | null | undefined): string {
    49→  if (!d) return "—";
    50→  const date = typeof d === "string" ? new Date(d) : d;
    51→  return date.toLocaleDateString("en-IN", {
    52→    day: "2-digit",
    53→    month: "short",
    54→    year: "numeric",
    55→  });
    56→}
    57→
    58→function fmtDateTime(d: Date | string | null | undefined): string {
    59→  if (!d) return "—";
    60→  const date = typeof d === "string" ? new Date(d) : d;
    61→  return date.toLocaleString("en-IN", {
    62→    day: "2-digit",
    63→    month: "short",
    64→    year: "numeric",
    65→    hour: "2-digit",
    66→    minute: "2-digit",
    67→  });
    68→}

    70→function titleCase(s: string): string {
    71→  return s.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase());
    72→}

    74→function escapeHtml(s: string | null | undefined): string {
    75→  if (s == null) return "";
    76→  return String(s)
    77→    .replace(/&/g, "&amp;")
    78→    .replace(/</g, "&lt;")
    79→    .replace(/>/g, "&gt;")
    80→    .replace(/"/g, "&quot;")
    81→    .replace(/'/g, "&#39;");
    82→}

    84→// ── Range bounds (reused from dashboard route) ───────────────────────────────
    85→
    86→function getRangeBounds(range: Range): { start: Date | undefined; end: Date; label: string } {
    87→  const now = new Date();
    88→  if (range === "all") {
    89→    return { start: undefined, end: now, label: "All time" };
    90→  }
    91→  const y = now.getFullYear();
    92→  const m = now.getMonth();
    93→  let start: Date;
    94→  let label: string;
    95→  if (range === "month") {
    96→    start = new Date(y, m, 1);
    97→    label = start.toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    98→  } else if (range === "quarter") {
    99→    const qStartMonth = Math.floor(m / 3) * 3;
   100→    start = new Date(y, qStartMonth, 1);
   101→    const qNum = Math.floor(qStartMonth / 3) + 1;
   102→    label = `Q${qNum} ${y}`;
   103→  } else {
   104→    start = new Date(y, 0, 1);
   105→    label = `Year ${y}`;
   106→  }
   107→  return { start, end: now, label };
   108→}

   489→function statusPill(status: string): string {
   490→  const s = status.toLowerCase();
   491→  let cls = "zinc";
   492→  if (["fully_paid", "fully_delivered", "delivered", "resolved", "occurred", "done", "paid", "scheduled"].includes(s)) {
   493→    cls = "emerald";
   494→  } else if (["partially_paid", "partially_delivered", "in_transit", "accrued"].includes(s)) {
   495→    cls = "amber";
   496→  } else if (["pending", "open", "no_show", "short_shipment", "defective_return", "overdue"].includes(s)) {
   497→    cls = "rose";
   498→  } else if (["closed", "rejected", "dismissed"].includes(s)) {
   499→    cls = "zinc";
   500→  }
   501→  return `<span class="pill ${cls}">${escapeHtml(titleCase(status))}</span>`;
   502→}
   503→
   504→function eligiblePill(eligible: boolean, forceEligible?: boolean): string {
   505→  if (!eligible) return `<span class="pill rose">No</span>`;
   506→  return forceEligible
   507→    ? `<span class="pill amber">Forced</span>`
   508→    : `<span class="pill emerald">Yes</span>`;
   509→}

   513→async function buildBrokerageStatement(range: Range): Promise<string> {
   514→  const { start, end, label } = getRangeBounds(range);
   515→
   516→  const brokerages = await db.brokerage.findMany({
   517→    where: start
   518→      ? {
   519→          OR: [
   520→            { eligibleAt: { gte: start, lte: end } },
   521→            { eligibleAt: null, createdAt: { gte: start, lte: end } },
   522→          ],
   523→        }
   524→      : undefined,
   525→    include: {
   526→      bill: { select: { billNumber: true, baseAmount: true, status: true, po: { select: { poNumber: true } } } },
   527→      client: { select: { name: true } },
   528→      supplier: { select: { name: true } },
   529→      payout: { select: { id: true, status: true, paidAt: true } },
   530→    },
   531→    orderBy: { eligibleAt: "asc" },
   532→  });
   533→
   534→  const payouts = await db.brokeragePayout.findMany({
   535→    where: start ? { createdAt: { gte: start, lte: end } } : undefined,
   536→    include: { client: { select: { name: true } }, brokerages: { select: { id: true, brokerageAmount: true } } },
   537→    orderBy: { createdAt: "desc" },
   538→  });
   539→
   540→  const totalBrokerage = brokerages.reduce((s, b) => s + b.brokerageAmount, 0);
   541→  const accruedTotal = brokerages
   542→    .filter((b) => b.eligible && b.payoutStatus === "accrued")
   543→    .reduce((s, b) => s + b.brokerageAmount, 0);
   544→  const scheduledTotal = brokerages
   545→    .filter((b) => b.eligible && b.payoutStatus === "scheduled")
   546→    .reduce((s, b) => s + b.brokerageAmount, 0);
   547→  const paidTotal = brokerages
   548→    .filter((b) => b.payoutStatus === "paid")
   549→    .reduce((s, b) => s + b.brokerageAmount, 0);
   550→  const pendingTotal = brokerages
   551→    .filter((b) => !b.eligible)
   552→    .reduce((s, b) => s + b.brokerageAmount, 0);
   553→  const billsCount = new Set(brokerages.map((b) => b.billId)).size;
   554→
   555→  // Group by client for the detail table (multiple sub-totals)
   556→  const byClient = new Map<string, { name: string; rows: typeof brokerages }>();
   557→  for (const b of brokerages) {
   558→    const key = b.clientId;
   559→    if (!byClient.has(key)) {
   560→      byClient.set(key, { name: b.client.name, rows: [] });
   561→    }
   562→    byClient.get(key)!.rows.push(b);
   563→  }
   564→  // Sort clients by name
   565→  const clientGroups = Array.from(byClient.entries())
   566→    .map(([id, g]) => ({ id, name: g.name, rows: g.rows }))
   567→    .sort((a, b) => a.name.localeCompare(b.name));
   568→
   569→  // Build the detail table — grouped by client with sub-totals
   570→  const detailRows: string[] = [];
   571→  let grandBase = 0;
   572→  let grandBrokerage = 0;
   573→  for (const group of clientGroups) {
   574→    detailRows.push(
   575→      `<tr class="group-header"><td colspan="9">${escapeHtml(group.name)} — ${group.rows.length} entr${group.rows.length === 1 ? "y" : "ies"}</td></tr>`,
   576→    );
   577→    let subBase = 0;
   578→    let subBrokerage = 0;
   579→    for (const b of group.rows) {
   580→      subBase += b.baseAmount;
   581→      subBrokerage += b.brokerageAmount;
   582→      grandBase += b.baseAmount;
   583→      grandBrokerage += b.brokerageAmount;
   584→      detailRows.push(
   585→        `<tr>
   586→          <td>${escapeHtml(b.bill.billNumber)}</td>
   587→          <td>${escapeHtml(b.client.name)}</td>
   588→          <td>${escapeHtml(b.supplier.name)}</td>
   589→          <td class="num">${fmtCurrency(b.baseAmount)}</td>
   590→          <td class="num">${b.commissionRate.toFixed(1)}%</td>
   591→          <td class="num emerald">${fmtCurrency(b.brokerageAmount)}</td>
   592→          <td class="center">${eligiblePill(b.eligible, b.forceEligible)}</td>
   593→          <td class="center">${statusPill(b.payoutStatus)}</td>
   594→          <td>${fmtDate(b.eligibleAt)}</td>
   595→        </tr>`,
   596→      );
   597→    }
   598→    detailRows.push(
   599→      `<tr class="sub-total">
   600→        <td colspan="3">Sub-total — ${escapeHtml(group.name)}</td>
   601→        <td class="num">${fmtCurrency(subBase)}</td>
   602→        <td></td>
   603→        <td class="num">${fmtCurrency(subBrokerage)}</td>
   604→        <td colspan="3"></td>
   605→      </tr>`,
   606→    );
   607→  }
   608→
   609→  if (brokerages.length === 0) {
   610→    detailRows.push(
   611→      `<tr><td colspan="9" style="text-align:center;padding:24px;color:#a1a1aa;">No brokerage entries in this range.</td></tr>`,
   612→    );
   613→  }
   614→
   615→  // Payout summary table
   616→  const payoutRows = payouts.length
   617→    ? payouts
   618→        .map(
   619→          (p) => `<tr>
   620→        <td>${escapeHtml(p.client.name)}</td>
   621→        <td class="center">${escapeHtml(titleCase(p.cadence))}</td>
   622→        <td>${fmtDate(p.periodStart)}</td>
   623→        <td>${fmtDate(p.periodEnd)}</td>
   624→        <td class="num">${fmtCurrency(p.totalAmount)}</td>
   625→        <td class="center">${statusPill(p.status)}</td>
   626→        <td class="center">${p.paidAt ? fmtDate(p.paidAt) : "—"}</td>
   627→        <td class="num">${p.brokerages.length}</td>
   628→      </tr>`,
   629→        )
   630→        .join("")
   631→    : `<tr><td colspan="8" style="text-align:center;padding:24px;color:#a1a1aa;">No payouts in this range.</td></tr>`;
   632→
   633→  const body = `
   634→    <div class="kpi-grid">
   635→      <div class="kpi-card">
   636→        <div class="kpi-label">Total brokerage</div>
   637→        <div class="kpi-value emerald">${fmtCurrency(totalBrokerage)}</div>
   638→        <div class="kpi-sub">${brokerages.length} entries</div>
   639→      </div>
   640→      <div class="kpi-card">
   641→        <div class="kpi-label">Eligible pending</div>
   642→        <div class="kpi-value amber">${fmtCurrency(accruedTotal + scheduledTotal)}</div>
   643→        <div class="kpi-sub">Accrued + scheduled</div>
   644→      </div>
   645→      <div class="kpi-card">
   646→        <div class="kpi-label">Paid out</div>
   647→        <div class="kpi-value emerald">${fmtCurrency(paidTotal)}</div>
   648→        <div class="kpi-sub">${payouts.length} payout batches</div>
   649→      </div>
   650→      <div class="kpi-card">
   651→        <div class="kpi-label">Not eligible</div>
   652→        <div class="kpi-value rose">${fmtCurrency(pendingTotal)}</div>
   653→        <div class="kpi-sub">${billsCount} bills</div>
   654→      </div>
   655→    </div>
   656→
   657→    <div class="section">
   658→      <div class="section-title">Brokerage ledger — grouped by client</div>
   659→      <div class="section-desc">All brokerage entries in the selected range. Sub-totals per client; grand total at the foot.</div>
   660→      <table class="report">
   661→        <thead>
   662→          <tr>
   663→            <th>Bill</th>
   664→            <th>Client</th>

   672→          </tr>

   674→        <tbody>

   724→async function buildClientLedger(clientId: string): Promise<string> {
   725→  const client = await db.client.findUnique({
   726→    where: { id: clientId },
   727→    include: {
   728→      bills: {
   729→        include: {
   730→          po: { select: { poNumber: true, supplier: { select: { name: true } } } },
   731→          payments: { orderBy: { date: "desc" } },
   732→          brokerage: true,
   733→        },
   734→        orderBy: { createdAt: "desc" },
   735→      },
   736→      brokerages: {
   737→        include: {
   738→          bill: { select: { billNumber: true } },
   739→          supplier: { select: { name: true } },
   740→          payout: true,
   741→        },
   742→        orderBy: { createdAt: "desc" },
   743→      },
   744→    },
   745→  });
   746→  if (!client) {
   747→    return htmlShell("Client Not Found", "—",
   748→      `<div style="padding:40px;text-align:center;color:#a1a1aa;">The requested client could not be found.</div>`);
   749→  }
   750→
   751→  // Build the ledger (bills + payments, running balance)
   752→  type Ledger = { date: Date; type: string; ref: string; debit: number; credit: number; balance: number };
   753→  const entries: Ledger[] = [];
   754→  for (const bill of client.bills) {
   755→    entries.push({ date: bill.createdAt, type: "Bill", ref: bill.billNumber, debit: bill.finalAmount, credit: 0, balance: 0 });
   756→    for (const p of bill.payments) {
   757→      entries.push({ date: p.date, type: "Payment", ref: p.reference || p.id.slice(-6), debit: 0, credit: p.amount, balance: 0 });

   761→  let bal = 0;
   762→  for (const e of entries) {
   763→    bal += e.debit - e.credit;
   764→    e.balance = bal;

   767→  entries.reverse();
   768→
   769→  const totalBusiness = client.bills.reduce((s, b) => s + b.finalAmount, 0);
   770→  const outstanding = client.bills.reduce((s, b) => s + (b.finalAmount - b.paidAmount), 0);

   772→  const brokeragePaid = client.brokerages.filter((b) => b.payoutStatus === "paid").reduce((s, b) => s + b.brokerageAmount, 0);

   783→    try {
   784→      orderedQty = (JSON.parse(p.lineItemsJson) as { setQty: number }[]).reduce((s, i) => s + i.setQty, 0);
   785→    } catch {
   786→      orderedQty = 0;
   787→    }
   788→    return {
   789→      poNumber: p.poNumber,
   790→      status: p.status,
   791→      ordered: p.totalValue,
   792→      orderedQty,
   793→      dispatchedQty,
   794→      fulfillment: orderedQty ? Math.round((dispatchedQty / orderedQty) * 100) : 0,
   795→      supplierName: p.supplier.name,
   796→      expectedDispatchDate: p.expectedDispatchDate,
   797→      revisedDispatchDate: p.revisedDispatchDate,
   798→    };
   799→  });
   800→
   801→  const ledgerRows = entries.length
   802→    ? entries
   803→        .map(
   804→          (e) => `<tr>
   805→        <td>${fmtDate(e.date)}</td>

   808→        <td class="num">${e.debit ? fmtCurrency(e.debit) : "—"}</td>
   809→        <td class="num emerald">${e.credit ? fmtCurrency(e.credit) : "—"}</td>
   810→        <td class="num">${fmtCurrency(e.balance)}</td>
   811→      </tr>`,
   812→        )
   813→        .join("")
   814→    : `<tr><td colspan="6" style="text-align:center;padding:24px;color:#a1a1aa;">No ledger entries yet.</td></tr>`;
   815→
   816→  const deliveryRows = deliverySummary.length
   817→    ? deliverySummary
   818→        .map(
   819→          (d) => `<tr>
   820→        <td>${escapeHtml(d.poNumber)}</td>
   821→        <td>${escapeHtml(d.supplierName)}</td>
   822→        <td class="center">${statusPill(d.status)}</td>
   823→        <td class="num">${d.orderedQty}</td>

   837→        <td>${escapeHtml(b.bill.billNumber)}</td>

   961→async function buildSupplierSummary(supplierId: string): Promise<string> {

   968→          po: { select: { poNumber: true } },
   969→          brokerage: true,

   972→        orderBy: { createdAt: "desc" },

   976→          po: {

   978→              poNumber: true, totalValue: true, lineItemsJson: true,

   980→              client: { select: { name: true } },
   981→            },
   982→          },
   983→        },

   985→      },
   986→      brokerages: {
   987→        include: {
   988→          bill: { select: { billNumber: true, baseAmount: true } },
   989→          client: { select: { name: true } },
   990→          payout: true,
   991→        },
   992→        orderBy: { createdAt: "desc" },

   994→    },

   996→  if (!supplier) {

  1000→
  1001→  const totalSupplied = supplier.bills.reduce((s, b) => s + b.baseAmount, 0);
  1002→  const outstandingBrokerage = supplier.brokerages
  1003→    .filter((b) => b.eligible && b.payoutStatus !== "paid")
  1004→    .reduce((s, b) => s + b.brokerageAmount, 0);
  1005→  const paidBrokerage = supplier.brokerages

  1020→    } catch {

  1022→    }

  1695→// Maps an action string to a print-safe pill color (matches UI tone mapping).
  1696→function auditActionPill(action: string): string {
  1697→  const a = action.toLowerCase();
  1698→  if (a === "create" || a === "payout") {
  1699→    return `<span class="pill emerald">${escapeHtml(titleCase(action))}</span>`;
  1700→  }
  1701→  if (a === "update") {
  1702→    return `<span class="pill amber">${escapeHtml(titleCase(action))}</span>`;
  1703→  }
  1704→  if (a === "delete") {
  1705→    return `<span class="pill rose">${escapeHtml(titleCase(action))}</span>`;
  1706→  }
  1707→  if (a === "force_eligible") {
  1708→    return `<span class="pill emerald" style="background:#f0fdfa;color:#0f766e;border-color:#99f6e4;">${escapeHtml(titleCase(action))}</span>`;
  1709→  }
  1710→  return `<span class="pill zinc">${escapeHtml(titleCase(action))}</span>`;

  1713→async function buildAuditTrail(params: URLSearchParams): Promise<string> {

  1718→  // Note: SQLite (Prisma's provider here) does not support `mode: "insensitive"`.
  1719→  // String comparison uses the default BINARY collation (case-sensitive).
  1720→  // Stored entityType values are PascalCase and the UI dropdown uses matching
  1721→  // values, so case-sensitive equals is sufficient.

  1728→    const d = new Date(fromRaw);
  1729→    if (!Number.isNaN(d.getTime())) range.gte = d;

  1731→  if (toRaw) {
  1732→    const d = new Date(toRaw);
  1733→    if (!Number.isNaN(d.getTime())) {
  1734→      d.setHours(23, 59, 59, 999);
  1735→      range.lte = d;

  1737→  }
  1738→  if (range.gte || range.lte) {
  1739→    where.createdAt = range;

  1743→    db.auditLog.findMany({

  1749→    db.auditLog.groupBy({ by: ["entityType"], where, _count: { _all: true } }),
  1750→    db.auditLog.aggregate({
  1751→      where,
  1752→      _min: { createdAt: true },

  1757→  const byAction: Record<string, number> = {};

  1759→  const byEntity: Record<string, number> = {};

  1770→  } else if (range.gte) {
  1771→    rangeLabel = `From ${fmtDate(range.gte)}`;
  1772→  } else if (range.lte) {

  1782→  const createCount = byAction["create"] ?? 0;
  1783→  const updateCount = byAction["update"] ?? 0;
  1784→  const deleteCount = byAction["delete"] ?? 0;
  1785→  const forceCount = byAction["force_eligible"] ?? 0;
  1786→  const payoutCount = byAction["payout"] ?? 0;

  1916→export async function GET(req: NextRequest) {
  1917→  const { searchParams } = new URL(req.url);
  1918→  const typeParam = (searchParams.get("type") ?? "").toLowerCase() as ReportType;
  1919→  if (!VALID_TYPES.includes(typeParam)) {
  1920→    return NextResponse.json(
  1921→      { error: `Invalid report type. Valid: ${VALID_TYPES.join(", ")}` },
  1922→      { status: 400 },
  1923→    );
  1924→  }
  1925→
  1926→  try {
  1927→    let html: string;
  1928→    if (typeParam === "brokerage-statement") {
  1929→      const rawRange = (searchParams.get("range") ?? "all").toLowerCase() as Range;
  1930→      const range: Range = VALID_RANGES.includes(rawRange) ? rawRange : "all";
  1931→      html = await buildBrokerageStatement(range);
  1932→    } else if (typeParam === "client-ledger") {
  1933→      const clientId = searchParams.get("clientId");
  1934→      if (!clientId) {
  1935→        return NextResponse.json({ error: "clientId is required for client-ledger report" }, { status: 400 });
  1936→      }
  1937→      html = await buildClientLedger(clientId);
  1938→    } else if (typeParam === "supplier-summary") {
  1939→      const supplierId = searchParams.get("supplierId");
  1940→      if (!supplierId) {
  1941→        return NextResponse.json({ error: "supplierId is required for supplier-summary report" }, { status: 400 });
  1942→      }
  1943→      html = await buildSupplierSummary(supplierId);
  1944→    } else if (typeParam === "purchase-order") {
  1945→      const poId = searchParams.get("poId");
  1946→      if (!poId) {
  1947→        return NextResponse.json({ error: "poId is required for purchase-order report" }, { status: 400 });
  1948→      }
  1949→      html = await buildPurchaseOrder(poId);
  1950→    } else if (typeParam === "party-ledger") {
  1951→      const partyType = (searchParams.get("partyType") ?? "").toLowerCase();
  1952→      const partyId = searchParams.get("partyId");
  1953→      if (partyType !== "client" && partyType !== "supplier") {
  1954→        return NextResponse.json({ error: "partyType must be 'client' or 'supplier' for party-ledger report" }, { status: 400 });
  1955→      }
  1956→      if (!partyId) {
  1957→        return NextResponse.json({ error: "partyId is required for party-ledger report" }, { status: 400 });
  1958→      }
  1959→      html = await buildPartyLedger(partyType, partyId);
  1960→    } else {
  1961→      // audit-trail
  1962→      html = await buildAuditTrail(searchParams);
  1963→    }
  1964→
  1965→    return new NextResponse(html, {
  1966→      status: 200,
  1967→      headers: {
  1968→        "Content-Type": "text/html; charset=utf-8",
  1969→        "Cache-Control": "no-store, no-cache, must-revalidate",
  1970→        "X-Content-Type-Options": "nosniff",
  1971→      },
  1972→    });
  1973→  } catch (e) {
  1974→    const message = e instanceof Error ? e.message : "Unknown error";
  1975→    return NextResponse.json({ error: `Report failed: ${message}` }, { status: 500 });
  1976→  }
  1977→}
