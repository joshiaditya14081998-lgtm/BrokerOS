    38→function fmtCurrency(n: number): string {
    39→  const v = Number.isFinite(n) ? n : 0;
    40→  return new Intl.NumberFormat("en-IN", {
    41→    style: "currency",
    42→    currency: "INR",
    43→    maximumFractionDigits: 0,
    44→  }).format(v);
    45→}

    47→function fmtDate(d: Date | string | null | undefined): string {
    48→  if (!d) return "—";
    49→  const date = typeof d === "string" ? new Date(d) : d;
    50→  return date.toLocaleDateString("en-IN", {
    51→    day: "2-digit",
    52→    month: "short",
    53→    year: "numeric",
    54→  });
    55→}
    56→
    57→function fmtDateTime(d: Date | string | null | undefined): string {
    58→  if (!d) return "—";
    59→  const date = typeof d === "string" ? new Date(d) : d;
    60→  return date.toLocaleString("en-IN", {
    61→    day: "2-digit",
    62→    month: "short",
    63→    year: "numeric",
    64→    hour: "2-digit",
    65→    minute: "2-digit",
    66→  });
    67→}

    69→function titleCase(s: string): string {
    70→  return s.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase());
    71→}
    72→
    73→function escapeHtml(s: string | null | undefined): string {
    74→  if (s == null) return "";
    75→  return String(s)
    76→    .replace(/&/g, "&amp;")
    77→    .replace(/</g, "&lt;")
    78→    .replace(/>/g, "&gt;")
    79→    .replace(/"/g, "&quot;")
    80→    .replace(/'/g, "&#39;");
    81→}

    83→// ── Range bounds (reused from dashboard route) ───────────────────────────────
    84→
    85→function getRangeBounds(range: Range): { start: Date | undefined; end: Date; label: string } {
    86→  const now = new Date();
    87→  if (range === "all") {
    88→    return { start: undefined, end: now, label: "All time" };
    89→  }
    90→  const y = now.getFullYear();
    91→  const m = now.getMonth();
    92→  let start: Date;
    93→  let label: string;
    94→  if (range === "month") {
    95→    start = new Date(y, m, 1);
    96→    label = start.toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    97→  } else if (range === "quarter") {
    98→    const qStartMonth = Math.floor(m / 3) * 3;
    99→    start = new Date(y, qStartMonth, 1);
   100→    const qNum = Math.floor(qStartMonth / 3) + 1;
   101→    label = `Q${qNum} ${y}`;
   102→  } else {
   103→    start = new Date(y, 0, 1);
   104→    label = `Year ${y}`;
   105→  }
   106→  return { start, end: now, label };
   107→}

   488→function statusPill(status: string): string {
   489→  const s = status.toLowerCase();
   490→  let cls = "zinc";
   491→  if (["fully_paid", "fully_delivered", "delivered", "resolved", "occurred", "done", "paid", "scheduled"].includes(s)) {
   492→    cls = "emerald";
   493→  } else if (["partially_paid", "partially_delivered", "in_transit", "accrued"].includes(s)) {
   494→    cls = "amber";
   495→  } else if (["pending", "open", "no_show", "short_shipment", "defective_return", "overdue"].includes(s)) {
   496→    cls = "rose";
   497→  } else if (["closed", "rejected", "dismissed"].includes(s)) {
   498→    cls = "zinc";
   499→  }
   500→  return `<span class="pill ${cls}">${escapeHtml(titleCase(status))}</span>`;
   501→}
   502→
   503→function eligiblePill(eligible: boolean, forceEligible?: boolean): string {
   504→  if (!eligible) return `<span class="pill rose">No</span>`;
   505→  return forceEligible
   506→    ? `<span class="pill amber">Forced</span>`
   507→    : `<span class="pill emerald">Yes</span>`;
   508→}

   510→// ── Brokerage statement report ───────────────────────────────────────────────

   512→async function buildBrokerageStatement(range: Range): Promise<string> {
   513→  const { start, end, label } = getRangeBounds(range);
   514→
   515→  const brokerages = await db.brokerage.findMany({
   516→    where: start
   517→      ? {
   518→          OR: [
   519→            { eligibleAt: { gte: start, lte: end } },
   520→            { eligibleAt: null, createdAt: { gte: start, lte: end } },
   521→          ],
   522→        }
   523→      : undefined,
   524→    include: {
   525→      bill: { select: { billNumber: true, baseAmount: true, status: true, po: { select: { poNumber: true } } } },
   526→      client: { select: { name: true } },
   527→      supplier: { select: { name: true } },
   528→      payout: { select: { id: true, status: true, paidAt: true } },
   529→    },
   530→    orderBy: { eligibleAt: "asc" },
   531→  });
   532→
   533→  const payouts = await db.brokeragePayout.findMany({
   534→    where: start ? { createdAt: { gte: start, lte: end } } : undefined,
   535→    include: { client: { select: { name: true } }, brokerages: { select: { id: true, brokerageAmount: true } } },
   536→    orderBy: { createdAt: "desc" },
   537→  });
   538→
   539→  const totalBrokerage = brokerages.reduce((s, b) => s + b.brokerageAmount, 0);
   540→  const accruedTotal = brokerages
   541→    .filter((b) => b.eligible && b.payoutStatus === "accrued")
   542→    .reduce((s, b) => s + b.brokerageAmount, 0);
   543→  const scheduledTotal = brokerages
   544→    .filter((b) => b.eligible && b.payoutStatus === "scheduled")
   545→    .reduce((s, b) => s + b.brokerageAmount, 0);
   546→  const paidTotal = brokerages
   547→    .filter((b) => b.payoutStatus === "paid")
   548→    .reduce((s, b) => s + b.brokerageAmount, 0);
   549→  const pendingTotal = brokerages
   550→    .filter((b) => !b.eligible)
   551→    .reduce((s, b) => s + b.brokerageAmount, 0);
   552→  const billsCount = new Set(brokerages.map((b) => b.billId)).size;
   553→
   554→  // Group by client for the detail table (multiple sub-totals)
   555→  const byClient = new Map<string, { name: string; rows: typeof brokerages }>();
   556→  for (const b of brokerages) {
   557→    const key = b.clientId;
   558→    if (!byClient.has(key)) {
   559→      byClient.set(key, { name: b.client.name, rows: [] });
   560→    }
   561→    byClient.get(key)!.rows.push(b);
   562→  }
   563→  // Sort clients by name
   564→  const clientGroups = Array.from(byClient.entries())
   565→    .map(([id, g]) => ({ id, name: g.name, rows: g.rows }))
   566→    .sort((a, b) => a.name.localeCompare(b.name));
   567→
   568→  // Build the detail table — grouped by client with sub-totals
   569→  const detailRows: string[] = [];
   570→  let grandBase = 0;
   571→  let grandBrokerage = 0;
   572→  for (const group of clientGroups) {
   573→    detailRows.push(
   574→      `<tr class="group-header"><td colspan="9">${escapeHtml(group.name)} — ${group.rows.length} entr${group.rows.length === 1 ? "y" : "ies"}</td></tr>`,
   575→    );
   576→    let subBase = 0;
   577→    let subBrokerage = 0;
   578→    for (const b of group.rows) {
   579→      subBase += b.baseAmount;
   580→      subBrokerage += b.brokerageAmount;
   581→      grandBase += b.baseAmount;
   582→      grandBrokerage += b.brokerageAmount;
   583→      detailRows.push(
   584→        `<tr>
   585→          <td>${escapeHtml(b.bill.billNumber)}</td>
   586→          <td>${escapeHtml(b.client.name)}</td>
   587→          <td>${escapeHtml(b.supplier.name)}</td>
   588→          <td class="num">${fmtCurrency(b.baseAmount)}</td>
   589→          <td class="num">${b.commissionRate.toFixed(1)}%</td>
   590→          <td class="num emerald">${fmtCurrency(b.brokerageAmount)}</td>
   591→          <td class="center">${eligiblePill(b.eligible, b.forceEligible)}</td>
   592→          <td class="center">${statusPill(b.payoutStatus)}</td>
   593→          <td>${fmtDate(b.eligibleAt)}</td>
   594→        </tr>`,
   595→      );
   596→    }
   597→    detailRows.push(
   598→      `<tr class="sub-total">
   599→        <td colspan="3">Sub-total — ${escapeHtml(group.name)}</td>
   600→        <td class="num">${fmtCurrency(subBase)}</td>
   601→        <td></td>
   602→        <td class="num">${fmtCurrency(subBrokerage)}</td>
   603→        <td colspan="3"></td>
   604→      </tr>`,
   605→    );
   606→  }
   607→
   608→  if (brokerages.length === 0) {
   609→    detailRows.push(
   610→      `<tr><td colspan="9" style="text-align:center;padding:24px;color:#a1a1aa;">No brokerage entries in this range.</td></tr>`,
   611→    );
   612→  }
   613→
   614→  // Payout summary table
   615→  const payoutRows = payouts.length
   616→    ? payouts
   617→        .map(
   618→          (p) => `<tr>
   619→        <td>${escapeHtml(p.client.name)}</td>
   620→        <td class="center">${escapeHtml(titleCase(p.cadence))}</td>
   621→        <td>${fmtDate(p.periodStart)}</td>
   622→        <td>${fmtDate(p.periodEnd)}</td>
   623→        <td class="num">${fmtCurrency(p.totalAmount)}</td>
   624→        <td class="center">${statusPill(p.status)}</td>
   625→        <td class="center">${p.paidAt ? fmtDate(p.paidAt) : "—"}</td>
   626→        <td class="num">${p.brokerages.length}</td>
   627→      </tr>`,
   628→        )
   629→        .join("")
   630→    : `<tr><td colspan="8" style="text-align:center;padding:24px;color:#a1a1aa;">No payouts in this range.</td></tr>`;
   631→
   632→  const body = `
   633→    <div class="kpi-grid">
   634→      <div class="kpi-card">
   635→        <div class="kpi-label">Total brokerage</div>
   636→        <div class="kpi-value emerald">${fmtCurrency(totalBrokerage)}</div>
   637→        <div class="kpi-sub">${brokerages.length} entries</div>
   638→      </div>
   639→      <div class="kpi-card">
   640→        <div class="kpi-label">Eligible pending</div>
   641→        <div class="kpi-value amber">${fmtCurrency(accruedTotal + scheduledTotal)}</div>
   642→        <div class="kpi-sub">Accrued + scheduled</div>
   643→      </div>
   644→      <div class="kpi-card">
   645→        <div class="kpi-label">Paid out</div>
   646→        <div class="kpi-value emerald">${fmtCurrency(paidTotal)}</div>
   647→        <div class="kpi-sub">${payouts.length} payout batches</div>
   648→      </div>
   649→      <div class="kpi-card">
   650→        <div class="kpi-label">Not eligible</div>
   651→        <div class="kpi-value rose">${fmtCurrency(pendingTotal)}</div>
   652→        <div class="kpi-sub">${billsCount} bills</div>
   653→      </div>
   654→    </div>
   655→
   656→    <div class="section">
   657→      <div class="section-title">Brokerage ledger — grouped by client</div>
   658→      <div class="section-desc">All brokerage entries in the selected range. Sub-totals per client; grand total at the foot.</div>
   659→      <table class="report">
   660→        <thead>
   661→          <tr>
   662→            <th>Bill</th>
   663→            <th>Client</th>
   664→            <th>Supplier</th>
   665→            <th class="num">Base (excl GST)</th>
   666→            <th class="num">Comm %</th>

   669→            <th class="center">Payout</th>
   670→            <th>Eligible At</th>
   671→          </tr>
   672→        </thead>
   673→        <tbody>
   674→          ${detailRows.join("")}
   675→        </tbody>
   676→        <tfoot>
   677→          <tr>
   678→            <td colspan="3">Grand total</td>
   679→            <td class="num">${fmtCurrency(grandBase)}</td>
   680→            <td></td>

   692→        <thead>

   695→            <th class="center">Cadence</th>

   723→async function buildClientLedger(clientId: string): Promise<string> {
   724→  const client = await db.client.findUnique({
   725→    where: { id: clientId },
   726→    include: {
   727→      bills: {
   728→        include: {
   729→          po: { select: { poNumber: true, supplier: { select: { name: true } } } },
   730→          payments: { orderBy: { date: "desc" } },
   731→          brokerage: true,
   732→        },
   733→        orderBy: { createdAt: "desc" },
   734→      },
   735→      brokerages: {
   736→        include: {
   737→          bill: { select: { billNumber: true } },
   738→          supplier: { select: { name: true } },
   739→          payout: true,
   740→        },
   741→        orderBy: { createdAt: "desc" },
   742→      },
   743→    },
   744→  });
   745→  if (!client) {
   746→    return htmlShell("Client Not Found", "—",
   747→      `<div style="padding:40px;text-align:center;color:#a1a1aa;">The requested client could not be found.</div>`);
   748→  }
   749→
   750→  // Build the ledger (bills + payments, running balance)
   751→  type Ledger = { date: Date; type: string; ref: string; debit: number; credit: number; balance: number };
   752→  const entries: Ledger[] = [];
   753→  for (const bill of client.bills) {
   754→    entries.push({ date: bill.createdAt, type: "Bill", ref: bill.billNumber, debit: bill.finalAmount, credit: 0, balance: 0 });
   755→    for (const p of bill.payments) {
   756→      entries.push({ date: p.date, type: "Payment", ref: p.reference || p.id.slice(-6), debit: 0, credit: p.amount, balance: 0 });
   757→    }
   758→  }
   759→  entries.sort((a, b) => a.date.getTime() - b.date.getTime());
   760→  let bal = 0;
   761→  for (const e of entries) {
   762→    bal += e.debit - e.credit;
   763→    e.balance = bal;
   764→  }
   765→  // Reverse to show newest first
   766→  entries.reverse();
   767→
   768→  const totalBusiness = client.bills.reduce((s, b) => s + b.finalAmount, 0);
   769→  const outstanding = client.bills.reduce((s, b) => s + (b.finalAmount - b.paidAmount), 0);
   770→  const brokerageEarned = client.brokerages.filter((b) => b.eligible).reduce((s, b) => s + b.brokerageAmount, 0);
   771→  const brokeragePaid = client.brokerages.filter((b) => b.payoutStatus === "paid").reduce((s, b) => s + b.brokerageAmount, 0);
   772→
   773→  // PO delivery summary
   774→  const pos = await db.purchaseOrder.findMany({
   775→    where: { clientId },
   776→    include: { dispatches: true, supplier: { select: { name: true } } },
   777→    orderBy: { createdAt: "desc" },
   778→  });
   779→  const deliverySummary = pos.map((p) => {
   780→    const dispatchedQty = p.dispatches.reduce((s, d) => s + d.dispatchedQty, 0);
   781→    let orderedQty = 0;
   782→    try {
   783→      orderedQty = (JSON.parse(p.lineItemsJson) as { setQty: number }[]).reduce((s, i) => s + i.setQty, 0);
   784→    } catch {
   785→      orderedQty = 0;
   786→    }
   787→    return {
   788→      poNumber: p.poNumber,
   789→      status: p.status,
   790→      ordered: p.totalValue,
   791→      orderedQty,
   792→      dispatchedQty,
   793→      fulfillment: orderedQty ? Math.round((dispatchedQty / orderedQty) * 100) : 0,
   794→      supplierName: p.supplier.name,
   795→      expectedDispatchDate: p.expectedDispatchDate,
   796→      revisedDispatchDate: p.revisedDispatchDate,
   797→    };
   798→  });
   799→
   800→  const ledgerRows = entries.length
   801→    ? entries
   802→        .map(
   803→          (e) => `<tr>
   804→        <td>${fmtDate(e.date)}</td>
   805→        <td>${escapeHtml(titleCase(e.type))}</td>
   806→        <td>${escapeHtml(e.ref)}</td>
   807→        <td class="num">${e.debit ? fmtCurrency(e.debit) : "—"}</td>
   808→        <td class="num emerald">${e.credit ? fmtCurrency(e.credit) : "—"}</td>
   809→        <td class="num">${fmtCurrency(e.balance)}</td>
   810→      </tr>`,
   811→        )
   812→        .join("")
   813→    : `<tr><td colspan="6" style="text-align:center;padding:24px;color:#a1a1aa;">No ledger entries yet.</td></tr>`;
   814→
   815→  const deliveryRows = deliverySummary.length
   816→    ? deliverySummary
   817→        .map(
   818→          (d) => `<tr>
   819→        <td>${escapeHtml(d.poNumber)}</td>
   820→        <td>${escapeHtml(d.supplierName)}</td>
   821→        <td class="center">${statusPill(d.status)}</td>
   822→        <td class="num">${d.orderedQty}</td>
   823→        <td class="num">${d.dispatchedQty}</td>
   824→        <td class="num">${d.fulfillment}%</td>
   825→        <td>${fmtDate(d.expectedDispatchDate)}</td>
   826→        <td>${d.revisedDispatchDate ? fmtDate(d.revisedDispatchDate) : "—"}</td>
   827→      </tr>`,
   828→        )
   829→        .join("")
   830→    : `<tr><td colspan="8" style="text-align:center;padding:24px;color:#a1a1aa;">No purchase orders yet.</td></tr>`;

   833→    ? client.brokerages
   834→        .map(
   835→          (b) => `<tr>
   836→        <td>${escapeHtml(b.bill.billNumber)}</td>
   837→        <td>${escapeHtml(b.supplier.name)}</td>
   838→        <td class="num">${fmtDate(b.createdAt)}</td>
   839→        <td class="num">${b.commissionRate.toFixed(1)}%</td>
   840→        <td class="num emerald">${fmtCurrency(b.brokerageAmount)}</td>

   844→      </tr>`,

   847→    : `<tr><td colspan="8" style="text-align:center;padding:24px;color:#a1a1aa;">No brokerage entries.</td></tr>`;
   848→

   850→

   852→    <div class="subject-card">
   853→      <h2>${escapeHtml(client.name)}</h2>

   861→        <div class="item"><div class="k">GST rate</div><div class="v">${client.gstRate}%</div></div>

  1378→// Maps an action string to a print-safe pill color (matches UI tone mapping).
  1379→function auditActionPill(action: string): string {
  1380→  const a = action.toLowerCase();
  1381→  if (a === "create" || a === "payout") {
  1382→    return `<span class="pill emerald">${escapeHtml(titleCase(action))}</span>`;
  1383→  }
  1384→  if (a === "update") {
  1385→    return `<span class="pill amber">${escapeHtml(titleCase(action))}</span>`;
  1386→  }
  1387→  if (a === "delete") {
  1388→    return `<span class="pill rose">${escapeHtml(titleCase(action))}</span>`;
  1389→  }
  1390→  if (a === "force_eligible") {
  1391→    return `<span class="pill emerald" style="background:#f0fdfa;color:#0f766e;border-color:#99f6e4;">${escapeHtml(titleCase(action))}</span>`;
  1392→  }
  1393→  return `<span class="pill zinc">${escapeHtml(titleCase(action))}</span>`;
  1394→}

  1396→async function buildAuditTrail(params: URLSearchParams): Promise<string> {
  1397→  const entityType = params.get("entityType")?.trim();
  1398→  const fromRaw = params.get("from");
  1399→  const toRaw = params.get("to");
  1400→
  1401→  // Note: SQLite (Prisma's provider here) does not support `mode: "insensitive"`.
  1402→  // String comparison uses the default BINARY collation (case-sensitive).
  1403→  // Stored entityType values are PascalCase and the UI dropdown uses matching
  1404→  // values, so case-sensitive equals is sufficient.
  1405→  const where: Prisma.AuditLogWhereInput = {};
  1406→  if (entityType) {
  1407→    where.entityType = entityType;
  1408→  }
  1409→  const range: { gte?: Date; lte?: Date } = {};
  1410→  if (fromRaw) {
  1411→    const d = new Date(fromRaw);
  1412→    if (!Number.isNaN(d.getTime())) range.gte = d;
  1413→  }
  1414→  if (toRaw) {
  1415→    const d = new Date(toRaw);
  1416→    if (!Number.isNaN(d.getTime())) {
  1417→      d.setHours(23, 59, 59, 999);
  1418→      range.lte = d;
  1419→    }
  1420→  }
  1421→  if (range.gte || range.lte) {
  1422→    where.createdAt = range;
  1423→  }
  1424→
  1425→  const [logs, byActionAgg, byEntityAgg, minMax] = await Promise.all([
  1426→    db.auditLog.findMany({
  1427→      where,
  1428→      orderBy: { createdAt: "desc" },
  1429→      take: 1000,
  1430→    }),
  1431→    db.auditLog.groupBy({ by: ["action"], where, _count: { _all: true } }),
  1432→    db.auditLog.groupBy({ by: ["entityType"], where, _count: { _all: true } }),
  1433→    db.auditLog.aggregate({
  1434→      where,
  1435→      _min: { createdAt: true },
  1436→      _max: { createdAt: true },
  1437→    }),
  1438→  ]);
  1439→
  1440→  const byAction: Record<string, number> = {};
  1441→  for (const g of byActionAgg) byAction[g.action] = g._count._all;
  1442→  const byEntity: Record<string, number> = {};
  1443→  for (const g of byEntityAgg) byEntity[g.entityType] = g._count._all;
  1444→
  1445→  const total = logs.length;
  1446→  const earliest = minMax._min.createdAt;
  1447→  const latest = minMax._max.createdAt;
  1448→
  1449→  // Build the date-range label shown in the report header.
  1450→  let rangeLabel: string;
  1451→  if (range.gte && range.lte) {
  1452→    rangeLabel = `${fmtDate(range.gte)} → ${fmtDate(range.lte)}`;
  1453→  } else if (range.gte) {
  1454→    rangeLabel = `From ${fmtDate(range.gte)}`;
  1455→  } else if (range.lte) {
  1456→    rangeLabel = `Up to ${fmtDate(range.lte)}`;
  1457→  } else {
  1458→    rangeLabel = "All time";
  1459→  }
  1460→  if (entityType) {
  1461→    rangeLabel += ` · ${entityType} only`;
  1462→  }
  1463→
  1464→  // KPI grid: total + per-action counts
  1465→  const createCount = byAction["create"] ?? 0;
  1466→  const updateCount = byAction["update"] ?? 0;
  1467→  const deleteCount = byAction["delete"] ?? 0;
  1468→  const forceCount = byAction["force_eligible"] ?? 0;
  1469→  const payoutCount = byAction["payout"] ?? 0;
  1470→
  1471→  // Action breakdown list (all actions observed, sorted by count desc)
  1472→  const actionRows = Object.entries(byAction)
  1473→    .sort((a, b) => b[1] - a[1])
  1474→    .map(
  1475→      ([action, n]) =>
  1476→        `<tr><td>${auditActionPill(action)}</td><td class="num">${n}</td><td class="num">${total ? Math.round((n / total) * 100) : 0}%</td></tr>`,
  1477→    )
  1478→    .join("") ||
  1479→    `<tr><td colspan="3" style="text-align:center;padding:24px;color:#a1a1aa;">No entries.</td></tr>`;
  1480→
  1481→  const entityRows = Object.entries(byEntity)
  1482→    .sort((a, b) => b[1] - a[1])
  1483→    .map(
  1484→      ([entity, n]) =>
  1485→        `<tr><td>${escapeHtml(titleCase(entity))}</td><td class="num">${n}</td><td class="num">${total ? Math.round((n / total) * 100) : 0}%</td></tr>`,
  1486→    )
  1487→    .join("") ||
  1488→    `<tr><td colspan="3" style="text-align:center;padding:24px;color:#a1a1aa;">No entries.</td></tr>`;
  1489→
  1490→  // Detail table — page break every 50 rows via a wrapper div with class
  1491→  // `section-break` on every 50th row group. Simpler approach: insert a
  1492→  // page-break row class on every 50th row.
  1493→  const PAGE_SIZE = 50;
  1494→  const detailRows = logs.length
  1495→    ? logs
  1496→        .map((l, i) => {
  1497→          const breakClass = i > 0 && i % PAGE_SIZE === 0 ? "page-break-row" : "";
  1498→          return `<tr class="${breakClass}">
  1499→            <td class="nowrap">${escapeHtml(fmtDateTime(l.createdAt))}</td>
  1500→            <td>${escapeHtml(l.userName ?? "System")}</td>
  1501→            <td>${escapeHtml(titleCase(l.entityType))}</td>
  1502→            <td>${auditActionPill(l.action)}</td>
  1503→            <td>${escapeHtml(l.reason ?? "—")}</td>
  1504→          </tr>`;
  1505→        })
  1506→        .join("")
  1507→    : `<tr><td colspan="5" style="text-align:center;padding:24px;color:#a1a1aa;">No audit entries match these filters.</td></tr>`;
  1508→
  1509→  const body = `
  1510→    <div class="kpi-grid">
  1511→      <div class="kpi-card">
  1512→        <div class="kpi-label">Total entries</div>
  1513→        <div class="kpi-value">${total}</div>
  1514→        <div class="kpi-sub">${range.gte || range.lte ? "In selected range" : "All time"}</div>
  1515→      </div>
  1516→      <div class="kpi-card">
  1517→        <div class="kpi-label">Created</div>
  1518→        <div class="kpi-value emerald">${createCount}</div>
  1519→        <div class="kpi-sub">New records</div>
  1520→      </div>
  1521→      <div class="kpi-card">
  1522→        <div class="kpi-label">Updated</div>
  1523→        <div class="kpi-value amber">${updateCount}</div>
  1524→        <div class="kpi-sub">Field edits</div>
  1525→      </div>
  1526→      <div class="kpi-card">
  1527→        <div class="kpi-label">Deleted</div>
  1528→        <div class="kpi-value rose">${deleteCount}</div>
  1529→        <div class="kpi-sub">Removals</div>
  1530→      </div>
  1531→      <div class="kpi-card">
  1532→        <div class="kpi-label">Force-eligible</div>
  1533→        <div class="kpi-value" style="color:#0f766e;">${forceCount}</div>
  1534→        <div class="kpi-sub">Manual overrides</div>
  1535→      </div>
  1536→      <div class="kpi-card">
  1537→        <div class="kpi-label">Payouts</div>
  1538→        <div class="kpi-value emerald">${payoutCount}</div>
  1539→        <div class="kpi-sub">Brokerage settled</div>
  1540→      </div>
  1541→      <div class="kpi-card">
  1542→        <div class="kpi-label">First entry</div>
  1543→        <div class="kpi-value" style="font-size:13px;">${earliest ? escapeHtml(fmtDate(earliest)) : "—"}</div>
  1544→        <div class="kpi-sub">${earliest ? escapeHtml(fmtDateTime(earliest)) : ""}</div>
  1545→      </div>
  1546→      <div class="kpi-card">
  1547→        <div class="kpi-label">Latest entry</div>
  1548→        <div class="kpi-value" style="font-size:13px;">${latest ? escapeHtml(fmtDate(latest)) : "—"}</div>
  1549→        <div class="kpi-sub">${latest ? escapeHtml(fmtDateTime(latest)) : ""}</div>
  1550→      </div>
  1551→    </div>
  1552→
  1553→    <div class="section">
  1554→      <div class="section-title">Breakdown by action</div>
  1555→      <table class="report">
  1556→        <thead>
  1557→          <tr><th>Action</th><th class="num">Count</th><th class="num">Share</th></tr>
  1558→        </thead>
  1559→        <tbody>${actionRows}</tbody>
  1560→      </table>
  1561→    </div>
  1562→
  1563→    <div class="section section-break">
  1564→      <div class="section-title">Breakdown by entity type</div>
  1565→      <table class="report">
  1566→        <thead>
  1567→          <tr><th>Entity</th><th class="num">Count</th><th class="num">Share</th></tr>
  1568→        </thead>
  1569→        <tbody>${entityRows}</tbody>
  1570→      </table>
  1571→    </div>
  1572→
  1573→    <div class="section section-break">
  1574→      <div class="section-title">Audit entries</div>
  1575→      <div class="section-desc">Every financial mutation is recorded below. Page breaks every ${PAGE_SIZE} entries.</div>
  1576→      <table class="report">
  1577→        <thead>
  1578→          <tr>
  1579→            <th>Time</th>
  1580→            <th>User</th>
  1581→            <th>Entity</th>
  1582→            <th>Action</th>
  1583→            <th>Reason</th>
  1584→          </tr>
  1585→        </thead>
  1586→        <tbody>${detailRows}</tbody>
  1587→      </table>
  1588→    </div>
  1589→  `;
  1590→
  1591→  const footerNote =
  1592→    "Immutable append-only log. Every financial mutation is recorded for dispute resolution and compliance.";
  1593→
  1594→  return htmlShell(REPORT_TITLES["audit-trail"], rangeLabel, body, footerNote);

  1599→export async function GET(req: NextRequest) {
  1600→  const { searchParams } = new URL(req.url);
  1601→  const typeParam = (searchParams.get("type") ?? "").toLowerCase() as ReportType;
  1602→  if (!VALID_TYPES.includes(typeParam)) {
  1603→    return NextResponse.json(
  1604→      { error: `Invalid report type. Valid: ${VALID_TYPES.join(", ")}` },
  1605→      { status: 400 },
  1606→    );
  1607→  }
  1608→
  1609→  try {
  1610→    let html: string;
  1611→    if (typeParam === "brokerage-statement") {
  1612→      const rawRange = (searchParams.get("range") ?? "all").toLowerCase() as Range;
  1613→      const range: Range = VALID_RANGES.includes(rawRange) ? rawRange : "all";
  1614→      html = await buildBrokerageStatement(range);
  1615→    } else if (typeParam === "client-ledger") {
  1616→      const clientId = searchParams.get("clientId");
  1617→      if (!clientId) {
  1618→        return NextResponse.json({ error: "clientId is required for client-ledger report" }, { status: 400 });
  1619→      }
  1620→      html = await buildClientLedger(clientId);
  1621→    } else if (typeParam === "supplier-summary") {
  1622→      const supplierId = searchParams.get("supplierId");
  1623→      if (!supplierId) {
  1624→        return NextResponse.json({ error: "supplierId is required for supplier-summary report" }, { status: 400 });
  1625→      }
  1626→      html = await buildSupplierSummary(supplierId);
  1627→    } else if (typeParam === "purchase-order") {
  1628→      const poId = searchParams.get("poId");
  1629→      if (!poId) {
  1630→        return NextResponse.json({ error: "poId is required for purchase-order report" }, { status: 400 });
  1631→      }
  1632→      html = await buildPurchaseOrder(poId);
  1633→    } else {
  1634→      // audit-trail
  1635→      html = await buildAuditTrail(searchParams);
  1636→    }
  1637→
  1638→    return new NextResponse(html, {
  1639→      status: 200,
  1640→      headers: {
  1641→        "Content-Type": "text/html; charset=utf-8",
  1642→        "Cache-Control": "no-store, no-cache, must-revalidate",
  1643→        "X-Content-Type-Options": "nosniff",
  1644→      },
  1645→    });
  1646→  } catch (e) {
  1647→    const message = e instanceof Error ? e.message : "Unknown error";
  1648→    return NextResponse.json({ error: `Report failed: ${message}` }, { status: 500 });
  1649→  }
  1650→}
