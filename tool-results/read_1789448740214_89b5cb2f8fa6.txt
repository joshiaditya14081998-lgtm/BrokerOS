   590→   590→// Saved Views — user-persisted filter combinations for list views
   591→   591→//
   592→   592→// Brokers often apply the same filter combos repeatedly (e.g. "VIP clients
   593→   593→// with outstanding > ₹50k", "Overdue dispatches", "Open disputes this week").
   594→   594→// A SavedView captures the filter state of a view as JSON so it can be
   595→   595→// re-applied with a single click from the SavedViewsBar or the Saved Views
   596→   596→// management page.

   598→   598→// `view` is a ViewKey string ("clients" | "suppliers" | "pos" | "bills" |
   599→   599→// "disputes" | "audit" | …). `filterJson` is the serialised filter state
   600→   600→// (shape varies per view — search query + active tags + status chip + date
   601→   601→// range, etc.). `entityType` is optional, set for tag-filtered views so the
   602→   602→// Saved Views page can group / describe them. `icon` is one of
   603→   603→// star|heart|pin|flag|bookmark (lucide icon names).

   610→   610→  name       String
   611→   611→  view       String   // ViewKey — clients | suppliers | pos | bills | disputes | audit | …
   612→   612→  filterJson String   // JSON of the filter state (shape varies per view)
   613→   613→  entityType String?  // optional — for tag-filtered views (Client | Supplier | PurchaseOrder)
   614→   614→  icon       String   @default("star") // star | heart | pin | flag | bookmark
   615→   615→  createdAt  DateTime @default(now())
   616→   616→  updatedAt  DateTime @updatedAt

   618→   618→  @@index([view])

   621→   621→// ─────────────────────────────────────────────────────────────────────────────
   622→   622→// Report Templates — user-defined custom report configurations
   623→   623→//
   624→   624→// Brokers have varied reporting needs — some want a weekly client performance
   625→   625→// summary, others a monthly supplier comparison, others a custom date-range
   626→   626→// brokerage breakdown. A ReportTemplate captures the user-defined report
   627→   627→// configuration (entity type, columns, filters, grouping, sorting, options)
   628→   628→// as JSON in `configJson` so the Custom Report Builder view can edit it and
   629→   629→// `/api/reports/custom?templateId=…` can render a print-optimized HTML PDF
   630→   630→// from it on demand.
   631→   631→//
   632→   632→// `type` is one of: summary | comparison | ledger | custom.
   633→   633→// `configJson` shape (see `src/lib/report-columns.ts` + ReportBuilderView):
   634→   634→//   {
   635→   635→//     entityType: "client"|"supplier"|"bill"|"payment"|"brokerage"|"dispatch"|"dispute",
   636→   636→//     columns: string[],                 // column keys from REPORT_COLUMNS[entityType]
   637→   637→//     filters: {
   638→   638→//       dateRange?: "all"|"week"|"month"|"quarter"|"year"|"custom",
   639→   639→//       fromDate?: string,               // ISO date — when dateRange === "custom"
   640→   640→//       toDate?: string,                 // ISO date — when dateRange === "custom"
   641→   641→//       status?: string,                 // entity-specific status (e.g. "fully_paid")
   642→   642→//       tagId?: string                   // restrict to entities with this Tag
   643→   643→//     },
   644→   644→//     groupBy?: string | null,           // column key to group rows by (sub-totals per group)
   645→   645→//     sortBy?: { field: string; direction: "asc"|"desc" },
   646→   646→//     title: string,                     // report heading (often === template name)
   647→   647→//     includeCharts: boolean             // embed an inline SVG bar chart in the PDF
   648→   648→//   }
   649→   649→// ─────────────────────────────────────────────────────────────────────────────
   650→   650→
   651→   651→model ReportTemplate {
   652→   652→  id          String   @id @default(cuid())
   653→   653→  brokerId      String
   654→   654→  broker        Broker   @relation(fields: [brokerId], references: [id], onDelete: Cascade)
   655→   655→  name        String
   656→   656→  description String?
   657→   657→  type        String   // summary | comparison | ledger | custom
   658→   658→  configJson  String   // JSON of the template config (see module doc)
   659→   659→  createdAt   DateTime @default(now())
   660→   660→  updatedAt   DateTime @updatedAt
   661→   661→
   662→   662→  @@index([type])
   663→   663→}
