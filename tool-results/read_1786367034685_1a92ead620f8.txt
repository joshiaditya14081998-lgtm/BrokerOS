    87→// ─────────────────────────────────────────────────────────────────────────────
    88→// ScoreRing — circular SVG progress for the supplier reliability score.
    89→// ─────────────────────────────────────────────────────────────────────────────
    90→function ScoreRing({ score, size = 48 }: { score: number; size?: number }) {
    91→  const radius = (size - 8) / 2;
    92→  const circumference = 2 * Math.PI * radius;
    93→  const offset = circumference - (score / 100) * circumference;
    94→  const color = score >= 85 ? COLORS.emerald
    95→    : score >= 70 ? COLORS.teal
    96→    : score >= 55 ? COLORS.amber
    97→    : COLORS.rose;
    98→  return (
    99→    <div className="relative" style={{ width: size, height: size }}>
   100→      <svg width={size} height={size} className="-rotate-90">
   101→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="oklch(0.7 0.02 160 / 0.2)" strokeWidth={4} />
   102→        <circle
   103→          cx={size / 2}
   104→          cy={size / 2}
   105→          r={radius}
   106→          fill="none"
   107→          stroke={color}
   108→          strokeWidth={4}
   109→          strokeDasharray={circumference}
   110→          strokeDashoffset={offset}
   111→          strokeLinecap="round"
   112→          className="transition-all duration-500"
   113→        />
   114→      </svg>
   115→      <span className="absolute inset-0 grid place-items-center text-xs font-semibold kpi-num" style={{ color }}>
   116→        {score}
   117→      </span>
   118→    </div>
   119→  );
   120→}
