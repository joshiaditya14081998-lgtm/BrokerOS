     6→     6→// GET /api/analytics — combined analytics view: supplier reliability,
     7→     7→// client credit exposure, brokerage forecast, business volume trends,
     8→     8→// and a deep forecasting section (6-mo brokerage with confidence interval,
     9→     9→// 12-mo seasonal trend analysis, client order velocity, supplier capacity).
    10→    10→//
    11→    11→// Optional query params:
    12→    12→//   ?tagId=X                  — filter all sections to entities tagged with X.
    13→    13→//   &entityType=Client|Supplier|PurchaseOrder
    14→    14→//                             — only filter that entity type; the other two
    15→    15→//                               remain unfiltered. If omitted, the tag filter
    16→    16→//                               applies to all three entity types.
    17→    17→//
    18→    18→// When a tag filter is active, the response includes `appliedFilter`
    19→    19→// ({ tagId, tagName, tagColor, entityType }) plus a `tagInsights` block with
    20→    20→// side-by-side "tagged vs all" comparisons the broker can use to size up the
    21→    21→// tagged segment's share of overall business.
    22→    22→//
    23→    23→// All values are computed on-read from existing tables (no denormalisation).
    24→    24→// ─────────────────────────────────────────────────────────────────────────────
    25→    25→
    26→    26→function tierFor(score: number): "excellent" | "good" | "average" | "needs attention" {
    27→    27→  if (score >= 85) return "excellent";
    28→    28→  if (score >= 70) return "good";
    29→    29→  if (score >= 55) return "average";
    30→    30→  return "needs attention";
    31→    31→}

    33→    33→// Parse line items JSON safely into a typed shape.
    34→    34→type LineItem = { styleName: string; color?: string | null; setQty: number; unitPrice: number; lineTotal?: number };
    35→    35→function parseLineItems(raw: string | null | undefined): LineItem[] {
    36→    36→  if (!raw) return [];
    37→    37→  try {
    38→    38→    const v = JSON.parse(raw);
    39→    39→    return Array.isArray(v) ? (v as LineItem[]) : [];
    40→    40→  } catch {
    41→    41→    return [];
    42→    42→  }
    43→    43→}

    45→    45→// Build N monthly buckets ending at the current month, oldest first.
    46→    46→function monthlyBuckets(count: number, now: Date): { label: string; start: Date; end: Date; monthIndex: number }[] {
    47→    47→  const out: { label: string; start: Date; end: Date; monthIndex: number }[] = [];
    48→    48→  for (let i = count - 1; i >= 0; i--) {
    49→    49→    const start = new Date(now.getFullYear(), now.getMonth() - i, 1);
    50→    50→    const end = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
    51→    51→    out.push({
    52→    52→      label: start.toLocaleDateString("en-IN", { month: "short" }) + (count >= 12 ? ` ${String(start.getFullYear()).slice(-2)}` : ""),
    53→    53→      start,
    54→    54→      end,
    55→    55→      monthIndex: start.getMonth(),
    56→    56→    });
    57→    57→  }
    58→    58→  return out;
    59→    59→}

    62→    62→// Indian garment-industry season mapping (month 0..11 → season key + label).
    63→    63→// festive  → Oct–Nov (Dussehra / Diwali peak)
    64→    64→// wedding  → Dec–Jan (wedding season)

    69→    69→type SeasonKey = "festive" | "wedding" | "summer" | "winter" | "normal";
    70→    70→function seasonForMonth(monthIndex: number): { key: SeasonKey; label: string } {
    71→    71→  switch (monthIndex) {
    72→    72→    case 11: return { key: "wedding", label: "Wedding season start" };
    73→    73→    case 0:  return { key: "wedding", label: "Wedding season peak" };
    74→    74→    case 2:  return { key: "festive", label: "Holi festive" };
    75→    75→    case 3:  return { key: "summer",  label: "Summer stock transition" };
    76→    76→    case 4:  return { key: "summer",  label: "Summer stock peak" };
    77→    77→    case 7:  return { key: "winter",  label: "Winter stock prep" };
    78→    78→    case 8:  return { key: "winter",  label: "Winter stock loading" };
    79→    79→    case 9:  return { key: "festive", label: "Festive season (Dussehra)" };
    80→    80→    case 10: return { key: "festive", label: "Festive season peak (Diwali)" };
    81→    81→    default: return { key: "normal",  label: "Regular season" };
    82→    82→  }
    83→    83→}

    85→    85→// ─────────────────────────────────────────────────────────────────────────────
    86→    86→// Tag filter resolution — given a tagId and an optional entityType scope,
    87→    87→// returns the matching set of entity IDs for each filtered entity type.
    88→    88→// When `entityType` is null, the filter applies to all three entity types.
    89→    89→// ─────────────────────────────────────────────────────────────────────────────
    90→    90→type TagFilter = {
    91→    91→  tagId: string;
    92→    92→  tagName: string;
    93→    93→  tagColor: string;
    94→    94→  // null = "all entity types"; otherwise a single type.
    95→    95→  entityType: TagEntityType | null;
    96→    96→  taggedClientIds: Set<string>;
    97→    97→  taggedSupplierIds: Set<string>;
    98→    98→  taggedPoIds: Set<string>;
    99→    99→};
   100→   100→
   101→   101→async function resolveTagFilter(tagId: string, entityType: TagEntityType | null): Promise<TagFilter | null> {
   102→   102→  const tag = await db.tag.findUnique({
   103→   103→    where: { id: tagId },
   104→   104→    select: { id: true, name: true, color: true },
   105→   105→  });
   106→   106→  if (!tag) return null;
   107→   107→
   108→   108→  const filterTypes: TagEntityType[] = entityType ? [entityType] : ["Client", "Supplier", "PurchaseOrder"];
   109→   109→  const where = { tagId, entityType: { in: filterTypes } };
   110→   110→  const rows = await db.entityTag.findMany({
   111→   111→    where,
   112→   112→    select: { entityType: true, entityId: true },
   113→   113→  });
   114→   114→  return {
   115→   115→    tagId: tag.id,
   116→   116→    tagName: tag.name,
   117→   117→    tagColor: tag.color,
   118→   118→    entityType,
   119→   119→    taggedClientIds: new Set(rows.filter((r) => r.entityType === "Client").map((r) => r.entityId)),
   120→   120→    taggedSupplierIds: new Set(rows.filter((r) => r.entityType === "Supplier").map((r) => r.entityId)),
   121→   121→    taggedPoIds: new Set(rows.filter((r) => r.entityType === "PurchaseOrder").map((r) => r.entityId)),
   122→   122→  };
   123→   123→}
   124→   124→
   125→   125→export async function GET(req: NextRequest) {
   126→   126→  const now = new Date();
   127→   127→  const url = req.nextUrl;
   128→   128→  const tagIdParam = url.searchParams.get("tagId");
   129→   129→  const entityTypeParam = url.searchParams.get("entityType");
   130→   130→  const entityType = entityTypeParam && isValidEntityType(entityTypeParam) ? entityTypeParam : null;
   131→   131→
   132→   132→  let tagFilter: TagFilter | null = null;
   133→   133→  if (tagIdParam) {
   134→   134→    tagFilter = await resolveTagFilter(tagIdParam, entityType);
   135→   135→    // If the tagId doesn't resolve to a real tag, we treat the request as
   136→   136→    // unfiltered rather than 404 — analytics should never hard-fail because
   137→   137→    // a stale URL/bookmark pointed at a deleted tag.
   138→   138→  }
   139→   139→
   140→   140→  // ── Pull all data we need in parallel ──────────────────────────────────────
   141→   141→  const [
   142→   142→    suppliers,
   143→   143→    clients,
   144→   144→    dispatches,
   145→   145→    pos,
   146→   146→    bills,
   147→   147→    payments,
   148→   148→    disputes,
   149→   149→    brokerages,
   150→   150→  ] = await Promise.all([
   151→   151→    db.supplier.findMany({ select: { id: true, name: true } }),
   152→   152→    db.client.findMany({ select: { id: true, name: true, payoutCadence: true } }),
   153→   153→    db.dispatch.findMany({
   154→   154→      select: {
   155→   155→        id: true, supplierId: true, poId: true, dispatchDate: true, dispatchedQty: true,
   156→   156→        status: true, po: { select: { totalValue: true, lineItemsJson: true, expectedDispatchDate: true, revisedDispatchDate: true, createdAt: true } },
   157→   157→      },
   158→   158→    }),
   159→   159→    db.purchaseOrder.findMany({
   160→   160→      select: { id: true, supplierId: true, clientId: true, lineItemsJson: true, totalValue: true, createdAt: true, status: true },
   161→   161→    }),
   162→   162→    db.bill.findMany({
   163→   163→      select: { id: true, clientId: true, supplierId: true, poId: true, finalAmount: true, paidAmount: true, status: true, createdAt: true },
   164→   164→    }),
   165→   165→    db.payment.findMany({ select: { id: true, billId: true, date: true, createdAt: true } }),
   166→   166→    db.dispute.findMany({ select: { id: true, poId: true, type: true, createdAt: true } }),
   167→   167→    db.brokerage.findMany({
   168→   168→      select: {
   169→   169→        id: true, clientId: true, supplierId: true, billId: true, brokerageAmount: true,
   170→   170→        eligible: true, payoutStatus: true, eligibleAt: true, createdAt: true,
   171→   171→        bill: { select: { status: true, finalAmount: true, paidAmount: true } },
   172→   172→      },
   173→   173→    }),
   174→   174→  ]);
   175→   175→
   176→   176→  // Quick lookup maps (pre-filter).
   177→   177→  const poById = new Map(pos.map((p) => [p.id, p]));
   178→   178→  const allClientIds = new Set(clients.map((c) => c.id));
   179→   179→  const allSupplierIds = new Set(suppliers.map((s) => s.id));
   180→   180→  const allPoIds = new Set(pos.map((p) => p.id));
   181→   181→
   182→   182→  // ── Apply tag filter to the main collections ──────────────────────────────
   183→   183→  // The filter strategy mirrors the task spec:
   184→   184→  //   • Clients → only tagged clients (when Client filter active).
   185→   185→  //   • Suppliers → only tagged suppliers (when Supplier filter active).
   186→   186→  //   • POs → tagged POs OR POs whose client/supplier is tagged
   187→   187→  //           (when PO filter active, OR when client/supplier filter is active
   188→   188→  //            and broker wants PO-level visibility into that segment).
   189→   189→  // When entityType is null, all three filters apply simultaneously; when
   190→   190→  // entityType is set, only that filter applies and the other collections stay
   191→   191→  // unfiltered.
   192→   192→  let fSuppliers = suppliers;
   193→   193→  let fClients = clients;
   194→   194→  let fPos = pos;
   195→   195→  let fDispatches = dispatches;
   196→   196→  let fBills = bills;
   197→   197→  let fPayments = payments;
   198→   198→  let fDisputes = disputes;
   199→   199→  let fBrokerages = brokerages;
   200→   200→
   201→   201→  if (tagFilter) {
   202→   202→    const { taggedClientIds, taggedSupplierIds, taggedPoIds } = tagFilter;
   203→   203→
   204→   204→    if (tagFilter.entityType === null || tagFilter.entityType === "Supplier") {
   205→   205→      fSuppliers = suppliers.filter((s) => taggedSupplierIds.has(s.id));
   206→   206→    }
   207→   207→    if (tagFilter.entityType === null || tagFilter.entityType === "Client") {
   208→   208→      fClients = clients.filter((c) => taggedClientIds.has(c.id));
   209→   209→    }
   210→   210→
   211→   211→    // PO filtering — tag + cascade. A PO survives if:
   212→   212→    //   - the PO itself is tagged (when PO filter active), OR
   213→   213→    //   - its client is tagged (when Client filter active), OR
   214→   214→    //   - its supplier is tagged (when Supplier filter active).
   215→   215→    if (tagFilter.entityType === null) {
   216→   216→      fPos = pos.filter((p) =>
   217→   217→        taggedPoIds.has(p.id) || taggedClientIds.has(p.clientId) || taggedSupplierIds.has(p.supplierId),
   218→   218→      );
   219→   219→    } else if (tagFilter.entityType === "PurchaseOrder") {
   220→   220→      fPos = pos.filter((p) => taggedPoIds.has(p.id));
   221→   221→    } else if (tagFilter.entityType === "Client") {
   222→   222→      fPos = pos.filter((p) => taggedClientIds.has(p.clientId));
   223→   223→    } else if (tagFilter.entityType === "Supplier") {
   224→   224→      fPos = pos.filter((p) => taggedSupplierIds.has(p.supplierId));
   225→   225→    }
   226→   226→
   227→   227→    // Cascade: keep only dispatches/bills/payments/disputes/brokerages that
   228→   228→    // reference a surviving PO, client, or supplier.
   229→   229→    const survivingPoIds = new Set(fPos.map((p) => p.id));
   230→   230→    const survivingClientIds = new Set(fClients.map((c) => c.id));
   231→   231→    const survivingSupplierIds = new Set(fSuppliers.map((s) => s.id));
   232→   232→
   233→   233→    fDispatches = dispatches.filter((d) => survivingPoIds.has(d.poId));
   234→   234→    fDisputes = disputes.filter((d) => survivingPoIds.has(d.poId));
   235→   235→
   236→   236→    // Bills carry their own clientId/supplierId/poId — keep any that match a
   237→   237→    // surviving entity (PO, client, or supplier depending on the active filter).
   238→   238→    fBills = bills.filter((b) => {
   239→   239→      if (survivingPoIds.has(b.poId)) return true;
   240→   240→      if (tagFilter!.entityType === null || tagFilter!.entityType === "Client") {
   241→   241→        if (survivingClientIds.has(b.clientId)) return true;
   242→   242→      }
   243→   243→      if (tagFilter!.entityType === null || tagFilter!.entityType === "Supplier") {
   244→   244→        if (survivingSupplierIds.has(b.supplierId)) return true;
   245→   245→      }
   246→   246→      return false;
   247→   247→    });
   248→   248→
   249→   249→    const survivingBillIds = new Set(fBills.map((b) => b.id));
   250→   250→    fPayments = payments.filter((p) => survivingBillIds.has(p.billId));
   251→   251→
   252→   252→    // Brokerages have their own clientId/supplierId/billId linkage.
   253→   253→    fBrokerages = brokerages.filter((b) => {
   254→   254→      if (survivingBillIds.has(b.billId)) return true;
   255→   255→      if (tagFilter!.entityType === null || tagFilter!.entityType === "Client") {
   256→   256→        if (survivingClientIds.has(b.clientId)) return true;
   257→   257→      }
   258→   258→      if (tagFilter!.entityType === null || tagFilter!.entityType === "Supplier") {
   259→   259→        if (survivingSupplierIds.has(b.supplierId)) return true;
   260→   260→      }
   261→   261→      return false;
   262→   262→    });
   263→   263→  }
   264→   264→
   265→   265→  // Quick lookup maps (post-filter)
   266→   266→  const billsByClient = new Map<string, typeof fBills>();
   267→   267→  for (const b of fBills) {
   268→   268→    const arr = billsByClient.get(b.clientId) ?? [];
   269→   269→    arr.push(b);
   270→   270→    billsByClient.set(b.clientId, arr);
   271→   271→  }
   272→   272→  const disputesByPo = new Map<string, typeof fDisputes>();
   273→   273→  for (const d of fDisputes) {
   274→   274→    const arr = disputesByPo.get(d.poId) ?? [];
   275→   275→    arr.push(d);
   276→   276→    disputesByPo.set(d.poId, arr);
   277→   277→  }
   278→   278→  const posBySupplier = new Map<string, typeof fPos>();
   279→   279→  for (const p of fPos) {
   280→   280→    const arr = posBySupplier.get(p.supplierId) ?? [];
   281→   281→    arr.push(p);
   282→   282→    posBySupplier.set(p.supplierId, arr);
   283→   283→  }
   284→   284→  const dispatchesBySupplier = new Map<string, typeof fDispatches>();
   285→   285→  for (const d of fDispatches) {
   286→   286→    const arr = dispatchesBySupplier.get(d.supplierId) ?? [];
   287→   287→    arr.push(d);
   288→   288→    dispatchesBySupplier.set(d.supplierId, arr);
   289→   289→  }
   290→   290→
   291→   291→  // ═══════════════════════════════════════════════════════════════════════════
   292→   292→  // (a) Supplier reliability scores
   293→   293→  // ═══════════════════════════════════════════════════════════════════════════
   294→   294→  const supplierScores = fSuppliers.map((s) => {
   295→   295→    const sDispatches = dispatchesBySupplier.get(s.id) ?? [];
   296→   296→    const sPos = posBySupplier.get(s.id) ?? [];
   297→   297→
   298→   298→    // Fulfillment rate: avg dispatchedQty/orderedQty across dispatches.
   299→   299→    // No dispatches → treat as 100 (nothing to fulfil, no failures yet).
   300→   300→    let fulfillment = 100;
   301→   301→    let totalSupplied = 0;
   302→   302→    if (sDispatches.length > 0) {
   303→   303→      let sumRatio = 0;
   304→   304→      for (const d of sDispatches) {
   305→   305→        const ordered = parseLineItems(d.po.lineItemsJson).reduce((a, i) => a + (i.setQty || 0), 0);
   306→   306→        const ratio = ordered > 0 ? Math.min(1, (d.dispatchedQty || 0) / ordered) : 0;
   307→   307→        sumRatio += ratio;
   308→   308→        // Approximate supplied value: dispatched qty share × PO total
   309→   309→        const poValue = d.po.totalValue || 0;
   310→   310→        const lineShare = ordered > 0 ? (d.dispatchedQty || 0) / ordered : 0;
   311→   311→        totalSupplied += poValue * lineShare;
   312→   312→      }
   313→   313→      fulfillment = Math.round((sumRatio / sDispatches.length) * 100);
   314→   314→    }
   315→   315→
   316→   316→    // On-time dispatch: % of dispatches where dispatchDate <= expectedDispatchDate
   317→   317→    // (use revisedDispatchDate if present). No dispatches/dates → 100.
   318→   318→    let onTimeRate = 100;
   319→   319→    if (sDispatches.length > 0) {
   320→   320→      const withDates = sDispatches.filter((d) => d.po.expectedDispatchDate || d.po.revisedDispatchDate);
   321→   321→      if (withDates.length > 0) {
   322→   322→        const onTime = withDates.filter((d) => {
   323→   323→          const expected = d.po.revisedDispatchDate ?? d.po.expectedDispatchDate;
   324→   324→          return expected && new Date(d.dispatchDate) <= new Date(expected);
   325→   325→        }).length;
   326→   326→        onTimeRate = Math.round((onTime / withDates.length) * 100);
   327→   327→      }
   328→   328→    }
   329→   329→
   330→   330→    // Short-shipment rate: % of dispatches marked short_shipment
   331→   331→    const shortShipmentRate = sDispatches.length > 0
   332→   332→      ? Math.round((sDispatches.filter((d) => d.status === "short_shipment").length / sDispatches.length) * 100)
   333→   333→      : 0;
   334→   334→
   335→   335→    // Dispute rate: disputes linked to their POs / total POs × 100
   336→   336→    let disputeRate = 0;
   337→   337→    if (sPos.length > 0) {
   338→   338→      const dispCount = sPos.reduce((acc, p) => acc + (disputesByPo.get(p.id)?.length ?? 0), 0);
   339→   339→      disputeRate = Math.round((dispCount / sPos.length) * 100);
   340→   340→    }
   341→   341→
   342→   342→    // Composite score (0–100)
   343→   343→    //   fulfillment (40) + onTime (30) + (100 - shortShip) (20) + (100 - dispute) (10)
   344→   344→    const score = Math.round(
   345→   345→      (fulfillment * 0.4) + (onTimeRate * 0.3) + ((100 - shortShipmentRate) * 0.2) + ((100 - disputeRate) * 0.1),
   346→   346→    );
   347→   347→
   348→   348→    return {
   349→   349→      supplierId: s.id,
   350→   350→      name: s.name,
   351→   351→      score: Math.max(0, Math.min(100, score)),
   352→   352→      fulfillment,
   353→   353→      onTimeRate,
   354→   354→      shortShipmentRate,
   355→   355→      disputeRate,
   356→   356→      dispatchCount: sDispatches.length,
   357→   357→      totalSupplied: Math.round(totalSupplied),
   358→   358→      tier: tierFor(score),
   359→   359→    };
   360→   360→  });
   361→   361→  supplierScores.sort((a, b) => b.score - a.score);
   362→   362→
   363→   363→  // ═══════════════════════════════════════════════════════════════════════════
   364→   364→  // (b) Client credit exposure
   365→   365→  // ═══════════════════════════════════════════════════════════════════════════
   366→   366→  const exposureBuckets = monthlyBuckets(6, now);
   367→   367→  // Group payments by bill for delay computation
   368→   368→  const paymentsByBill = new Map<string, typeof fPayments>();
   369→   369→  for (const p of fPayments) {
   370→   370→    const arr = paymentsByBill.get(p.billId) ?? [];
   371→   371→    arr.push(p);
   372→   372→    paymentsByBill.set(p.billId, arr);
   373→   373→  }
   374→   374→
   375→   375→  const clientExposure = fClients.map((c) => {
   376→   376→    const cBills = billsByClient.get(c.id) ?? [];
   377→   377→    const outstanding = cBills.reduce((s, b) => s + (b.finalAmount - b.paidAmount), 0);
   378→   378→
   379→   379→    // Exposure trend: outstanding snapshot at end of each of last 6 months.
   380→   380→    // Approximation: include only bills created before month end; use current
   381→   381→    // paidAmount as the running-paid value (no payment history snapshot).
   382→   382→    const exposureTrend = exposureBuckets.map((m) => {
   383→   383→      const val = cBills
   384→   384→        .filter((b) => new Date(b.createdAt) <= m.end)
   385→   385→        .reduce((s, b) => s + (b.finalAmount - b.paidAmount), 0);
   386→   386→      return { month: m.label, value: Math.round(val) };
   387→   387→    });
   388→   388→
   389→   389→    // Avg payment delay (days) for fully-paid bills: createdAt → last payment date
   390→   390→    let avgPaymentDelay = 0;
   391→   391→    const paidBills = cBills.filter((b) => b.status === "fully_paid");
   392→   392→    if (paidBills.length > 0) {
   393→   393→      let totalDelay = 0;
   394→   394→      let counted = 0;
   395→   395→      for (const b of paidBills) {
   396→   396→        const bPays = paymentsByBill.get(b.id) ?? [];
   397→   397→        if (bPays.length === 0) continue;
   398→   398→        const lastPay = bPays.reduce((latest, p) => (new Date(p.date) > latest ? new Date(p.date) : latest), new Date(0));
   399→   399→        const diff = Math.round((lastPay.getTime() - new Date(b.createdAt).getTime()) / (1000 * 60 * 60 * 24));
   400→   400→        if (diff >= 0) {
   401→   401→          totalDelay += diff;
   402→   402→          counted += 1;
   403→   403→        }
   404→   404→      }
   405→   405→      avgPaymentDelay = counted > 0 ? Math.round(totalDelay / counted) : 0;
   406→   406→    }
   407→   407→
   408→   408→    // Return rate: defective_return disputes / total bills × 100
   409→   409→    // Disputes are linked via PO; need POs that belong to this client.
   410→   410→    const clientPoIds = new Set(fPos.filter((p) => p.clientId === c.id).map((p) => p.id));
   411→   411→    const defectiveReturns = fDisputes.filter((d) => d.type === "defective_return" && clientPoIds.has(d.poId)).length;
   412→   412→    const returnRate = cBills.length > 0 ? Math.round((defectiveReturns / cBills.length) * 100) : 0;
   413→   413→
   414→   414→    return {
   415→   415→      clientId: c.id,
   416→   416→      name: c.name,
   417→   417→      outstanding: Math.round(outstanding),
   418→   418→      exposureTrend,
   419→   419→      avgPaymentDelay,
   420→   420→      returnRate,
   421→   421→      billCount: cBills.length,
   422→   422→    };
   423→   423→  });
   424→   424→  clientExposure.sort((a, b) => b.outstanding - a.outstanding);
   425→   425→
   426→   426→  // ═══════════════════════════════════════════════════════════════════════════
   427→   427→  // (c) Brokerage forecast
   428→   428→  // ═══════════════════════════════════════════════════════════════════════════
   429→   429→  const pendingTotal = fBrokerages
   430→   430→    .filter((b) => !b.eligible)
   431→   431→    .reduce((s, b) => s + b.brokerageAmount, 0);
   432→   432→
   433→   433→  const eligibleUnpaid = fBrokerages.filter((b) => b.eligible && b.payoutStatus !== "paid");
   434→   434→  const eligibleUnpaidTotal = eligibleUnpaid.reduce((s, b) => s + b.brokerageAmount, 0);
   435→   435→
   436→   436→  // Breakdown by client cadence
   437→   437→  const clientCadence = new Map(fClients.map((c) => [c.id, c.payoutCadence]));
   438→   438→  const byCadence = { immediate: 0, "4_month": 0, "12_month": 0 };
   439→   439→  for (const b of eligibleUnpaid) {
   440→   440→    const cad = clientCadence.get(b.clientId) ?? "immediate";
   441→   441→    if (cad === "immediate") byCadence.immediate += b.brokerageAmount;
   442→   442→    else if (cad === "4_month_cumulative") byCadence["4_month"] += b.brokerageAmount;
   443→   443→    else if (cad === "12_month_cumulative") byCadence["12_month"] += b.brokerageAmount;
   444→   444→  }
   445→   445→
   446→   446→  // Projected next 3 months payout:
   447→   447→  //   immediate → entire amount in current month
   448→   448→  //   4_month  → 1/4 per month for next 4 months (so months 1,2,3 get 1/4 each)
   449→   449→  //   12_month → 1/12 per month for next 12 months (so months 1,2,3 get 1/12 each)
   450→   450→  const projectedByMonth: { month: string; amount: number }[] = [];
   451→   451→  for (let i = 0; i < 3; i++) {
   452→   452→    const mStart = new Date(now.getFullYear(), now.getMonth() + i, 1);
   453→   453→    const label = mStart.toLocaleDateString("en-IN", { month: "short" }) + ` ${String(mStart.getFullYear()).slice(-2)}`;
   454→   454→    const immediateShare = byCadence.immediate * (i === 0 ? 1 : 0);
   455→   455→    const fourShare = byCadence["4_month"] / 4;
   456→   456→    const twelveShare = byCadence["12_month"] / 12;
   457→   457→    projectedByMonth.push({
   458→   458→      month: label,
   459→   459→      amount: Math.round(immediateShare + fourShare + twelveShare),
   460→   460→    });
   461→   461→  }
   462→   462→
   463→   463→  const brokerageForecast = {
   464→   464→    pendingTotal: Math.round(pendingTotal),
   465→   465→    eligibleUnpaidTotal: Math.round(eligibleUnpaidTotal),
   466→   466→    projectedByMonth,
   467→   467→    byCadence: {
   468→   468→      immediate: Math.round(byCadence.immediate),
   469→   469→      "4_month": Math.round(byCadence["4_month"]),
   470→   470→      "12_month": Math.round(byCadence["12_month"]),
   471→   471→    },
   472→   472→  };
   473→   473→
   474→   474→  // ═══════════════════════════════════════════════════════════════════════════
   475→   475→  // (d) Business volume trends
   476→   476→  // ═══════════════════════════════════════════════════════════════════════════
   477→   477→  const volBuckets = monthlyBuckets(12, now);
   478→   478→  const monthlyBilled = volBuckets.map((m) => ({
   479→   479→    month: m.label,
   480→   480→    value: Math.round(
   481→   481→      fBills
   482→   482→        .filter((b) => new Date(b.createdAt) >= m.start && new Date(b.createdAt) < m.end)
   483→   483→        .reduce((s, b) => s + b.finalAmount, 0),
   484→   484→    ),
   485→   485→  }));
   486→   486→
   487→   487→  // Dispatched value per month: approximate each dispatch's value as
   488→   488→  // (dispatchedQty / orderedQty) × po.totalValue, bucketed by dispatchDate.
   489→   489→  const monthlyDispatched = volBuckets.map((m) => {
   490→   490→    let val = 0;
   491→   491→    for (const d of fDispatches) {
   492→   492→      const dd = new Date(d.dispatchDate);
   493→   493→      if (dd < m.start || dd >= m.end) continue;
   494→   494→      const ordered = parseLineItems(d.po.lineItemsJson).reduce((a, i) => a + (i.setQty || 0), 0);
   495→   495→      const share = ordered > 0 ? Math.min(1, (d.dispatchedQty || 0) / ordered) : 0;
   496→   496→      val += (d.po.totalValue || 0) * share;
   497→   497→    }
   498→   498→    return { month: m.label, value: Math.round(val) };
   499→   499→  });
   500→   500→
   501→   501→  // Top 5 client/supplier pairs by total business value (bill finalAmount)
   502→   502→  const pairMap = new Map<string, { clientName: string; supplierName: string; value: number }>();
   503→   503→  const clientName = new Map(fClients.map((c) => [c.id, c.name]));
   504→   504→  const supplierName = new Map(fSuppliers.map((s) => [s.id, s.name]));
   505→   505→  // Fall back to ALL clients/suppliers for naming — when a tag filter excludes
   506→   506→  // the client/supplier themselves, the pair may still reference them via a
   507→   507→  // surviving bill (because bill.supplierId matched the tagged PO segment).
   508→   508→  const clientNameAll = new Map(clients.map((c) => [c.id, c.name]));
   509→   509→  const supplierNameAll = new Map(suppliers.map((s) => [s.id, s.name]));
   510→   510→  for (const b of fBills) {
   511→   511→    const key = `${b.clientId}|${b.supplierId}`;
   512→   512→    const existing = pairMap.get(key) ?? {
   513→   513→      clientName: clientName.get(b.clientId) ?? clientNameAll.get(b.clientId) ?? "—",
   514→   514→      supplierName: supplierName.get(b.supplierId) ?? supplierNameAll.get(b.supplierId) ?? "—",
   515→   515→      value: 0,
   516→   516→    };
   517→   517→    existing.value += b.finalAmount;
   518→   518→    pairMap.set(key, existing);
   519→   519→  }
   520→   520→  const topPairs = Array.from(pairMap.values())
   521→   521→    .map((p) => ({ ...p, value: Math.round(p.value) }))
   522→   522→    .sort((a, b) => b.value - a.value)
   523→   523→    .slice(0, 5);
   524→   524→
   525→   525→  const volumeTrends = { monthlyBilled, monthlyDispatched, topPairs };
   526→   526→
   527→   527→  // ═══════════════════════════════════════════════════════════════════════════
   528→   528→  // (e) Deep forecasting — 6-mo brokerage w/ confidence interval,
   529→   529→  //     12-mo seasonal trend analysis, client order velocity,
   530→   530→  //     supplier capacity utilisation.
   531→   531→  // ═══════════════════════════════════════════════════════════════════════════
   532→   532→
   533→   533→  // ── (e.1) Historical eligible-brokerage monthly series (last 6 months) ──────
   534→   534→  const histBuckets = monthlyBuckets(6, now);
   535→   535→  const histMonthly = histBuckets.map((m) => {
   536→   536→    const val = fBrokerages
   537→   537→      .filter((b) => {
   538→   538→        if (!b.eligible) return false;
   539→   539→        const ts = b.eligibleAt ? new Date(b.eligibleAt) : new Date(b.createdAt);
   540→   540→        return ts >= m.start && ts < m.end;
   541→   541→      })
   542→   542→      .reduce((s, b) => s + b.brokerageAmount, 0);
   543→   543→    return { label: m.label, value: Math.round(val) };
   544→   544→  });
   545→   545→  const histValues = histMonthly.map((m) => m.value);
   546→   546→  const histMax = histValues.length > 0 ? Math.max(...histValues) : 0;
   547→   547→  const histMin = histValues.length > 0 ? Math.min(...histValues) : 0;
   548→   548→  const histAvg = histValues.length > 0 ? histValues.reduce((a, b) => a + b, 0) / histValues.length : 0;
   549→   549→
   550→   550→  // ── (e.2) 6-month brokerage forecast with confidence interval ──────────────
   551→   551→  const pendingShare = brokerageForecast.pendingTotal / 6;
   552→   552→  const cadenceImmediate = byCadence.immediate;
   553→   553→  const cadence4Share = byCadence["4_month"] / 4;
   554→   554→  const cadence12Share = byCadence["12_month"] / 12;
   555→   555→  const brokerage6Month: { month: string; projected: number; upper: number; lower: number }[] = [];
   556→   556→  for (let i = 0; i < 6; i++) {
   557→   557→    const mStart = new Date(now.getFullYear(), now.getMonth() + i, 1);
   558→   558→    const label = mStart.toLocaleDateString("en-IN", { month: "short" }) + ` ${String(mStart.getFullYear()).slice(-2)}`;
   559→   559→    const cadenceShare = (i === 0 ? cadenceImmediate : 0) + (i < 4 ? cadence4Share : 0) + cadence12Share;
   560→   560→    const projected = Math.round(pendingShare + cadenceShare + histAvg);
   561→   561→    const upper = Math.max(projected, Math.round(histMax * 1.2));
   562→   562→    const lower = Math.min(projected, Math.round(histMin * 0.8));
   563→   563→    brokerage6Month.push({ month: label, projected, upper, lower });
   564→   564→  }
   565→   565→
   566→   566→  // ── (e.3) 12-month seasonal trend analysis ─────────────────────────────────
   567→   567→  const seasonalTrends = volBuckets.map((m, i) => {
   568→   568→    const value = monthlyBilled[i]?.value ?? 0;
   569→   569→    const prev = i > 0 ? (monthlyBilled[i - 1]?.value ?? 0) : 0;
   570→   570→    const growthRate = prev > 0 ? Math.round(((value - prev) / prev) * 100) : (value > 0 ? 100 : 0);
   571→   571→    const season = seasonForMonth(m.monthIndex);
   572→   572→    return {
   573→   573→      month: m.label,
   574→   574→      value,
   575→   575→      growthRate,
   576→   576→      seasonLabel: season.label,
   577→   577→      seasonKey: season.key,
   578→   578→      isPeak: false,
   579→   579→      isLow: false,
   580→   580→    };
   581→   581→  });
   582→   582→  // Mark top-2 peaks (by value)
   583→   583→  const peakIndices = [...seasonalTrends.map((s, i) => ({ s, i }))]
   584→   584→    .sort((a, b) => b.s.value - a.s.value)
   585→   585→    .slice(0, 2)
   586→   586→    .filter((x) => x.s.value > 0)
   587→   587→    .map((x) => x.i);
   588→   588→  // Mark bottom-2 lows (only among non-zero months)
   589→   589→  const nonZero = seasonalTrends.map((s, i) => ({ s, i })).filter((x) => x.s.value > 0);
   590→   590→  const lowIndices = nonZero
   591→   591→    .sort((a, b) => a.s.value - b.s.value)
   592→   592→    .slice(0, 2)
   593→   593→    .map((x) => x.i);
   594→   594→  for (const i of peakIndices) seasonalTrends[i].isPeak = true;
   595→   595→  for (const i of lowIndices) seasonalTrends[i].isLow = true;
   596→   596→
   597→   597→  // ── (e.4) Client business velocity ─────────────────────────────────────────
   598→   598→  const posByClient = new Map<string, typeof fPos>();
   599→   599→  for (const p of fPos) {
   600→   600→    const arr = posByClient.get(p.clientId) ?? [];
   601→   601→    arr.push(p);
   602→   602→    posByClient.set(p.clientId, arr);
   603→   603→  }
   604→   604→  const clientVelocity = fClients
   605→   605→    .map((c) => {
   606→   606→      const cPos = (posByClient.get(c.id) ?? [])
   607→   607→        .map((p) => new Date(p.createdAt).getTime())
   608→   608→        .sort((a, b) => a - b);
   609→   609→      if (cPos.length === 0) {
   610→   610→        return {
   611→   611→          clientId: c.id,
   612→   612→          name: c.name,
   613→   613→          lastOrderDate: null as string | null,
   614→   614→          avgDaysBetweenOrders: 0,
   615→   615→          predictedNextOrderDate: null as string | null,
   616→   616→          orderCount: 0,
   617→   617→        };
   618→   618→      }
   619→   619→      const lastTs = cPos[cPos.length - 1];
   620→   620→      const lastOrderDate = new Date(lastTs).toISOString();
   621→   621→      let avgDays = 0;
   622→   622→      if (cPos.length >= 2) {
   623→   623→        let totalGap = 0;
   624→   624→        for (let i = 1; i < cPos.length; i++) totalGap += (cPos[i] - cPos[i - 1]) / (1000 * 60 * 60 * 24);
   625→   625→        avgDays = Math.round(totalGap / (cPos.length - 1));
   626→   626→      }
   627→   627→      const predictedTs = avgDays > 0 ? lastTs + avgDays * 24 * 60 * 60 * 1000 : null;
   628→   628→      return {
   629→   629→        clientId: c.id,
   630→   630→        name: c.name,
   631→   631→        lastOrderDate,
   632→   632→        avgDaysBetweenOrders: avgDays,
   633→   633→        predictedNextOrderDate: predictedTs ? new Date(predictedTs).toISOString() : null,
   634→   634→        orderCount: cPos.length,
   635→   635→      };
   636→   636→    })
   637→   637→    .filter((c) => c.orderCount > 0)
   638→   638→    .sort((a, b) => {
   639→   639→      // Sort by predicted-next-order-date proximity (closest first), nulls last.
   640→   640→      const aTs = a.predictedNextOrderDate ? new Date(a.predictedNextOrderDate).getTime() : Infinity;
   641→   641→      const bTs = b.predictedNextOrderDate ? new Date(b.predictedNextOrderDate).getTime() : Infinity;
   642→   642→      return aTs - bTs;
   643→   643→    })
   644→   644→    .slice(0, 5);
   645→   645→
   646→   646→  // ── (e.5) Supplier capacity utilisation ────────────────────────────────────
   647→   647→  const supplierCapacity = fSuppliers.map((s) => {
   648→   648→    const sDispatches = dispatchesBySupplier.get(s.id) ?? [];
   649→   649→    const sPos = posBySupplier.get(s.id) ?? [];
   650→   650→
   651→   651→    // Avg dispatch lead time (only counts dispatches with a valid po.createdAt)
   652→   652→    let avgLeadTimeDays = 0;
   653→   653→    const leadTimes: number[] = [];
   654→   654→    for (const d of sDispatches) {
   655→   655→      const poCreated = d.po.createdAt ? new Date(d.po.createdAt).getTime() : 0;
   656→   656→      if (!poCreated) continue;
   657→   657→      const diff = Math.round((new Date(d.dispatchDate).getTime() - poCreated) / (1000 * 60 * 60 * 24));
   658→   658→      if (diff >= 0) leadTimes.push(diff);
   659→   659→    }
   660→   660→    if (leadTimes.length > 0) {
   661→   661→      avgLeadTimeDays = Math.round(leadTimes.reduce((a, b) => a + b, 0) / leadTimes.length);
   662→   662→    }
   663→   663→
   664→   664→    // Pending POs = currently open or partially-delivered
   665→   665→    const pendingPOs = sPos.filter((p) => p.status === "open" || p.status === "partially_delivered").length;
   666→   666→
   667→   667→    // Historical monthly avg PO count over the last 12 months
   668→   668→    const twelveMonthsAgo = new Date(now.getFullYear(), now.getMonth() - 11, 1);
   669→   669→    const recentPOs = sPos.filter((p) => new Date(p.createdAt) >= twelveMonthsAgo).length;
   670→   670→    const monthlyAvgPOs = recentPOs / 12;
   671→   671→
   672→   672→    // Status thresholds:
   673→   673→    //   overloaded  — pendingPOs >= 2× monthlyAvgPOs AND pendingPOs >= 5
   674→   674→    //   high        — pendingPOs >  monthlyAvgPOs AND pendingPOs >= 3
   675→   675→    //   normal      — otherwise
   676→   676→    let status: "normal" | "high" | "overloaded" = "normal";
   677→   677→    if (pendingPOs >= 5 && monthlyAvgPOs > 0 && pendingPOs >= 2 * monthlyAvgPOs) status = "overloaded";
   678→   678→    else if (pendingPOs >= 3 && pendingPOs > monthlyAvgPOs) status = "high";
   679→   679→
   680→   680→    return {
   681→   681→      supplierId: s.id,
   682→   682→      name: s.name,
   683→   683→      avgLeadTimeDays,
   684→   684→      pendingPOs,
   685→   685→      monthlyAvgPOs: Math.round(monthlyAvgPOs * 10) / 10,
   686→   686→      status,
   687→   687→    };
   688→   688→  });
   689→   689→  // Sort by status severity (overloaded > high > normal), then by pending POs desc
   690→   690→  const statusRank: Record<string, number> = { overloaded: 0, high: 1, normal: 2 };
   691→   691→  supplierCapacity.sort((a, b) => {
   692→   692→    const r = statusRank[a.status] - statusRank[b.status];
   693→   693→    if (r !== 0) return r;
   694→   694→    return b.pendingPOs - a.pendingPOs;
   695→   695→  });
   696→   696→
   697→   697→  const forecast = {
   698→   698→    brokerage6Month,
   699→   699→    seasonalTrends,
   700→   700→    clientVelocity,
   701→   701→    supplierCapacity,
   702→   702→  };
   703→   703→
   704→   704→  // ═══════════════════════════════════════════════════════════════════════════
   705→   705→  // Tag insights — when a tag filter is active, compute a tagged-vs-all
   706→   706→  // comparison block. Always uses BOTH the filtered ("tagged") set and the
   707→   707→  // full unfiltered set of entities — the broker needs the side-by-side
   708→   708→  // view to size up the segment's share of overall business.
   709→   709→  // ═══════════════════════════════════════════════════════════════════════════
   710→   710→  let tagInsights: unknown = null;
   711→   711→  let appliedFilter: {
   712→   712→    tagId: string;
   713→   713→    tagName: string;
   714→   714→    tagColor: string;
   715→   715→    entityType: TagEntityType | null;
   716→   716→    clientCount: number;
   717→   717→    supplierCount: number;
   718→   718→    purchaseOrderCount: number;
   719→   719→  } | null = null;
   720→   720→
   721→   721→  if (tagFilter) {
   722→   722→    // Total counts for the tag — drawn from ALL entities of each type, not the
   723→   723→    // filtered subset (so the broker sees "this tag has 4 VIP clients out of 9
   724→   724→    // total clients" even when the entityType scope is "Client" only).
   725→   725→    const taggedClientCount = tagFilter.taggedClientIds.size;
   726→   726→    const taggedSupplierCount = tagFilter.taggedSupplierIds.size;
   727→   727→    const taggedPoCount = tagFilter.taggedPoIds.size;
   728→   728→
   729→   729→    appliedFilter = {
   730→   730→      tagId: tagFilter.tagId,
   731→   731→      tagName: tagFilter.tagName,
   732→   732→      tagColor: tagFilter.tagColor,
   733→   733→      entityType: tagFilter.entityType,
   734→   734→      clientCount: taggedClientCount,
   735→   735→      supplierCount: taggedSupplierCount,
   736→   736→      purchaseOrderCount: taggedPoCount,
   737→   737→    };
   738→   738→
   739→   739→    // ── Client comparison (computed when Client filter active OR all) ────────
   740→   740→    type ClientCompare = {
   741→   741→      taggedCount: number;
   742→   742→      totalCount: number;
   743→   743→      taggedBusiness: number;
   744→   744→      allBusiness: number;
   745→   745→      taggedOutstanding: number;
   746→   746→      allOutstanding: number;
   747→   747→      taggedBrokerage: number;
   748→   748→      allBrokerage: number;
   749→   749→      taggedAvgOutstanding: number;
   750→   750→      allAvgOutstanding: number;
   751→   751→    };
   752→   752→    let clientComparison: ClientCompare | null = null;
   753→   753→    const includeClients = tagFilter.entityType === null || tagFilter.entityType === "Client";
   754→   754→    if (includeClients) {
   755→   755→      const taggedClientIds = tagFilter.taggedClientIds;
   756→   756→      const taggedClientSet = new Set(taggedClientIds);
   757→   757→      // Tagged totals — using ALL bills/brokerages (unfiltered), not fBills,
   758→   758→      // so the comparison reflects the full segment size.
   759→   759→      let tBus = 0, tOut = 0, tBrok = 0;
   760→   760→      let aBus = 0, aOut = 0, aBrok = 0;
   761→   761→      const billsByClientAll = new Map<string, typeof bills>();
   762→   762→      for (const b of bills) {
   763→   763→        const arr = billsByClientAll.get(b.clientId) ?? [];
   764→   764→        arr.push(b);
   765→   765→        billsByClientAll.set(b.clientId, arr);
   766→   766→        aBus += b.finalAmount;
   767→   767→        aOut += (b.finalAmount - b.paidAmount);
   768→   768→      }
   769→   769→      for (const b of brokerages) {
   770→   770→        aBrok += b.brokerageAmount;
   771→   771→      }
   772→   772→      for (const cid of taggedClientSet) {
   773→   773→        const cb = billsByClientAll.get(cid) ?? [];
   774→   774→        for (const b of cb) {
   775→   775→          tBus += b.finalAmount;
   776→   776→          tOut += (b.finalAmount - b.paidAmount);
   777→   777→        }
   778→   778→        for (const b of brokerages) {
   779→   779→          if (b.clientId === cid) tBrok += b.brokerageAmount;
   780→   780→        }
   781→   781→      }
   782→   782→      clientComparison = {
   783→   783→        taggedCount: taggedClientCount,
   784→   784→        totalCount: allClientIds.size,
   785→   785→        taggedBusiness: Math.round(tBus),
   786→   786→        allBusiness: Math.round(aBus),
   787→   787→        taggedOutstanding: Math.round(tOut),
   788→   788→        allOutstanding: Math.round(aOut),
   789→   789→        taggedBrokerage: Math.round(tBrok),
   790→   790→        allBrokerage: Math.round(aBrok),
   791→   791→        taggedAvgOutstanding: taggedClientCount > 0 ? Math.round(tOut / taggedClientCount) : 0,
   792→   792→        allAvgOutstanding: allClientIds.size > 0 ? Math.round(aOut / allClientIds.size) : 0,
   793→   793→      };
   794→   794→    }
   795→   795→
   796→   796→    // ── Supplier comparison ──────────────────────────────────────────────────
   797→   797→    type SupplierCompare = {
   798→   798→      taggedCount: number;
   799→   799→      totalCount: number;
   800→   800→      taggedSupplied: number;
   801→   801→      allSupplied: number;
   802→   802→      taggedFulfillment: number;
   803→   803→      allFulfillment: number;
   804→   804→    };
   805→   805→    let supplierComparison: SupplierCompare | null = null;
   806→   806→    const includeSuppliers = tagFilter.entityType === null || tagFilter.entityType === "Supplier";
   807→   807→    if (includeSuppliers) {
   808→   808→      const taggedSupplierSet = new Set(tagFilter.taggedSupplierIds);
   809→   809→      // Build dispatches-by-supplier over ALL dispatches (unfiltered).
   810→   810→      const dispatchesBySupplierAll = new Map<string, typeof dispatches>();
   811→   811→      for (const d of dispatches) {
   812→   812→        const arr = dispatchesBySupplierAll.get(d.supplierId) ?? [];
   813→   813→        arr.push(d);
   814→   814→        dispatchesBySupplierAll.set(d.supplierId, arr);
   815→   815→      }
   816→   816→      const computeSupplierMetrics = (supplierIds: Iterable<string>) => {
   817→   817→        let supplied = 0;
   818→   818→        const ratios: number[] = [];
   819→   819→        for (const sid of supplierIds) {
   820→   820→          const ds = dispatchesBySupplierAll.get(sid) ?? [];
   821→   821→          for (const d of ds) {
   822→   822→            const ordered = parseLineItems(d.po.lineItemsJson).reduce((a, i) => a + (i.setQty || 0), 0);
   823→   823→            const ratio = ordered > 0 ? Math.min(1, (d.dispatchedQty || 0) / ordered) : 0;
   824→   824→            ratios.push(ratio);
   825→   825→            const poValue = d.po.totalValue || 0;
   826→   826→            const lineShare = ordered > 0 ? (d.dispatchedQty || 0) / ordered : 0;
   827→   827→            supplied += poValue * lineShare;
   828→   828→          }
   829→   829→        }
   830→   830→        const fulfillment = ratios.length > 0
   831→   831→          ? Math.round((ratios.reduce((a, b) => a + b, 0) / ratios.length) * 100)
   832→   832→          : 0;
   833→   833→        return { supplied: Math.round(supplied), fulfillment };
   834→   834→      };
   835→   835→      const taggedMetrics = computeSupplierMetrics(taggedSupplierSet);
   836→   836→      const allMetrics = computeSupplierMetrics(allSupplierIds);
   837→   837→      supplierComparison = {
   838→   838→        taggedCount: taggedSupplierCount,
   839→   839→        totalCount: allSupplierIds.size,
   840→   840→        taggedSupplied: taggedMetrics.supplied,
   841→   841→        allSupplied: allMetrics.supplied,
   842→   842→        taggedFulfillment: taggedMetrics.fulfillment,
   843→   843→        allFulfillment: allMetrics.fulfillment,
   844→   844→      };
   845→   845→    }
   846→   846→
   847→   847→    // ── PO comparison ─────────────────────────────────────────────────────────
   848→   848→    type PoCompare = {
   849→   849→      taggedCount: number;
   850→   850→      totalCount: number;
   851→   851→      taggedValue: number;
   852→   852→      allValue: number;
   853→   853→      taggedFulfillment: number;
   854→   854→      allFulfillment: number;
   855→   855→    };
   856→   856→    let poComparison: PoCompare | null = null;
   857→   857→    const includePos = tagFilter.entityType === null || tagFilter.entityType === "PurchaseOrder";
   858→   858→    if (includePos) {
   859→   859→      const taggedPoSet = new Set(tagFilter.taggedPoIds);
   860→   860→      // Build dispatches-by-po over ALL dispatches (unfiltered).
   861→   861→      const dispatchesByPoAll = new Map<string, typeof dispatches>();
   862→   862→      for (const d of dispatches) {
   863→   863→        const arr = dispatchesByPoAll.get(d.poId) ?? [];
   864→   864→        arr.push(d);
   865→   865→        dispatchesByPoAll.set(d.poId, arr);
   866→   866→      }
   867→   867→      const computePoMetrics = (poIds: Iterable<string>) => {
   868→   868→        let totalValue = 0;
   869→   869→        const ratios: number[] = [];
   870→   870→        for (const pid of poIds) {
   871→   871→          const po = poById.get(pid);
   872→   872→          if (!po) continue;
   873→   873→          totalValue += po.totalValue || 0;
   874→   874→          const ds = dispatchesByPoAll.get(pid) ?? [];
   875→   875→          if (ds.length > 0) {
   876→   876→            const ordered = parseLineItems(po.lineItemsJson).reduce((a, i) => a + (i.setQty || 0), 0);
   877→   877→            const totalDispatched = ds.reduce((s, d) => s + (d.dispatchedQty || 0), 0);
   878→   878→            const ratio = ordered > 0 ? Math.min(1, totalDispatched / ordered) : 0;
   879→   879→            ratios.push(ratio);
   880→   880→          }
   881→   881→        }
   882→   882→        const fulfillment = ratios.length > 0
   883→   883→          ? Math.round((ratios.reduce((a, b) => a + b, 0) / ratios.length) * 100)
   884→   884→          : 0;
   885→   885→        return { totalValue: Math.round(totalValue), fulfillment };
   886→   886→      };
   887→   887→      const taggedMetrics = computePoMetrics(taggedPoSet);
   888→   888→      const allMetrics = computePoMetrics(allPoIds);
   889→   889→      poComparison = {
   890→   890→        taggedCount: taggedPoCount,
   891→   891→        totalCount: allPoIds.size,
   892→   892→        taggedValue: taggedMetrics.totalValue,
   893→   893→        allValue: allMetrics.totalValue,
   894→   894→        taggedFulfillment: taggedMetrics.fulfillment,
   895→   895→        allFulfillment: allMetrics.fulfillment,
   896→   896→      };
   897→   897→    }
   898→   898→
   899→   899→    tagInsights = {
   900→   900→      client: clientComparison,
   901→   901→      supplier: supplierComparison,
   902→   902→      purchaseOrder: poComparison,
   903→   903→    };
   904→   904→  }
   905→   905→
   906→   906→  // ═══════════════════════════════════════════════════════════════════════════
   907→   907→  return NextResponse.json({
   908→   908→    appliedFilter,
   909→   909→    tagInsights,
   910→   910→    // Total (unfiltered) entity counts — the UI uses these to render
   911→   911→    // "(N of M suppliers)" headers when a tag filter narrows the result set.
   912→   912→    totals: {
   913→   913→      supplierCount: allSupplierIds.size,
   914→   914→      clientCount: allClientIds.size,
   915→   915→      purchaseOrderCount: allPoIds.size,
   916→   916→    },
   917→   917→    suppliers: supplierScores,
   918→   918→    clients: clientExposure,
   919→   919→    brokerage: brokerageForecast,
   920→   920→    volume: volumeTrends,
   921→   921→    forecast,
   922→   922→  });
   923→   923→}
