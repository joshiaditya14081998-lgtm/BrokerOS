   289→function DispatchesTab({ dispatches, logs }: { dispatches: Dispatch[]; logs: DispatchDateLog[] }) {
   290→  if (dispatches.length === 0 && logs.length === 0) {
   291→    return <EmptyState title="No dispatches yet" hint="Shipments will appear here once recorded." />;
   292→  }
   293→  return (
   294→    <>
   295→      {dispatches.length === 0 ? null : dispatches.map((d) => {
   296→        const items = safeParse<DispatchItem[]>(d.itemsJson, []);
   297→        return (
   298→          <div key={d.id} className="glass rounded-xl p-4">
   299→            <div className="flex items-start justify-between gap-2">
   300→              <div>
   301→                <p className="text-sm font-semibold">{formatDate(d.dispatchDate)}</p>
   302→                <p className="text-xs text-muted-foreground">
   303→                  Dispatched {d.dispatchedQty} sets · {d.photos.length} photo{d.photos.length === 1 ? "" : "s"}
   304→                </p>
   305→              </div>
   306→              <StatusChip status={d.status} />
   307→            </div>
   308→            {items.length > 0 ? (
   309→              <div className="mt-2 grid grid-cols-1 gap-1 sm:grid-cols-2">
   310→                {items.map((it, i) => (
   311→                  <div key={i} className="flex items-center justify-between rounded-lg bg-card/40 px-2 py-1 text-xs">
   312→                    <span className="truncate">
   313→                      <span className="font-medium">{it.styleName}</span>
   314→                      {it.color ? <span className="ml-1 text-muted-foreground">· {it.color}</span> : null}
   315→                    </span>
   316→                    <span className="ml-2 shrink-0 font-semibold">{it.qty}</span>
   317→                  </div>
   318→                ))}
   319→              </div>
   320→            ) : null}
   321→            {d.notes ? <p className="mt-2 text-xs text-muted-foreground">“{d.notes}”</p> : null}
   322→            {d.disputes.length > 0 ? (
   323→              <div className="mt-2 space-y-1 border-t border-border/40 pt-2">
   324→                <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Disputes</p>

   326→                  <div key={dis.id} className="flex items-center justify-between text-[11px]">

   328→                      <AlertTriangle className="size-3 text-rose-500" />
   329→                      {titleCase(dis.type)} · {dis.quantityAffected} affected
   330→                    </span>
   331→                    <StatusChip status={dis.status} />
   332→                  </div>
   333→                ))}
   334→              </div>
   335→            ) : null}
   336→            <div className="mt-3">
   337→              <PhotoUpload
   338→                entityType="Dispatch"
   339→                entityId={d.id}
   340→                stage="dispatch"
   341→                label="Dispatch photos"
   342→                hint="Loading bay, bale counts, vehicle number, delivery challan."
   343→              />
   344→            </div>
   345→          </div>
   346→        );
   347→      })}
   348→
   349→      {logs.length > 0 ? (
   350→        <div className="glass rounded-xl p-4">
   351→          <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Dispatch date revisions</p>
   352→          <div className="mt-2 space-y-2">
   353→            {logs.map((l) => (
   354→              <div key={l.id} className="flex items-start gap-2 text-xs">
   355→                <Clock className="mt-0.5 size-3.5 shrink-0 text-amber-500" />
   356→                <div className="min-w-0 flex-1">
   357→                  <div className="flex flex-wrap items-center gap-1.5">
   358→                    <span className="text-muted-foreground line-through">{formatDate(l.oldDate)}</span>
   359→                    <ChevronRight className="size-3 text-muted-foreground" />
   360→                    <span className="font-medium">{formatDate(l.newDate)}</span>
   361→                    <span className="text-muted-foreground">· {formatDateTime(l.createdAt)}</span>

   365→              </div>
   366→            ))}
   367→          </div>
   368→        </div>
   369→      ) : null}
   370→    </>
   371→  );
   372→}

   482→function BrokerageCard({ brokerage }: { brokerage: Brokerage }) {
   483→  return (
   484→    <div className="glass rounded-xl p-4">
   485→      <div className="mb-2 flex items-center gap-1.5">
   486→        <BadgePercent className="size-4 text-emerald-500" />
   487→        <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Brokerage</p>
   488→      </div>
   489→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   490→        <KV label="Commission" value={`${brokerage.commissionRate}%`} />
   491→        <KV label="Base (excl GST)" value={formatCurrency(brokerage.baseAmount)} />
   492→        <KV label="Brokerage amt" value={formatCurrency(brokerage.brokerageAmount)} tone="emerald" />
   493→        <KV label="Eligible" value={brokerage.eligible ? "Yes" : "No"} tone={brokerage.eligible ? "emerald" : "default"} />

   496→      </div>
   497→      {brokerage.forceEligible && brokerage.forceReason ? (
   498→        <p className="mt-2 rounded-lg bg-amber-500/10 px-3 py-1.5 text-[11px] text-amber-700 dark:text-amber-300">
   499→          Force-eligible: {brokerage.forceReason}
   500→        </p>
   501→      ) : null}
   502→      {brokerage.payout ? (
   503→        <p className="mt-2 text-[11px] text-muted-foreground">
   504→          Payout: {titleCase(brokerage.payout.status)} · {titleCase(brokerage.payout.cadence)}
   505→          {brokerage.payout.paidAt ? ` · paid ${formatDate(brokerage.payout.paidAt)}` : ""}
   506→        </p>
   507→      ) : null}
   508→    </div>
   509→  );
   510→}
   511→
   512→type TimelineEntry = {
   513→  id: string;
   514→  date: string;
   515→  kind: "log" | "payment" | "dispute" | "bill";
   516→  title: string;
   517→  detail: string | null;
   518→};
   519→
   520→function TimelineTab({ po }: { po: PoDetail }) {
   521→  const entries: TimelineEntry[] = React.useMemo(() => {
   522→    const list: TimelineEntry[] = [];
   523→    for (const l of po.dispatchDateLogs) {
   524→      list.push({
   525→        id: l.id, date: l.createdAt, kind: "log",
   526→        title: `Dispatch date revised → ${formatDate(l.newDate)}`,
   527→        detail: l.reason || (l.oldDate ? `from ${formatDate(l.oldDate)}` : null),
   528→      });
   529→    }
   530→    if (po.bill) {
   531→      list.push({
   532→        id: `bill-${po.bill.id}`, date: po.bill.createdAt, kind: "bill",
   533→        title: `Bill ${po.bill.billNumber} generated`,
   534→        detail: `Final: ${formatCurrency(po.bill.finalAmount)} · status: ${titleCase(po.bill.status)}`,
   535→      });
   536→      for (const p of po.bill.payments) {
   537→        list.push({
   538→          id: `pay-${p.id}`, date: p.date, kind: "payment",
   539→          title: `Payment · ${titleCase(p.mode)}`,
   540→          detail: `${formatCurrency(p.amount)}${p.reference ? ` · ref ${p.reference}` : ""}`,
   541→        });
   542→      }
   543→      if (po.bill.brokerage) {
   544→        const b = po.bill.brokerage;
   545→        list.push({
   546→          id: `brok-${b.id}`, date: b.eligibleAt ?? po.bill.createdAt, kind: "bill",
   547→          title: `Brokerage ${b.eligible ? "eligible" : "accrued"}`,
   548→          detail: `${formatCurrency(b.brokerageAmount)} · ${titleCase(b.payoutStatus)}`,
   549→        });
   550→      }
   551→    }
   552→    for (const d of po.disputes) {
   553→      list.push({
   554→        id: `disp-${d.id}`, date: d.createdAt, kind: "dispute",
   555→        title: `Dispute · ${titleCase(d.type)}`,
   556→        detail: d.description || `${d.quantityAffected} units · ${formatCurrency(d.valueAffected)}`,
   557→      });
   558→    }
   559→    return list.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   560→  }, [po]);
   561→
   562→  if (entries.length === 0) {
   563→    return (
   564→      <GlassCard className="p-8">
   565→        <EmptyState title="No timeline events" hint="Dispatches, payments, and disputes will appear here." icon={<Clock className="size-5" />} />
   566→      </GlassCard>
   567→    );
   568→  }
   569→
   570→  return (
   571→    <div className="glass rounded-xl p-4">
   572→      <ol className="relative space-y-4 border-l border-border/60 pl-4">
   573→        {entries.map((e) => (
   574→          <li key={e.id} className="relative">
   575→            <span className={`absolute -left-[1.41rem] top-1 grid size-3.5 place-items-center rounded-full border-2 border-background ${kindClass(e.kind)}`}>
   576→              <span className="size-1.5 rounded-full bg-current opacity-80" />
   577→            </span>
   578→            <div className="flex flex-wrap items-center gap-1.5">
   579→              <span className="text-sm font-medium">{e.title}</span>
   580→              <span className="text-[10px] uppercase tracking-wider text-muted-foreground">
   581→                {kindLabel(e.kind)}
   582→              </span>
   583→            </div>
   584→            <p className="text-xs text-muted-foreground">{formatDateTime(e.date)}</p>
   585→            {e.detail ? <p className="mt-0.5 text-xs text-foreground/80">{e.detail}</p> : null}
   586→          </li>
   587→        ))}
   588→      </ol>
   589→    </div>
   590→  );
   591→}
   592→
   593→function kindClass(kind: TimelineEntry["kind"]): string {
   594→  switch (kind) {
   595→    case "payment": return "bg-emerald-500/20 text-emerald-600 dark:text-emerald-400";
   596→    case "log": return "bg-amber-500/20 text-amber-600 dark:text-amber-400";
   597→    case "dispute": return "bg-rose-500/20 text-rose-600 dark:text-rose-400";
   598→    default: return "bg-teal-500/20 text-teal-600 dark:text-teal-400";
   599→  }
   600→}
   601→
   602→function kindLabel(kind: TimelineEntry["kind"]): string {
   603→  switch (kind) {
   604→    case "payment": return "Payment";
   605→    case "log": return "Revision";
   606→    case "dispute": return "Dispute";
   607→    default: return "Billing";
   608→  }
   609→}

   611→function Stat({
   612→  icon, label, value, tone = "default",
   613→}: {
   614→  icon: React.ReactNode;
   615→  label: string;
   616→  value: string;
   617→  tone?: "default" | "emerald" | "amber";
   618→}) {
   619→  const toneClass = tone === "emerald"
   620→    ? "text-emerald-600 dark:text-emerald-400"
   621→    : tone === "amber"
   622→      ? "text-amber-600 dark:text-amber-400"
   623→      : "text-foreground";
   624→  return (
   625→    <div className="rounded-xl border border-border/40 bg-card/40 p-3">
   626→      <div className="flex items-center gap-1.5 text-muted-foreground">
   627→        {icon}
   628→        <span className="text-[10px] uppercase tracking-wider">{label}</span>
   629→      </div>
   630→      <p className={`kpi-num mt-1 text-base font-semibold ${toneClass}`}>{value}</p>
   631→    </div>
   632→  );
   633→}
