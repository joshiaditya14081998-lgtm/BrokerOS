   215→// ─────────────────────────────────────────────────────────────────────────────
   216→// ScoreRing — circular SVG progress for the supplier reliability score.
   217→// ─────────────────────────────────────────────────────────────────────────────
   218→function ScoreRing({ score, size = 48 }: { score: number; size?: number }) {
   219→  const radius = (size - 8) / 2;
   220→  const circumference = 2 * Math.PI * radius;
   221→  const offset = circumference - (score / 100) * circumference;
   222→  const color = score >= 85 ? COLORS.emerald
   223→    : score >= 70 ? COLORS.teal
   224→    : score >= 55 ? COLORS.amber
   225→    : COLORS.rose;
   226→  return (
   227→    <div className="relative" style={{ width: size, height: size }}>
   228→      <svg width={size} height={size} className="-rotate-90">
   229→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="oklch(0.7 0.02 160 / 0.2)" strokeWidth={4} />
   230→        <circle
   231→          cx={size / 2}
   232→          cy={size / 2}
   233→          r={radius}
   234→          fill="none"
   235→          stroke={color}
   236→          strokeWidth={4}
   237→          strokeDasharray={circumference}
   238→          strokeDashoffset={offset}
   239→          strokeLinecap="round"
   240→          className="transition-all duration-500"
   241→        />
   242→      </svg>
   243→      <span className="absolute inset-0 grid place-items-center text-xs font-semibold kpi-num" style={{ color }}>
   244→        {score}
   245→      </span>
   246→    </div>
   247→  );
   248→}

   356→// ─────────────────────────────────────────────────────────────────────────────
   357→// TagInsightsCard — top-of-view summary card shown when a tag filter is
   358→// active. Surfaces tag name + color, entity counts, total business value,
   359→// average-metric comparison, and a tagged-vs-all share bar.
   360→// ─────────────────────────────────────────────────────────────────────────────
   361→function TagInsightsCard({ appliedFilter, insights }: { appliedFilter: AppliedFilter; insights: TagInsights }) {
   362→  const swatchClass = TAG_SWATCH_CLASS[appliedFilter.tagColor] ?? TAG_SWATCH_CLASS.emerald;
   363→  const colorClass = TAG_COLOR_CLASS[appliedFilter.tagColor] ?? TAG_COLOR_CLASS.emerald;
   364→
   365→  // Build the comparison "share of total" bars for the included entity types.
   366→  type ShareRow = {
   367→    label: string;
   368→    tagged: number;
   369→    total: number;
   370→    icon: React.ReactNode;
   371→  };
   372→  const shareRows: ShareRow[] = [];
   373→  if (insights.client) {
   374→    shareRows.push({
   375→      label: "Clients",
   376→      tagged: insights.client.taggedCount,
   377→      total: insights.client.totalCount,
   378→      icon: <Users className="size-3.5" />,
   379→    });
   380→  }
   381→  if (insights.supplier) {
   382→    shareRows.push({
   383→      label: "Suppliers",
   384→      tagged: insights.supplier.taggedCount,
   385→      total: insights.supplier.totalCount,
   386→      icon: <Factory className="size-3.5" />,
   387→    });
   388→  }
   389→  if (insights.purchaseOrder) {
   390→    shareRows.push({
   391→      label: "Purchase Orders",
   392→      tagged: insights.purchaseOrder.taggedCount,
   393→      total: insights.purchaseOrder.totalCount,
   394→      icon: <BarChart3 className="size-3.5" />,
   395→    });
   396→  }
   397→
   398→  // Average-metric comparison insights (one per included entity type).
   399→  type InsightRow = {
   400→    label: string;
   401→    taggedValue: string;
   402→    allValue: string;
   403→    delta: number; // % difference vs all
   404→  };
   405→  const insightRows: InsightRow[] = [];
   406→  if (insights.client) {
   407→    const delta = insights.client.allAvgOutstanding > 0
   408→      ? Math.round(((insights.client.taggedAvgOutstanding - insights.client.allAvgOutstanding) / insights.client.allAvgOutstanding) * 100)
   409→      : 0;
   410→    insightRows.push({
   411→      label: "Avg outstanding / client",
   412→      taggedValue: formatCurrency(insights.client.taggedAvgOutstanding, { compact: true }),
   413→      allValue: formatCurrency(insights.client.allAvgOutstanding, { compact: true }),
   414→      delta,
   415→    });
   416→  }
   417→  if (insights.supplier) {
   418→    const delta = insights.supplier.allFulfillment > 0
   419→      ? insights.supplier.taggedFulfillment - insights.supplier.allFulfillment
   420→      : 0;
   421→    insightRows.push({
   422→      label: "Avg fulfillment / supplier",
   423→      taggedValue: `${insights.supplier.taggedFulfillment}%`,
   424→      allValue: `${insights.supplier.allFulfillment}%`,
   425→      delta,
   426→    });
   427→  }
   428→  if (insights.purchaseOrder) {
   429→    const avgTagged = insights.purchaseOrder.taggedCount > 0
   430→      ? insights.purchaseOrder.taggedValue / insights.purchaseOrder.taggedCount
   431→      : 0;
   432→    const avgAll = insights.purchaseOrder.totalCount > 0
   433→      ? insights.purchaseOrder.allValue / insights.purchaseOrder.totalCount
   434→      : 0;
   435→    const delta = avgAll > 0 ? Math.round(((avgTagged - avgAll) / avgAll) * 100) : 0;
   436→    insightRows.push({
   437→      label: "Avg PO value",
   438→      taggedValue: formatCurrency(Math.round(avgTagged), { compact: true }),
   439→      allValue: formatCurrency(Math.round(avgAll), { compact: true }),
   440→      delta,
   441→    });
   442→  }
   443→
   444→  return (
   445→    <GlassCard className="relative overflow-hidden p-5">
   446→      <div className="pointer-events-none absolute inset-0 bg-gradient-to-br from-emerald-500/5 to-transparent" />
   447→      <div className="relative">
   448→        <div className="flex flex-wrap items-start justify-between gap-3">
   449→          <div className="flex items-center gap-3">
   450→            <div className={cn("grid size-10 place-items-center rounded-xl border", colorClass)}>
   451→              <TagIcon className="size-5" />
   452→            </div>
   453→            <div>
   454→              <div className="flex items-center gap-2">
   455→                <h3 className="text-lg font-semibold tracking-tight text-foreground">
   456→                  {appliedFilter.tagName}
   457→                </h3>
   458→                <span className={cn("size-2.5 rounded-full", swatchClass)} aria-hidden />
   459→              </div>
   460→              <p className="text-xs text-muted-foreground">
   461→                {appliedFilter.entityType === null
   462→                  ? "Filter applied across Clients, Suppliers & Purchase Orders"
   463→                  : `Filter scoped to ${appliedFilter.entityType === "PurchaseOrder" ? "Purchase Orders" : `${appliedFilter.entityType}s`}`}
   464→              </p>
   465→            </div>
   466→          </div>
   467→          <Badge variant="outline" className="border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300">
   468→            <PieChart className="size-3" />
   469→            Tag insights
   470→          </Badge>
   471→        </div>
   472→
   473→        {/* Share-of-total bars */}
   474→        <div className="mt-5 grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   475→          {shareRows.map((row) => {
   476→            const pct = row.total > 0 ? Math.round((row.tagged / row.total) * 100) : 0;
   477→            return (
   478→              <div key={row.label} className="rounded-xl border border-border/50 bg-card/40 p-3">
   479→                <div className="flex items-center justify-between gap-2">
   480→                  <span className="inline-flex items-center gap-1.5 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   481→                    {row.icon}
   482→                    {row.label}
   483→                  </span>
   484→                  <span className="kpi-num text-xs text-muted-foreground">
   485→                    {row.tagged} of {row.total}
   486→                  </span>
   487→                </div>
   488→                <div className="mt-2 flex items-center gap-2">
   489→                  <div className="h-2 flex-1 overflow-hidden rounded-full bg-muted">
   490→                    <div
   491→                      className="h-full rounded-full bg-emerald-500 transition-all duration-500"
   492→                      style={{ width: `${Math.max(2, pct)}%` }}
   493→                    />
   494→                  </div>
   495→                  <span className="kpi-num w-10 text-right text-xs font-medium text-foreground">{pct}%</span>
   496→                </div>
   497→              </div>
   498→            );
   499→          })}
   500→        </div>
   501→
   502→        {/* Tagged vs all comparison metrics */}
   503→        <div className="mt-4">
   504→          <p className="mb-2 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   505→            Tagged vs. all entities
   506→          </p>
   507→          <div className="overflow-x-auto rounded-xl border border-border/50">
   508→            <Table>
   509→              <TableHeader>
   510→                <TableRow>
   511→                  <TableHead>Metric</TableHead>
   512→                  <TableHead className="text-right">Tagged</TableHead>
   513→                  <TableHead className="text-right">All</TableHead>
   514→                  <TableHead className="text-right">Δ vs. all</TableHead>
   515→                </TableRow>
   516→              </TableHeader>
   517→              <TableBody>
   518→                {insightRows.map((row) => {
   519→                  const positive = row.delta > 0;
   520→                  const neutral = row.delta === 0;
   521→                  const tone = neutral
   522→                    ? "text-muted-foreground"
   523→                    : positive

   549→// TagComparisonChart — grouped BarChart comparing Tagged vs All across the
   550→// key metrics for each included entity type.
   551→// ─────────────────────────────────────────────────────────────────────────────
   552→function TagComparisonChart({ insights, tagColor }: { insights: TagInsights; tagColor: string }) {
   553→  // Build chart data — one row per entity-type/metric.
   554→  type ChartRow = { metric: string; tagged: number; all: number };
   555→  const rows: ChartRow[] = [];
   556→  if (insights.client) {
   557→    rows.push({ metric: "Client · Business", tagged: insights.client.taggedBusiness, all: insights.client.allBusiness });
   558→    rows.push({ metric: "Client · Outstanding", tagged: insights.client.taggedOutstanding, all: insights.client.allOutstanding });
   559→    rows.push({ metric: "Client · Brokerage", tagged: insights.client.taggedBrokerage, all: insights.client.allBrokerage });
   560→  }
   561→  if (insights.supplier) {
   562→    rows.push({ metric: "Supplier · Supplied", tagged: insights.supplier.taggedSupplied, all: insights.supplier.allSupplied });
   563→    // Fulfillment is a % — scale to ₹10k-per-% so it shows up on the same axis
   564→    // (we surface the raw % in the tooltip via formatter).
   565→    rows.push({
   566→      metric: "Supplier · Fulfillment",
   567→      tagged: insights.supplier.taggedFulfillment * 1000,
   568→      all: insights.supplier.allFulfillment * 1000,
   569→    });
   570→  }
   571→  if (insights.purchaseOrder) {
   572→    rows.push({ metric: "PO · Value", tagged: insights.purchaseOrder.taggedValue, all: insights.purchaseOrder.allValue });
   573→    rows.push({
   574→      metric: "PO · Fulfillment",
   575→      tagged: insights.purchaseOrder.taggedFulfillment * 1000,
   576→      all: insights.purchaseOrder.allFulfillment * 1000,
   577→    });
   578→  }
   579→
   580→  if (rows.length === 0) return null;
   581→
   582→  const tagSwatchColor = TAG_SWATCH_CLASS[tagColor] ? COLORS[colorsKeyFor(tagColor)] : COLORS.emerald;
   583→  const tooltipFormatter = (value: number, name: string, item: { payload?: { metric?: string } }) => {
   584→    const metric = item?.payload?.metric ?? "";
   585→    if (metric.includes("Fulfillment")) {
   586→      return [`${Math.round(value / 1000)}%`, name];
   587→    }
   588→    return [formatCurrency(value), name];
   589→  };
   590→
   591→  return (
   592→    <GlassCard className="p-5">
   593→      <SectionHeader
   594→        title="Tagged vs. All — Comparison"
   595→        description="Side-by-side view of how the tagged segment stacks up against the full portfolio"
   596→        action={
   597→          <div className="hidden items-center gap-3 text-xs sm:flex">
   598→            <span className="inline-flex items-center gap-1.5 text-muted-foreground">
   599→              <span className="size-2.5 rounded-full" style={{ background: tagSwatchColor }} />
   600→              Tagged
   601→            </span>
   602→            <span className="inline-flex items-center gap-1.5 text-muted-foreground">
   603→              <span className="size-2.5 rounded-full" style={{ background: COLORS.teal }} />
   604→              All
   605→            </span>
   606→          </div>
   607→        }
   608→      />
   609→      <div className="mt-4 h-72 w-full">
   610→        <ResponsiveContainer width="100%" height="100%">
   611→          <BarChart data={rows} margin={{ top: 4, right: 8, bottom: 0, left: 0 }}>
   612→            <CartesianGrid strokeDasharray="3 3" stroke="oklch(0.7 0.02 160 / 0.2)" vertical={false} />
   613→            <XAxis
   614→              dataKey="metric"
   615→              tick={{ fontSize: 10, fill: "oklch(0.5 0.02 165)" }}
   616→              axisLine={false}
   617→              tickLine={false}
   618→              interval={0}
   619→              angle={-25}
   620→              textAnchor="end"
   621→              height={60}
   622→            />
   623→            <YAxis
   624→              tick={{ fontSize: 11, fill: "oklch(0.5 0.02 165)" }}
   625→              axisLine={false}
   626→              tickLine={false}
   627→              tickFormatter={(v) => formatCurrency(v as number, { compact: true })}
   628→              width={55}
   629→            />
   630→            <Tooltip
   631→              cursor={{ fill: "oklch(0.7 0.02 160 / 0.08)" }}
   632→              contentStyle={{ background: "oklch(0.16 0.012 170 / 0.92)", border: "1px solid oklch(1 0 0 / 0.12)", borderRadius: 12, color: "white", backdropFilter: "blur(12px)" }}
   633→              formatter={(v: number, name: string, item) => tooltipFormatter(v, name, item as { payload?: { metric?: string } })}
   634→            />
   635→            <Legend
   636→              wrapperStyle={{ fontSize: 11 }}
   637→              formatter={(v) => (v === "tagged" ? "Tagged" : "All")}
   638→            />
   639→            <Bar dataKey="tagged" name="tagged" fill={tagSwatchColor} radius={[4, 4, 0, 0]} />
   640→            <Bar dataKey="all" name="all" fill={COLORS.teal} radius={[4, 4, 0, 0]} fillOpacity={0.55} />
   641→          </BarChart>
   642→        </ResponsiveContainer>
   643→      </div>
   644→      <p className="mt-2 text-[11px] text-muted-foreground">
   645→        Fulfillment metrics are scaled for visual comparison — hover for the true percentage.
   646→      </p>
   647→    </GlassCard>
   648→  );
   649→}

   651→// Maps a tag color name (from the shared palette) to the oklch chart color so
   652→// the TagComparisonChart bars match the tag's pill swatch.
   653→function colorsKeyFor(color: string): keyof typeof COLORS {
   654→  if (color === "emerald") return "emerald";
   655→  if (color === "amber") return "amber";
   656→  if (color === "rose") return "rose";
   657→  if (color === "teal") return "teal";
   658→  if (color === "plum") return "plum";
   659→  return "emerald";
   660→}
