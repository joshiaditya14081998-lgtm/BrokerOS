     8→// ─────────────────────────────────────────────────────────────────────────────
     9→// Daily Broker Digest — a consolidated snapshot of what needs the broker's
    10→// attention today, served in three formats:
    11→//   • text  → plain-text summary (also feeds the KPI strip in the UI)
    12→//   • html  → standalone HTML email body (inline CSS, emerald accent, tables)
    13→//             that the broker can copy/paste into Gmail/Outlook or send later
    14→//   • email → AI-generated "morning brief" via z-ai-web-dev-sdk LLM
    15→//             (returns { subject, body }) — friendly 3-4 paragraph summary
    16→//
    17→// All three formats share the same underlying data, gathered once via Prisma.
    18→// ─────────────────────────────────────────────────────────────────────────────
    19→
    20→const DAY_MS = 24 * 60 * 60 * 1000;
    21→
    22→type Priority = "urgent" | "high" | "normal";
    23→
    24→type ActionItem = {
    25→  type: string;
    26→  title: string;
    27→  description: string;
    28→  priority: Priority;
    29→  dueDate: string;
    30→};
    31→
    32→type DigestData = {
    33→  generatedAt: string; // ISO
    34→  generatedAtDisplay: string; // "12 Aug 2026, 09:30 AM"
    35→  dateLabel: string; // "12 Aug 2026"
    36→  actionItems: ActionItem[];
    37→  outstandingReceivable: {
    38→    total: number;
    39→    topClients: { name: string; due: number }[];
    40→  };
    41→  overdueDispatches: {
    42→    poNumber: string;
    43→    supplierName: string;
    44→    dueDate: string;
    45→    daysOverdue: number;
    46→  }[];
    47→  paymentsDueNext7Days: {
    48→    billNumber: string;
    49→    clientName: string;
    50→    amount: number;
    51→    dueDate: string;
    52→  }[];
    53→  brokeragePendingPayout: {
    54→    accrued: number; // eligible, awaiting payout
    55→    scheduled: number; // already in a payout batch
    56→    total: number;
    57→  };
    58→  openDisputesCount: number;
    59→  pendingNotificationsCount: number;
    60→  summary: {
    61→    pendingActions: number;
    62→    outstandingTotal: number;
    63→    duePaymentsCount: number; // bills due within next 7 days (or already overdue)
    64→    openDisputes: number;
    65→  };
    66→};
    67→
    68→function fmtINR(n: number): string {
    69→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
    70→}
    71→
    72→function fmtINRCompact(n: number): string {
    73→  const v = Number.isFinite(n) ? n : 0;
    74→  if (Math.abs(v) >= 10000000) return `₹${(v / 10000000).toFixed(2)}Cr`;
    75→  if (Math.abs(v) >= 100000) return `₹${(v / 100000).toFixed(2)}L`;
    76→  if (Math.abs(v) >= 1000) return `₹${(v / 1000).toFixed(1)}K`;
    77→  return fmtINR(v);
    78→}
    79→
    80→function fmtDate(d: Date | string): string {
    81→  const dt = typeof d === "string" ? new Date(d) : d;
    82→  return dt.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
    83→}
    84→
    85→function fmtDateTime(d: Date): string {
    86→  return d.toLocaleString("en-IN", {
    87→    day: "2-digit",
    88→    month: "short",
    89→    year: "numeric",
    90→    hour: "2-digit",
    91→    minute: "2-digit",
    92→  });
    93→}
    94→
    95→function pluralDays(n: number): string {
    96→  const abs = Math.max(1, Math.abs(n));
    97→  return `${abs} day${abs === 1 ? "" : "s"}`;
    98→}

   100→const PRIORITY_RANK: Record<Priority, number> = { urgent: 0, high: 1, normal: 2 };
   101→
   102→// ─────────────────────────────────────────────────────────────────────────────
   103→// gatherDigestData — single pass over Prisma, mirrors the action-center logic
   104→// (simplified) plus the digest-specific aggregations (top clients, payments
   105→// due next 7d, brokerage pending payout).
   106→// ─────────────────────────────────────────────────────────────────────────────
   107→async function gatherDigestData(): Promise<DigestData> {
   108→  const now = new Date();
   109→  const in7d = new Date(now.getTime() + 7 * DAY_MS);
   110→
   111→  const [
   112→    visits,
   113→    overduePOs,
   114→    openBills,
   115→    eligibleBrokerages,
   116→    openDisputes,
   117→    pendingNotifications,
   118→  ] = await Promise.all([
   119→    db.visit.findMany({
   120→      where: {
   121→        status: { in: ["scheduled", "followed_up"] },
   122→        plannedDate: { lte: new Date(now.getTime() + 3 * DAY_MS) },
   123→      },
   124→      include: { client: { select: { name: true } } },
   125→      orderBy: { plannedDate: "asc" },
   126→    }),
   127→    db.purchaseOrder.findMany({
   128→      where: { status: "open", expectedDispatchDate: { not: null, lt: now } },
   129→      include: { supplier: { select: { name: true } } },
   130→      orderBy: { expectedDispatchDate: "asc" },
   131→    }),
   132→    db.bill.findMany({
   133→      where: { status: { not: "fully_paid" } },
   134→      include: { client: { select: { name: true, defaultPaymentCycleDays: true } } },
   135→      orderBy: { createdAt: "asc" },
   136→    }),
   137→    db.brokerage.findMany({
   138→      where: { eligible: true, payoutStatus: { in: ["accrued", "scheduled"] } },
   139→      include: { client: { select: { name: true } } },
   140→    }),
   141→    db.dispute.findMany({
   142→      where: { status: "open" },
   143→      include: { po: { select: { poNumber: true } } },
   144→      orderBy: { createdAt: "asc" },
   145→    }),
   146→    db.notification.count({ where: { status: "pending" } }),
   147→  ]);
   148→
   149→  // ── Action items (simplified from /api/action-center) ────────────────────
   150→  const actionItems: ActionItem[] = [];
   151→
   152→  for (const v of visits) {
   153→    const daysOverdue = Math.floor((now.getTime() - v.plannedDate.getTime()) / DAY_MS);
   154→    const isPast = v.plannedDate < now;
   155→    if (v.status === "scheduled" && isPast) {
   156→      actionItems.push({
   157→        type: "visit_followup",
   158→        title: `Follow up with ${v.client.name}`,
   159→        description: `Visit was ${pluralDays(daysOverdue)} ago. Call to reschedule.`,
   160→        priority: daysOverdue > 2 ? "urgent" : "high",
   161→        dueDate: v.plannedDate.toISOString(),
   162→      });
   163→    } else if (
   164→      v.status === "followed_up" &&
   165→      v.plannedDate < new Date(now.getTime() - 3 * DAY_MS)
   166→    ) {
   167→      actionItems.push({
   168→        type: "visit_followup",
   169→        title: `Re-follow-up: ${v.client.name}`,
   170→        description: `Original visit was ${pluralDays(daysOverdue)} ago and still pending closure.`,
   171→        priority: "high",
   172→        dueDate: v.plannedDate.toISOString(),
   173→      });
   174→    } else if (v.status === "scheduled" && !isPast) {
   175→      const daysAhead = Math.max(
   176→        0,
   177→        Math.floor((v.plannedDate.getTime() - now.getTime()) / DAY_MS),
   178→      );
   179→      actionItems.push({
   180→        type: "visit_followup",
   181→        title: `Upcoming visit: ${v.client.name}`,
   182→        description:
   183→          daysAhead === 0
   184→            ? "Visit scheduled for today."
   185→            : `Visit scheduled in ${pluralDays(daysAhead)}.`,
   186→        priority: "normal",
   187→        dueDate: v.plannedDate.toISOString(),
   188→      });
   189→    }
   190→  }
   191→
   192→  // Overdue dispatches → also feed the dispatches list separately
   193→  const overdueDispatchList: DigestData["overdueDispatches"] = [];
   194→  for (const po of overduePOs) {
   195→    const effective = po.revisedDispatchDate ?? po.expectedDispatchDate;
   196→    if (!effective) continue;
   197→    const daysOverdue = Math.max(
   198→      1,
   199→      Math.floor((now.getTime() - effective.getTime()) / DAY_MS),
   200→    );
   201→    actionItems.push({
   202→      type: "dispatch_due",
   203→      title: `Record dispatch: ${po.poNumber}`,
   204→      description: `Supplier ${po.supplier.name} dispatch is overdue.`,
   205→      priority: "high",
   206→      dueDate: effective.toISOString(),
   207→    });
   208→    overdueDispatchList.push({
   209→      poNumber: po.poNumber,
   210→      supplierName: po.supplier.name,
   211→      dueDate: effective.toISOString(),
   212→      daysOverdue,
   213→    });
   214→  }
   215→
   216→  // Payments due — collect bills whose computed due date is within 7 days OR
   217→  // already past due.
   218→  const paymentsDueNext7Days: DigestData["paymentsDueNext7Days"] = [];
   219→  for (const bill of openBills) {
   220→    const remaining = Math.max(0, bill.finalAmount - bill.paidAmount);
   221→    if (remaining <= 0) continue;
   222→    const dueDate = new Date(
   223→      bill.createdAt.getTime() + bill.client.defaultPaymentCycleDays * DAY_MS,
   224→    );
   225→    const diffDays = Math.floor((now.getTime() - dueDate.getTime()) / DAY_MS);
   226→    const isPast = dueDate < now;
   227→    if (isPast) {
   228→      const priority: Priority = diffDays > 14 ? "urgent" : "high";
   229→      actionItems.push({
   230→        type: "payment_due",
   231→        title: `Collect payment: ${bill.billNumber}`,
   232→        description: `${bill.client.name} owes ${fmtINR(remaining)}. ${pluralDays(diffDays)} overdue.`,
   233→        priority,
   234→        dueDate: dueDate.toISOString(),
   235→      });
   236→      paymentsDueNext7Days.push({
   237→        billNumber: bill.billNumber,
   238→        clientName: bill.client.name,
   239→        amount: remaining,
   240→        dueDate: dueDate.toISOString(),
   241→      });
   242→    } else if (dueDate <= in7d) {
   243→      const daysAhead = Math.max(
   244→        0,
   245→        Math.floor((dueDate.getTime() - now.getTime()) / DAY_MS),
   246→      );
   247→      actionItems.push({
   248→        type: "payment_due",
   249→        title: `Collect payment: ${bill.billNumber}`,
   250→        description:
   251→          daysAhead === 0
   252→            ? `${bill.client.name} owes ${fmtINR(remaining)}. Due today.`
   253→            : `${bill.client.name} owes ${fmtINR(remaining)}. Due in ${pluralDays(daysAhead)}.`,
   254→        priority: "normal",
   255→        dueDate: dueDate.toISOString(),
   256→      });
   257→      paymentsDueNext7Days.push({
   258→        billNumber: bill.billNumber,
   259→        clientName: bill.client.name,
   260→        amount: remaining,
   261→        dueDate: dueDate.toISOString(),
   262→      });
   263→    }
   264→  }
   265→
   266→  // Brokerage pending payout — aggregate per client, one action per client
   267→  const brokerageByClient = new Map<string, { total: number; name: string; earliest: Date }>();
   268→  let accruedTotal = 0;
   269→  let scheduledTotal = 0;
   270→  for (const b of eligibleBrokerages) {
   271→    if (b.payoutStatus === "accrued") accruedTotal += b.brokerageAmount;
   272→    if (b.payoutStatus === "scheduled") scheduledTotal += b.brokerageAmount;
   273→    const eligibleAt = b.eligibleAt ?? b.createdAt;
   274→    const prev = brokerageByClient.get(b.clientId);
   275→    if (prev) {
   276→      prev.total += b.brokerageAmount;
   277→      if (eligibleAt < prev.earliest) prev.earliest = eligibleAt;
   278→    } else {
   279→      brokerageByClient.set(b.clientId, {
   280→        total: b.brokerageAmount,
   281→        name: b.client.name,
   282→        earliest: eligibleAt,
   283→      });
   284→    }
   285→  }
   286→  for (const [, agg] of brokerageByClient) {
   287→    actionItems.push({
   288→      type: "brokerage_payout",
   289→      title: `Pay out brokerage: ${agg.name}`,
   290→      description: `${fmtINR(agg.total)} eligible for payout.`,
   291→      priority: "high",
   292→      dueDate: agg.earliest.toISOString(),
   293→    });
   294→  }
   295→
   296→  // Open disputes
   297→  for (const d of openDisputes) {
   298→    const daysOpen = Math.max(
   299→      1,
   300→      Math.floor((now.getTime() - d.createdAt.getTime()) / DAY_MS),
   301→    );
   302→    actionItems.push({
   303→      type: "dispute_resolve",
   304→      title: `Resolve dispute: ${d.po.poNumber}`,
   305→      description: `${d.type.replace(/_/g, " ")} — open for ${pluralDays(daysOpen)}.`,
   306→      priority: daysOpen > 7 ? "urgent" : "high",
   307→      dueDate: d.createdAt.toISOString(),
   308→    });
   309→  }
   310→
   311→  // Sort actions: priority asc, then dueDate asc — cap at 12 for the digest
   312→  actionItems.sort((a, b) => {
   313→    const pr = PRIORITY_RANK[a.priority] - PRIORITY_RANK[b.priority];
   314→    if (pr !== 0) return pr;
   315→    return new Date(a.dueDate).getTime() - new Date(b.dueDate).getTime();
   316→  });
   317→
   318→  // Outstanding receivable — top 3 clients by due amount
   319→  const byClient = new Map<string, { name: string; due: number }>();
   320→  for (const bill of openBills) {
   321→    const remaining = Math.max(0, bill.finalAmount - bill.paidAmount);
   322→    if (remaining <= 0) continue;
   323→    const prev = byClient.get(bill.clientId);
   324→    if (prev) {
   325→      prev.due += remaining;
   326→    } else {
   327→      byClient.set(bill.clientId, { name: bill.client.name, due: remaining });
   328→    }
   329→  }
   330→  const topClients = Array.from(byClient.values())
   331→    .sort((a, b) => b.due - a.due)
   332→    .slice(0, 3);
   333→  const outstandingTotal = Array.from(byClient.values()).reduce((s, c) => s + c.due, 0);
   334→
   335→  return {
   336→    generatedAt: now.toISOString(),
   337→    generatedAtDisplay: fmtDateTime(now),
   338→    dateLabel: fmtDate(now),
   339→    actionItems: actionItems.slice(0, 12),
   340→    outstandingReceivable: {
   341→      total: outstandingTotal,
   342→      topClients,
   343→    },
   344→    overdueDispatches: overdueDispatchList.slice(0, 6),
   345→    paymentsDueNext7Days: paymentsDueNext7Days
   346→      .sort((a, b) => new Date(a.dueDate).getTime() - new Date(b.dueDate).getTime())
   347→      .slice(0, 6),
   348→    brokeragePendingPayout: {
   349→      accrued: accruedTotal,
   350→      scheduled: scheduledTotal,
   351→      total: accruedTotal + scheduledTotal,
   352→    },
   353→    openDisputesCount: openDisputes.length,
   354→    pendingNotificationsCount: pendingNotifications,
   355→    summary: {
   356→      pendingActions: actionItems.length,
   357→      outstandingTotal,
   358→      duePaymentsCount: paymentsDueNext7Days.length,
   359→      openDisputes: openDisputes.length,
   360→    },
   361→  };
   362→}

   368→function buildTextDigest(d: DigestData): string {
   369→  const lines: string[] = [];
   370→  lines.push(`BROKER OS — DAILY DIGEST`);
   371→  lines.push(`${d.generatedAtDisplay}`);
   372→  lines.push("=".repeat(48));
   373→  lines.push("");
   374→  lines.push(`Pending actions: ${d.summary.pendingActions}`);
   375→  lines.push(`Outstanding receivable: ${fmtINR(d.summary.outstandingTotal)}`);
   376→  lines.push(`Payments due (next 7 days): ${d.summary.duePaymentsCount}`);
   377→  lines.push(`Open disputes: ${d.summary.openDisputes}`);
   378→  lines.push(`Pending notifications: ${d.pendingNotificationsCount}`);
   379→  lines.push("");
   380→
   381→  lines.push("─ TODAY'S PRIORITISED ACTIONS ─");
   382→  if (d.actionItems.length === 0) {
   383→    lines.push("All caught up — no pending actions.");
   384→  } else {
   385→    d.actionItems.forEach((a, i) => {
   386→      const prio = a.priority.toUpperCase();
   387→      lines.push(`${i + 1}. [${prio}] ${a.title}`);
   388→      lines.push(`   ${a.description}`);
   389→      lines.push(`   Due: ${fmtDate(a.dueDate)}`);
   390→    });
   391→  }
   392→  lines.push("");
   393→
   394→  lines.push("─ OUTSTANDING RECEIVABLES ─");
   395→  lines.push(`Total: ${fmtINR(d.outstandingReceivable.total)}`);
   396→  if (d.outstandingReceivable.topClients.length > 0) {
   397→    lines.push("Top clients with due:");
   398→    d.outstandingReceivable.topClients.forEach((c) => {
   399→      lines.push(`   • ${c.name}: ${fmtINR(c.due)}`);
   400→    });
   401→  }
   402→  lines.push("");
   403→
   404→  lines.push("─ OVERDUE DISPATCHES ─");
   405→  if (d.overdueDispatches.length === 0) {
   406→    lines.push("No overdue dispatches.");
   407→  } else {
   408→    d.overdueDispatches.forEach((x) => {
   409→      lines.push(`   • ${x.poNumber} — ${x.supplierName} — ${pluralDays(x.daysOverdue)} overdue (was ${fmtDate(x.dueDate)})`);
   410→    });
   411→  }
   412→  lines.push("");
   413→
   414→  lines.push("─ PAYMENTS DUE (NEXT 7 DAYS) ─");
   415→  if (d.paymentsDueNext7Days.length === 0) {
   416→    lines.push("No payments due in the next 7 days.");
   417→  } else {
   418→    d.paymentsDueNext7Days.forEach((p) => {
   419→      lines.push(`   • ${p.billNumber} — ${p.clientName} — ${fmtINR(p.amount)} — due ${fmtDate(p.dueDate)}`);
   420→    });
   421→  }
   422→  lines.push("");
   423→
   424→  lines.push("─ BROKERAGE PENDING PAYOUT ─");
   425→  lines.push(`   Accrued (eligible, awaiting payout): ${fmtINR(d.brokeragePendingPayout.accrued)}`);
   426→  lines.push(`   Scheduled (in a payout batch):       ${fmtINR(d.brokeragePendingPayout.scheduled)}`);
   427→  lines.push(`   Total pending:                       ${fmtINR(d.brokeragePendingPayout.total)}`);
   428→  lines.push("");
   429→
   430→  lines.push("─ OPEN DISPUTES ─");
   431→  lines.push(`${d.openDisputesCount} open dispute${d.openDisputesCount === 1 ? "" : "s"}.`);
   432→  lines.push("");
   433→
   434→  lines.push("=".repeat(48));
   435→  lines.push("Generated by Garment Broker OS");
   436→  return lines.join("\n");
   437→}

   439→function escapeHtml(s: string): string {
   440→  return s
   441→    .replace(/&/g, "&amp;")
   442→    .replace(/</g, "&lt;")
   443→    .replace(/>/g, "&gt;")
   444→    .replace(/"/g, "&quot;")
   445→    .replace(/'/g, "&#39;");
   446→}

   448→function priorityBadge(p: Priority): string {
   449→  const map: Record<Priority, { bg: string; fg: string; label: string }> = {
   450→    urgent: { bg: "#fee2e2", fg: "#991b1b", label: "URGENT" },
   451→    high: { bg: "#fef3c7", fg: "#92400e", label: "HIGH" },
   452→    normal: { bg: "#d1fae5", fg: "#065f46", label: "NORMAL" },
   453→  };
   454→  const v = map[p];
   455→  return `<span style="display:inline-block;padding:2px 8px;border-radius:9999px;font-size:10px;font-weight:700;letter-spacing:0.04em;background:${v.bg};color:${v.fg};">${v.label}</span>`;
   456→}

   458→function buildHtmlDigest(d: DigestData): string {
   459→  const accent = "#059669"; // emerald-600
   460→  const accentDark = "#047857";
   461→  const subtle = "#f1f5f9";
   462→  const border = "#e2e8f0";
   463→  const muted = "#64748b";
   464→  const text = "#0f172a";
   465→
   466→  // Action items rows
   467→  const actionRows = d.actionItems.length
   468→    ? d.actionItems
   469→        .map(
   470→          (a, i) => `
   471→        <tr>
   472→          <td style="padding:10px 12px;border-bottom:1px solid ${border};vertical-align:top;width:32px;color:${muted};font-weight:600;">${i + 1}</td>
   473→          <td style="padding:10px 12px;border-bottom:1px solid ${border};vertical-align:top;">
   474→            <div style="font-weight:600;color:${text};font-size:14px;">${escapeHtml(a.title)}</div>
   475→            <div style="color:${muted};font-size:12px;margin-top:2px;">${escapeHtml(a.description)}</div>
   476→          </td>
   477→          <td style="padding:10px 12px;border-bottom:1px solid ${border};vertical-align:top;white-space:nowrap;">${priorityBadge(a.priority)}</td>
   478→          <td style="padding:10px 12px;border-bottom:1px solid ${border};vertical-align:top;white-space:nowrap;color:${muted};font-size:12px;">${escapeHtml(fmtDate(a.dueDate))}</td>
   479→        </tr>`,
   480→        )
   481→        .join("")
   482→    : `<tr><td colspan="4" style="padding:20px 12px;text-align:center;color:${muted};font-size:13px;">All caught up — no pending actions today.</td></tr>`;
   483→
   484→  // Top clients rows
   485→  const clientRows = d.outstandingReceivable.topClients.length
   486→    ? d.outstandingReceivable.topClients
   487→        .map(
   488→          (c) => `
   489→        <tr>
   490→          <td style="padding:8px 12px;border-bottom:1px solid ${border};color:${text};font-size:13px;">${escapeHtml(c.name)}</td>
   491→          <td style="padding:8px 12px;border-bottom:1px solid ${border};color:${text};font-size:13px;text-align:right;font-weight:600;white-space:nowrap;">${escapeHtml(fmtINR(c.due))}</td>
   492→        </tr>`,
   493→        )
   494→        .join("")
   495→    : `<tr><td colspan="2" style="padding:14px 12px;text-align:center;color:${muted};font-size:13px;">No outstanding receivables.</td></tr>`;
   496→
   497→  // Dispatch rows
   498→  const dispatchRows = d.overdueDispatches.length
   499→    ? d.overdueDispatches
   500→        .map(
   501→          (x) => `
   502→        <tr>

   719→// AI brief via z-ai-web-dev-sdk
   720→// Builds a concise data summary → LLM → returns { subject, body }.
   721→// Falls back to a templated brief if the LLM call fails (so the UI always
   722→// gets a usable response).
   723→// ─────────────────────────────────────────────────────────────────────────────
   724→async function buildEmailBrief(d: DigestData): Promise<{ subject: string; body: string; aiGenerated: boolean }> {
   725→  const subject = `Broker OS Daily Digest — ${d.dateLabel}`;
   726→
   727→  // Build a compact data summary for the LLM to write from.
   728→  const dataSummary = {
   729→    date: d.dateLabel,
   730→    generatedAt: d.generatedAtDisplay,
   731→    pendingActions: d.summary.pendingActions,
   732→    outstandingReceivableINR: fmtINR(d.outstandingReceivable.total),
   733→    paymentsDueNext7Days: d.summary.duePaymentsCount,
   734→    openDisputes: d.summary.openDisputes,
   735→    pendingNotifications: d.pendingNotificationsCount,
   736→    topActions: d.actionItems.slice(0, 6).map((a) => ({
   737→      priority: a.priority,
   738→      title: a.title,
   739→      description: a.description,
   740→    })),
   741→    topClientsWithDue: d.outstandingReceivable.topClients.map((c) => ({
   742→      name: c.name,
   743→      dueINR: fmtINR(c.due),
   744→    })),
   745→    overdueDispatches: d.overdueDispatches.slice(0, 4).map((x) => ({
   746→      poNumber: x.poNumber,
   747→      supplier: x.supplierName,
   748→      daysOverdue: x.daysOverdue,
   749→    })),
   750→    paymentsDue: d.paymentsDueNext7Days.slice(0, 4).map((p) => ({
   751→      billNumber: p.billNumber,
   752→      client: p.clientName,
   753→      amountINR: fmtINR(p.amount),
   754→      dueDate: fmtDate(p.dueDate),
   755→    })),
   756→    brokeragePendingPayoutINR: fmtINR(d.brokeragePendingPayout.total),
   757→    brokerageAccruedINR: fmtINR(d.brokeragePendingPayout.accrued),
   758→    brokerageScheduledINR: fmtINR(d.brokeragePendingPayout.scheduled),
   759→  };
   760→
   761→  const systemPrompt =
   762→    "You are a concise, professional assistant briefing a garment broker on their day. " +
   763→    "Summarize the key actions needed today based on the data. Be specific with names and amounts. " +
   764→    "Keep it under 200 words. Use Indian Rupee formatting (e.g. ₹1,25,000). " +
   765→    "Write 3-4 short paragraphs in a friendly but professional tone, as if briefing the broker over morning coffee. " +
   766→    "Do not use markdown headings or bullet points — just plain paragraphs. " +
   767→    "Start with a one-line greeting that includes the date.";
   768→
   769→  const userPrompt =
   770→    "Here is today's broker digest data (JSON):\n\n" +
   771→    JSON.stringify(dataSummary, null, 2) +
   772→    "\n\nWrite the morning brief now.";
   773→
   774→  try {
   775→    const zai = await ZAI.create();
   776→    const completion = await zai.chat.completions.create({
   777→      messages: [
   778→        { role: "assistant", content: systemPrompt },
   779→        { role: "user", content: userPrompt },
   780→      ],
   781→      thinking: { type: "disabled" },
   782→    });
   783→    const body =
   784→      completion?.choices?.[0]?.message?.content?.trim() ||
   785→      buildFallbackBrief(d);
   786→    return { subject, body, aiGenerated: !!completion?.choices?.[0]?.message?.content };
   787→  } catch {
   788→    return { subject, body: buildFallbackBrief(d), aiGenerated: false };
   789→  }
   790→}

   792→// Templated fallback brief — used if the LLM is unavailable or returns empty.

   794→  const parts: string[] = [];

   801→    const top = d.outstandingReceivable.topClients[0];
   802→    parts.push(
   803→      `Your top due is from ${top.name} at ${fmtINR(top.due)}. ` +
   804→        `${d.summary.duePaymentsCount} payment${d.summary.duePaymentsCount === 1 ? "" : "s"} fall due in the next 7 days — follow up promptly to keep cashflow healthy.`,
   805→    );

   807→  if (d.overdueDispatches.length > 0) {

"is" : "are"} overdue, the most overdue being ${first.poNumber} from ${first.supplierName} (${pluralDays(first.daysOverdue)} late). Chase the supplier and record the dispatch as soon as it ships.`,

   814→    `${fmtINR(d.brokeragePendingPayout.total)} in brokerage is awaiting payout ` +

   819→  return parts.join("\n\n");
   820→}

   825→export async function GET(req: NextRequest) {
   826→  const url = new URL(req.url);
   827→  const format = (url.searchParams.get("format") ?? "text").toLowerCase();
   828→  const validFormats = ["text", "html", "email"];
   829→  if (!validFormats.includes(format)) {
   830→    return NextResponse.json(
   831→      { error: `Invalid format. Use one of: ${validFormats.join(", ")}.` },
   832→      { status: 400 },
   833→    );
   834→  }
   835→
   836→  const data = await gatherDigestData();
   837→
   838→  const noStore = { "Cache-Control": "no-store, no-cache, must-revalidate" };
   839→
   840→  if (format === "text") {
   841→    return NextResponse.json(
   842→      { format: "text", text: buildTextDigest(data), stats: data.summary, generatedAt: data.generatedAt },
   843→      { headers: noStore },
   844→    );
   845→  }
   846→
   847→  if (format === "html") {
   848→    return NextResponse.json(
   849→      { format: "html", html: buildHtmlDigest(data), stats: data.summary, generatedAt: data.generatedAt },
   850→      { headers: noStore },
   851→    );
   852→  }
   853→
   854→  // format === "email" — AI brief via LLM
   855→  const brief = await buildEmailBrief(data);
   856→  return NextResponse.json(
   857→    {
   858→      format: "email",
   859→      subject: brief.subject,
   860→      body: brief.body,
   861→      aiGenerated: brief.aiGenerated,
   862→      stats: data.summary,
   863→      generatedAt: data.generatedAt,
   864→    },
   865→    { headers: noStore },
   866→  );
   867→}
