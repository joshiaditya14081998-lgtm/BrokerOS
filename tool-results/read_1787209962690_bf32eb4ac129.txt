    59→function backupAgeDays(iso: string | null): number | null {
    60→  if (!iso) return null;
    61→  const t = new Date(iso).getTime();
    62→  if (Number.isNaN(t)) return null;
    63→  return Math.floor((Date.now() - t) / (24 * 60 * 60 * 1000));
    64→}

   748→// ─────────────────────────────────────────────────────────────────────────────
   749→// SortableCard — wraps a dashboard section with dnd-kit's useSortable hook.
   750→//
   751→// In customize mode, a thin bar appears above the card content with:
   752→//   • a grip drag handle (the activator for dnd-kit)
   753→//   • the card label
   754→//   • a "Hide" button (eye-off icon) that removes the card from the layout
   755→//
   756→// Out of customize mode, the bar is hidden and a small floating grip appears
   757→// at the top-right corner of the card on hover — clicking it enters customize
   758→// mode (so the hover affordance isn't a tease). The floating grip uses
   759→// `pointer-events-none` until hovered so it never blocks clicks on underlying
   760→// card content (e.g. KPI tile drill-downs).
   761→//
   762→// While dragging, the card lifts with `scale-[1.02] shadow-xl z-10` per spec.
   763→// ─────────────────────────────────────────────────────────────────────────────
   764→function SortableCard({
   765→  id,
   766→  label,
   767→  customizeMode,
   768→  children,
   769→}: {
   770→  id: DashboardCardId;
   771→  label: string;
   772→  customizeMode: boolean;
   773→  children: React.ReactNode;
   774→}) {
   775→  const {
   776→    attributes,
   777→    listeners,
   778→    setNodeRef,
   779→    setActivatorNodeRef,
   780→    transform,
   781→    transition,
   782→    isDragging,
   783→  } = useSortable({ id });
   784→
   785→  const toggleCard = useDashboardLayout((s) => s.toggleCard);
   786→  const setCustomizeMode = useDashboardLayout((s) => s.setCustomizeMode);
   787→
   788→  const style: React.CSSProperties = {
   789→    transform: CSS.Transform.toString(transform),
   790→    transition,
   791→  };
   792→
   793→  return (
   794→    <div
   795→      ref={setNodeRef}
   796→      style={style}
   797→      className={cn(
   798→        "group/sort relative",
   799→        isDragging && "z-10 scale-[1.02] shadow-xl",
   800→      )}
   801→    >
   802→      {customizeMode ? (
   803→        <div className="mb-2 flex items-center justify-between rounded-xl border border-dashed border-emerald-500/40 bg-emerald-500/5 px-3 py-1.5">
   804→          <div className="flex min-w-0 items-center gap-2">
   805→            <button
   806→              type="button"
   807→              ref={setActivatorNodeRef}
   808→              {...attributes}
   809→              {...listeners}
   810→              className="grid size-6 shrink-0 cursor-grab place-items-center rounded-md text-muted-foreground transition-colors hover:bg-emerald-500/10 hover:text-emerald-600 active:cursor-grabbing dark:hover:text-emerald-400"
   811→              aria-label={`Drag ${label} to reorder`}
   812→            >
   813→              <GripVertical className="size-4" />
   814→            </button>
   815→            <span className="truncate text-xs font-medium text-muted-foreground">{label}</span>
   816→          </div>
   817→          <Button
   818→            type="button"
   819→            variant="ghost"
   820→            size="sm"
   821→            className="h-7 shrink-0 gap-1.5 px-2 text-[11px] text-muted-foreground hover:text-rose-600 dark:hover:text-rose-400"
   822→            onClick={() => toggleCard(id)}
   823→            aria-label={`Hide ${label} from dashboard`}
   824→          >
   825→            <EyeOff className="size-3.5" />
   826→            <span className="hidden sm:inline">Hide</span>
   827→          </Button>
   828→        </div>
   829→      ) : (
   830→        // Floating grip — visible on hover only. Clicking it enters customize
   831→        // mode so the broker can start rearranging. pointer-events-none until
   832→        // hovered so it never blocks underlying card interactions.
   833→        <button
   834→          type="button"
   835→          onClick={() => setCustomizeMode(true)}
   836→          aria-label="Customize dashboard layout"
   837→          className="absolute right-2 top-2 z-20 grid size-7 cursor-pointer place-items-center rounded-md bg-background/70 text-muted-foreground opacity-0 backdrop-blur-sm transition-all hover:text-emerald-600 group-hover/sort:opacity-100 group-hover/sort:pointer-events-auto pointer-events-none dark:hover:text-emerald-400"
   838→        >
   839→          <GripVertical className="size-3.5" />
   840→        </button>
   841→      )}
   842→      {children}
   843→    </div>
   844→  );
   845→}
