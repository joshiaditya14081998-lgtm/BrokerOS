    34→function fmtCurrency(n: number): string {
    35→  const v = Number.isFinite(n) ? n : 0;
    36→  return new Intl.NumberFormat("en-IN", {
    37→    style: "currency",
    38→    currency: "INR",
    39→    maximumFractionDigits: 0,
    40→  }).format(v);
    41→}

    43→function fmtDate(d: Date | string | null | undefined): string {
    44→  if (!d) return "—";
    45→  const date = typeof d === "string" ? new Date(d) : d;
    46→  return date.toLocaleDateString("en-IN", {
    47→    day: "2-digit",
    48→    month: "short",
    49→    year: "numeric",
    50→  });
    51→}
    52→
    53→function fmtDateTime(d: Date | string | null | undefined): string {
    54→  if (!d) return "—";
    55→  const date = typeof d === "string" ? new Date(d) : d;
    56→  return date.toLocaleString("en-IN", {
    57→    day: "2-digit",
    58→    month: "short",
    59→    year: "numeric",
    60→    hour: "2-digit",
    61→    minute: "2-digit",
    62→  });
    63→}
    64→
    65→function titleCase(s: string): string {
    66→  return s.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase());
    67→}
    68→
    69→function escapeHtml(s: string | null | undefined): string {
    70→  if (s == null) return "";
    71→  return String(s)
    72→    .replace(/&/g, "&amp;")
    73→    .replace(/</g, "&lt;")
    74→    .replace(/>/g, "&gt;")
    75→    .replace(/"/g, "&quot;")
    76→    .replace(/'/g, "&#39;");
    77→}

    79→// ── Range bounds (reused from dashboard route) ───────────────────────────────
    80→
    81→function getRangeBounds(range: Range): { start: Date | undefined; end: Date; label: string } {
    82→  const now = new Date();
    83→  if (range === "all") {
    84→    return { start: undefined, end: now, label: "All time" };
    85→  }
    86→  const y = now.getFullYear();
    87→  const m = now.getMonth();
    88→  let start: Date;
    89→  let label: string;
    90→  if (range === "month") {
    91→    start = new Date(y, m, 1);
    92→    label = start.toLocaleDateString("en-IN", { month: "long", year: "numeric" });
    93→  } else if (range === "quarter") {
    94→    const qStartMonth = Math.floor(m / 3) * 3;
    95→    start = new Date(y, qStartMonth, 1);
    96→    const qNum = Math.floor(qStartMonth / 3) + 1;
    97→    label = `Q${qNum} ${y}`;
    98→  } else {
    99→    start = new Date(y, 0, 1);
   100→    label = `Year ${y}`;
   101→  }
   102→  return { start, end: now, label };
   103→}

   388→// ── Status pill helper ───────────────────────────────────────────────────────

   390→function statusPill(status: string): string {
   391→  const s = status.toLowerCase();
   392→  let cls = "zinc";
   393→  if (["fully_paid", "fully_delivered", "delivered", "resolved", "occurred", "done", "paid", "scheduled"].includes(s)) {
   394→    cls = "emerald";
   395→  } else if (["partially_paid", "partially_delivered", "in_transit", "accrued"].includes(s)) {
   396→    cls = "amber";
   397→  } else if (["pending", "open", "no_show", "short_shipment", "defective_return", "overdue"].includes(s)) {
   398→    cls = "rose";
   399→  } else if (["closed", "rejected", "dismissed"].includes(s)) {
   400→    cls = "zinc";
   401→  }
   402→  return `<span class="pill ${cls}">${escapeHtml(titleCase(status))}</span>`;
   403→}
   404→
   405→function eligiblePill(eligible: boolean, forceEligible?: boolean): string {
   406→  if (!eligible) return `<span class="pill rose">No</span>`;
   407→  return forceEligible
   408→    ? `<span class="pill amber">Forced</span>`
   409→    : `<span class="pill emerald">Yes</span>`;
   410→}

   412→// ── Brokerage statement report ───────────────────────────────────────────────
   413→
   414→async function buildBrokerageStatement(range: Range): Promise<string> {
   415→  const { start, end, label } = getRangeBounds(range);
   416→
   417→  const brokerages = await db.brokerage.findMany({
   418→    where: start
   419→      ? {
   420→          OR: [
   421→            { eligibleAt: { gte: start, lte: end } },
   422→            { eligibleAt: null, createdAt: { gte: start, lte: end } },
   423→          ],
   424→        }
   425→      : undefined,
   426→    include: {
   427→      bill: { select: { billNumber: true, baseAmount: true, status: true, po: { select: { poNumber: true } } } },
   428→      client: { select: { name: true } },
   429→      supplier: { select: { name: true } },
   430→      payout: { select: { id: true, status: true, paidAt: true } },
   431→    },
   432→    orderBy: { eligibleAt: "asc" },
   433→  });
   434→
   435→  const payouts = await db.brokeragePayout.findMany({
   436→    where: start ? { createdAt: { gte: start, lte: end } } : undefined,
   437→    include: { client: { select: { name: true } }, brokerages: { select: { id: true, brokerageAmount: true } } },
   438→    orderBy: { createdAt: "desc" },
   439→  });
   440→
   441→  const totalBrokerage = brokerages.reduce((s, b) => s + b.brokerageAmount, 0);
   442→  const accruedTotal = brokerages
   443→    .filter((b) => b.eligible && b.payoutStatus === "accrued")
   444→    .reduce((s, b) => s + b.brokerageAmount, 0);
   445→  const scheduledTotal = brokerages
   446→    .filter((b) => b.eligible && b.payoutStatus === "scheduled")
   447→    .reduce((s, b) => s + b.brokerageAmount, 0);
   448→  const paidTotal = brokerages
   449→    .filter((b) => b.payoutStatus === "paid")
   450→    .reduce((s, b) => s + b.brokerageAmount, 0);
   451→  const pendingTotal = brokerages
   452→    .filter((b) => !b.eligible)
   453→    .reduce((s, b) => s + b.brokerageAmount, 0);
   454→  const billsCount = new Set(brokerages.map((b) => b.billId)).size;
   455→
   456→  // Group by client for the detail table (multiple sub-totals)
   457→  const byClient = new Map<string, { name: string; rows: typeof brokerages }>();
   458→  for (const b of brokerages) {
   459→    const key = b.clientId;
   460→    if (!byClient.has(key)) {
   461→      byClient.set(key, { name: b.client.name, rows: [] });
   462→    }
   463→    byClient.get(key)!.rows.push(b);
   464→  }
   465→  // Sort clients by name
   466→  const clientGroups = Array.from(byClient.entries())
   467→    .map(([id, g]) => ({ id, name: g.name, rows: g.rows }))
   468→    .sort((a, b) => a.name.localeCompare(b.name));
   469→
   470→  // Build the detail table — grouped by client with sub-totals
   471→  const detailRows: string[] = [];
   472→  let grandBase = 0;
   473→  let grandBrokerage = 0;
   474→  for (const group of clientGroups) {
   475→    detailRows.push(
   476→      `<tr class="group-header"><td colspan="9">${escapeHtml(group.name)} — ${group.rows.length} entr${group.rows.length === 1 ? "y" : "ies"}</td></tr>`,
   477→    );
   478→    let subBase = 0;
   479→    let subBrokerage = 0;
   480→    for (const b of group.rows) {
   481→      subBase += b.baseAmount;
   482→      subBrokerage += b.brokerageAmount;
   483→      grandBase += b.baseAmount;
   484→      grandBrokerage += b.brokerageAmount;
   485→      detailRows.push(
   486→        `<tr>
   487→          <td>${escapeHtml(b.bill.billNumber)}</td>
   488→          <td>${escapeHtml(b.client.name)}</td>
   489→          <td>${escapeHtml(b.supplier.name)}</td>
   490→          <td class="num">${fmtCurrency(b.baseAmount)}</td>
   491→          <td class="num">${b.commissionRate.toFixed(1)}%</td>
   492→          <td class="num emerald">${fmtCurrency(b.brokerageAmount)}</td>
   493→          <td class="center">${eligiblePill(b.eligible, b.forceEligible)}</td>
   494→          <td class="center">${statusPill(b.payoutStatus)}</td>
   495→          <td>${fmtDate(b.eligibleAt)}</td>
   496→        </tr>`,
   497→      );
   498→    }
   499→    detailRows.push(
   500→      `<tr class="sub-total">
   501→        <td colspan="3">Sub-total — ${escapeHtml(group.name)}</td>
   502→        <td class="num">${fmtCurrency(subBase)}</td>
   503→        <td></td>
   504→        <td class="num">${fmtCurrency(subBrokerage)}</td>
   505→        <td colspan="3"></td>
   506→      </tr>`,
   507→    );
   508→  }
   509→
   510→  if (brokerages.length === 0) {
   511→    detailRows.push(
   512→      `<tr><td colspan="9" style="text-align:center;padding:24px;color:#a1a1aa;">No brokerage entries in this range.</td></tr>`,
   513→    );
   514→  }
   515→
   516→  // Payout summary table
   517→  const payoutRows = payouts.length
   518→    ? payouts
   519→        .map(

  1131→export async function GET(req: NextRequest) {
  1132→  const { searchParams } = new URL(req.url);
  1133→  const typeParam = (searchParams.get("type") ?? "").toLowerCase() as ReportType;
  1134→  if (!VALID_TYPES.includes(typeParam)) {
  1135→    return NextResponse.json(
  1136→      { error: `Invalid report type. Valid: ${VALID_TYPES.join(", ")}` },
  1137→      { status: 400 },
  1138→    );
  1139→  }
  1140→
  1141→  try {
  1142→    let html: string;
  1143→    if (typeParam === "brokerage-statement") {
  1144→      const rawRange = (searchParams.get("range") ?? "all").toLowerCase() as Range;
  1145→      const range: Range = VALID_RANGES.includes(rawRange) ? rawRange : "all";
  1146→      html = await buildBrokerageStatement(range);
  1147→    } else if (typeParam === "client-ledger") {
  1148→      const clientId = searchParams.get("clientId");
  1149→      if (!clientId) {
  1150→        return NextResponse.json({ error: "clientId is required for client-ledger report" }, { status: 400 });
  1151→      }
  1152→      html = await buildClientLedger(clientId);
  1153→    } else {
  1154→      const supplierId = searchParams.get("supplierId");
  1155→      if (!supplierId) {
  1156→        return NextResponse.json({ error: "supplierId is required for supplier-summary report" }, { status: 400 });
  1157→      }
  1158→      html = await buildSupplierSummary(supplierId);
  1159→    }
  1160→
  1161→    return new NextResponse(html, {
  1162→      status: 200,
  1163→      headers: {
  1164→        "Content-Type": "text/html; charset=utf-8",
  1165→        "Cache-Control": "no-store, no-cache, must-revalidate",
  1166→        "X-Content-Type-Options": "nosniff",
  1167→      },
  1168→    });
  1169→  } catch (e) {
  1170→    const message = e instanceof Error ? e.message : "Unknown error";
  1171→    return NextResponse.json({ error: `Report failed: ${message}` }, { status: 500 });
  1172→  }
  1173→}
