   139→export function PoDetailSheet({ id, onClose }: { id: string; onClose: () => void }) {
   140→  const { data, loading } = useApi<{ po: PoDetail }>(`/api/purchase-orders/${id}`);

   148→            <SheetTitle className="text-lg">
   149→              {loading || !po ? "Loading…" : po.poNumber}

   153→                variant="outline"

   310→function DispatchesTab({ dispatches, logs }: { dispatches: Dispatch[]; logs: DispatchDateLog[] }) {
   311→  if (dispatches.length === 0 && logs.length === 0) {
   312→    return <EmptyState title="No dispatches yet" hint="Shipments will appear here once recorded." />;
   313→  }
   314→  return (
   315→    <>
   316→      {dispatches.length === 0 ? null : dispatches.map((d) => {
   317→        const items = safeParse<DispatchItem[]>(d.itemsJson, []);
   318→        return (
   319→          <div key={d.id} className="glass rounded-xl p-4">
   320→            <div className="flex items-start justify-between gap-2">
   321→              <div>
   322→                <p className="text-sm font-semibold">{formatDate(d.dispatchDate)}</p>
   323→                <p className="text-xs text-muted-foreground">
   324→                  Dispatched {d.dispatchedQty} sets · {d.photos.length} photo{d.photos.length === 1 ? "" : "s"}
   325→                </p>
   326→              </div>
   327→              <StatusChip status={d.status} />
   328→            </div>
   329→            {items.length > 0 ? (
   330→              <div className="mt-2 grid grid-cols-1 gap-1 sm:grid-cols-2">
   331→                {items.map((it, i) => (
   332→                  <div key={i} className="flex items-center justify-between rounded-lg bg-card/40 px-2 py-1 text-xs">
   333→                    <span className="truncate">
   334→                      <span className="font-medium">{it.styleName}</span>
   335→                      {it.color ? <span className="ml-1 text-muted-foreground">· {it.color}</span> : null}
   336→                    </span>
   337→                    <span className="ml-2 shrink-0 font-semibold">{it.qty}</span>
   338→                  </div>
   339→                ))}
   340→              </div>
   341→            ) : null}
   342→            {d.notes ? <p className="mt-2 text-xs text-muted-foreground">“{d.notes}”</p> : null}
   343→            {d.disputes.length > 0 ? (
   344→              <div className="mt-2 space-y-1 border-t border-border/40 pt-2">
   345→                <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Disputes</p>
   346→                {d.disputes.map((dis) => (
   347→                  <div key={dis.id} className="flex items-center justify-between text-[11px]">
   348→                    <span className="inline-flex items-center gap-1 text-muted-foreground">
   349→                      <AlertTriangle className="size-3 text-rose-500" />
   350→                      {titleCase(dis.type)} · {dis.quantityAffected} affected
   351→                    </span>
   352→                    <StatusChip status={dis.status} />
   353→                  </div>
   354→                ))}
   355→              </div>
   356→            ) : null}
   357→            <div className="mt-3">
   358→              <PhotoUpload
   359→                entityType="Dispatch"
   360→                entityId={d.id}
   361→                stage="dispatch"
   362→                label="Dispatch photos"
   363→                hint="Loading bay, bale counts, vehicle number, delivery challan."
   364→              />
   365→            </div>
   366→          </div>
   367→        );
   368→      })}
   369→
   370→      {logs.length > 0 ? (
   371→        <div className="glass rounded-xl p-4">
   372→          <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Dispatch date revisions</p>
   373→          <div className="mt-2 space-y-2">
   374→            {logs.map((l) => (
   375→              <div key={l.id} className="flex items-start gap-2 text-xs">
   376→                <Clock className="mt-0.5 size-3.5 shrink-0 text-amber-500" />
   377→                <div className="min-w-0 flex-1">
   378→                  <div className="flex flex-wrap items-center gap-1.5">
   379→                    <span className="text-muted-foreground line-through">{formatDate(l.oldDate)}</span>
   380→                    <ChevronRight className="size-3 text-muted-foreground" />
   381→                    <span className="font-medium">{formatDate(l.newDate)}</span>
   382→                    <span className="text-muted-foreground">· {formatDateTime(l.createdAt)}</span>

   384→                  {l.reason ? <p className="mt-0.5 text-muted-foreground">{l.reason}</p> : null}
   385→                </div>
   386→              </div>
   387→            ))}
   388→          </div>
   389→        </div>
   390→      ) : null}
   391→    </>
   392→  );
   393→}

   507→    <div className="glass rounded-xl p-4">
   508→      <div className="mb-2 flex items-center gap-1.5">
   509→        <BadgePercent className="size-4 text-emerald-500" />
   510→        <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Brokerage</p>
   511→      </div>
   512→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   513→        <KV label="Commission" value={`${brokerage.commissionRate}%`} />
   514→        <KV label="Base (excl GST)" value={fmtCurrency(brokerage.baseAmount)} />
   515→        <KV label="Brokerage amt" value={fmtCurrency(brokerage.brokerageAmount)} tone="emerald" />
   516→        <KV label="Eligible" value={brokerage.eligible ? "Yes" : "No"} tone={brokerage.eligible ? "emerald" : "default"} />
   517→        <KV label="Payout status" value={titleCase(brokerage.payoutStatus)} tone={brokerage.payoutStatus === "paid" ? "emerald" : "default"} />
   518→        {brokerage.eligibleAt ? <KV label="Eligible since" value={formatDate(brokerage.eligibleAt)} /> : null}
   519→      </div>
   520→      {brokerage.forceEligible && brokerage.forceReason ? (
   521→        <p className="mt-2 rounded-lg bg-amber-500/10 px-3 py-1.5 text-[11px] text-amber-700 dark:text-amber-300">
   522→          Force-eligible: {brokerage.forceReason}
   523→        </p>
   524→      ) : null}
   525→      {brokerage.payout ? (
   526→        <p className="mt-2 text-[11px] text-muted-foreground">
   527→          Payout: {titleCase(brokerage.payout.status)} · {titleCase(brokerage.payout.cadence)}
   528→          {brokerage.payout.paidAt ? ` · paid ${formatDate(brokerage.payout.paidAt)}` : ""}
   529→        </p>
   530→      ) : null}
   531→    </div>
   532→  );
   533→}

   535→type TimelineEntry = {
   536→  id: string;
   537→  date: string;
   538→  kind: "log" | "payment" | "dispute" | "bill";
   539→  title: string;
   540→  detail: string | null;
   541→};

   543→function TimelineTab({ po }: { po: PoDetail }) {
   544→  const { format: fmtCurrency } = useCurrencyFormat();
   545→  const entries: TimelineEntry[] = React.useMemo(() => {
   546→    const list: TimelineEntry[] = [];
   547→    for (const l of po.dispatchDateLogs) {
   548→      list.push({
   549→        id: l.id, date: l.createdAt, kind: "log",
   550→        title: `Dispatch date revised → ${formatDate(l.newDate)}`,
   551→        detail: l.reason || (l.oldDate ? `from ${formatDate(l.oldDate)}` : null),
   552→      });
   553→    }
   554→    if (po.bill) {
   555→      list.push({
   556→        id: `bill-${po.bill.id}`, date: po.bill.createdAt, kind: "bill",
   557→        title: `Bill ${po.bill.billNumber} generated`,
   558→        detail: `Final: ${fmtCurrency(po.bill.finalAmount)} · status: ${titleCase(po.bill.status)}`,
   559→      });
   560→      for (const p of po.bill.payments) {
   561→        list.push({
   562→          id: `pay-${p.id}`, date: p.date, kind: "payment",
   563→          title: `Payment · ${titleCase(p.mode)}`,
   564→          detail: `${fmtCurrency(p.amount)}${p.reference ? ` · ref ${p.reference}` : ""}`,
   565→        });
   566→      }
   567→      if (po.bill.brokerage) {
   568→        const b = po.bill.brokerage;
   569→        list.push({
   570→          id: `brok-${b.id}`, date: b.eligibleAt ?? po.bill.createdAt, kind: "bill",
   571→          title: `Brokerage ${b.eligible ? "eligible" : "accrued"}`,
   572→          detail: `${fmtCurrency(b.brokerageAmount)} · ${titleCase(b.payoutStatus)}`,
   573→        });
   574→      }
   575→    }
   576→    for (const d of po.disputes) {
   577→      list.push({
   578→        id: `disp-${d.id}`, date: d.createdAt, kind: "dispute",
   579→        title: `Dispute · ${titleCase(d.type)}`,
   580→        detail: d.description || `${d.quantityAffected} units · ${fmtCurrency(d.valueAffected)}`,
   581→      });
   582→    }
   583→    return list.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   584→  }, [po, fmtCurrency]);
   585→
   586→  if (entries.length === 0) {
   587→    return (
   588→      <GlassCard className="p-8">
   589→        <EmptyState title="No timeline events" hint="Dispatches, payments, and disputes will appear here." icon={<Clock className="size-5" />} />
   590→      </GlassCard>
   591→    );
   592→  }
   593→
   594→  return (
   595→    <div className="glass rounded-xl p-4">
   596→      <ol className="relative space-y-4 border-l border-border/60 pl-4">
   597→        {entries.map((e) => (
   598→          <li key={e.id} className="relative">
   599→            <span className={`absolute -left-[1.41rem] top-1 grid size-3.5 place-items-center rounded-full border-2 border-background ${kindClass(e.kind)}`}>
   600→              <span className="size-1.5 rounded-full bg-current opacity-80" />
   601→            </span>
   602→            <div className="flex flex-wrap items-center gap-1.5">
   603→              <span className="text-sm font-medium">{e.title}</span>
   604→              <span className="text-[10px] uppercase tracking-wider text-muted-foreground">
   605→                {kindLabel(e.kind)}
   606→              </span>
   607→            </div>
   608→            <p className="text-xs text-muted-foreground">{formatDateTime(e.date)}</p>
   609→            {e.detail ? <p className="mt-0.5 text-xs text-foreground/80">{e.detail}</p> : null}
   610→          </li>
   611→        ))}
   612→      </ol>
   613→    </div>
   614→  );
   615→}
   616→
   617→function kindClass(kind: TimelineEntry["kind"]): string {
   618→  switch (kind) {
   619→    case "payment": return "bg-emerald-500/20 text-emerald-600 dark:text-emerald-400";
   620→    case "log": return "bg-amber-500/20 text-amber-600 dark:text-amber-400";
   621→    case "dispute": return "bg-rose-500/20 text-rose-600 dark:text-rose-400";
   622→    default: return "bg-teal-500/20 text-teal-600 dark:text-teal-400";
   623→  }
   624→}
   625→
   626→function kindLabel(kind: TimelineEntry["kind"]): string {
   627→  switch (kind) {
   628→    case "payment": return "Payment";
   629→    case "log": return "Revision";
   630→    case "dispute": return "Dispute";
   631→    default: return "Billing";
   632→  }
   633→}

   635→function Stat({
   636→  icon, label, value, tone = "default",
   637→}: {
   638→  icon: React.ReactNode;
   639→  label: string;
   640→  value: string;
   641→  tone?: "default" | "emerald" | "amber";
   642→}) {
   643→  const toneClass = tone === "emerald"
   644→    ? "text-emerald-600 dark:text-emerald-400"
   645→    : tone === "amber"
   646→      ? "text-amber-600 dark:text-amber-400"
   647→      : "text-foreground";
   648→  return (
   649→    <div className="rounded-xl border border-border/40 bg-card/40 p-3">
   650→      <div className="flex items-center gap-1.5 text-muted-foreground">
   651→        {icon}
   652→        <span className="text-[10px] uppercase tracking-wider">{label}</span>
   653→      </div>
   654→      <p className={`kpi-num mt-1 text-base font-semibold ${toneClass}`}>{value}</p>
   655→    </div>
   656→  );
   657→}

   661→}: {

   663→  value: string;

   665→}) {

   667→    ? "text-emerald-600 dark:text-emerald-400"
   668→    : tone === "amber"
   669→      ? "text-amber-600 dark:text-amber-400"
   670→      : "text-foreground";
   671→  return (
   672→    <div>
   673→      <p className="text-[10px] uppercase tracking-wider text-muted-foreground">{label}</p>
   674→      <p className={`text-sm font-medium ${toneClass}`}>{value}</p>
   675→    </div>
   676→  );
