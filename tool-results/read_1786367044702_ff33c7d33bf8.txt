   152→function formatBytes(bytes: number): string {
   153→  if (bytes < 1024) return `${bytes} B`;
   154→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   155→  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
   156→}
   157→
   158→function formatRelative(iso: string | null): string {
   159→  if (!iso) return "Never";
   160→  const t = new Date(iso).getTime();
   161→  if (Number.isNaN(t)) return "Never";
   162→  const diff = Date.now() - t;
   163→  const sec = Math.floor(diff / 1000);
   164→  if (sec < 60) return "Just now";
   165→  const min = Math.floor(sec / 60);
   166→  if (min < 60) return `${min} min ago`;
   167→  const hr = Math.floor(min / 60);
   168→  if (hr < 24) return `${hr} hr ago`;
   169→  const day = Math.floor(hr / 24);
   170→  if (day < 30) return `${day} day${day === 1 ? "" : "s"} ago`;
   171→  const mon = Math.floor(day / 30);
   172→  if (mon < 12) return `${mon} month${mon === 1 ? "" : "s"} ago`;
   173→  return new Date(iso).toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   174→}
   175→
   176→function formatDateTime(iso: string | null | undefined): string {
   177→  if (!iso) return "Unknown";
   178→  const d = new Date(iso);
   179→  if (Number.isNaN(d.getTime())) return "Unknown";
   180→  return d.toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" });
   181→}

   183→function parseBackupFile(text: string, fileName: string, fileSize: number): { ok: true; backup: ParsedBackup } | { ok: false; error: string } {
   184→  let json: unknown;
   185→  try {
   186→    json = JSON.parse(text);
   187→  } catch {
   188→    return { ok: false, error: `"${fileName}" is not valid JSON.` };
   189→  }
   190→  if (typeof json !== "object" || json === null || Array.isArray(json)) {
   191→    return { ok: false, error: "Backup file must be a JSON object." };
   192→  }
   193→  const obj = json as Record<string, unknown>;
   194→  if (obj.version !== 1) {
   195→    return { ok: false, error: `Unsupported backup version (expected 1, got ${String(obj.version)}).` };
   196→  }
   197→  if (typeof obj.tables !== "object" || obj.tables === null || Array.isArray(obj.tables)) {
   198→    return { ok: false, error: "Backup file is missing the 'tables' object." };
   199→  }
   200→  const tables = obj.tables as Record<string, unknown>;
   201→  for (const t of BACKUP_TABLE_LABELS) {
   202→    if (!Array.isArray(tables[t.key])) {
   203→      return { ok: false, error: `Backup file is missing the "${t.key}" table.` };
   204→    }
   205→  }
   206→  return {
   207→    ok: true,
   208→    backup: {
   209→      version: 1,
   210→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : undefined,
   211→      tables: tables as Record<string, unknown[]>,
   212→      bytes: fileSize,
   213→    },
   214→  };
   215→}

   487→    } catch (e) {
   488→      toast.error(e instanceof Error ? e.message : "Restore failed");

   490→    }
